-	script	::CardSeparation_mal	-1,{
	disable_items;
	if (checkweight(1201,1) == 0) {
		mes "Vous avez trop de types d'objets. Essayons de continuer après avoir réduit ces objets.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "Impossible de continuer car vous avez trop d'objets lourds. Essayons de continuer après avoir réduit le poids.";
		close;
	}
	if (strnpcinfo(1) == "Jeremy") {
		set .@Jeremy,1;
		set .@n$, "[Jeremy]";
		setarray .@equip_name$[0],  "Armor",  "Shoes",  "Garment", "Upper Hat";
		setarray .@equip_slot[0], EQI_ARMOR,EQI_SHOES,EQI_GARMENT,EQI_HEAD_TOP;

		mes .@n$;
		mes "Longtemps sans nous voir~";
		mes "J'ai appris une nouvelle compétence qui sépare les cartes des Armures, Chaussures, Vêtements et Casques. Voulez-vous l'essayer ?";
	} else {
		set .@Jeremy,0;
		set .@n$, "[Richard]";
		setarray .@equip_name$[0], "Left hand", "Right hand";
		setarray .@equip_slot[0],   EQI_HAND_L,   EQI_HAND_R;

		mes .@n$;
		mes "Le malin Jeremy ne veut pas toucher aux armes et boucliers à cause de la sélection de plusieurs cartes qui pourraient être endommagées. C'est pourquoi j'ai préparé une compétence de séparation de cartes pour les armes et boucliers...";
	}
	next;
	mes .@n$;
	mes "Généralement, le coût est de 1 000 000 de Zénies. Pendant la séparation des cartes, vous pouvez utiliser ^990000des objets spéciaux^000000 qui réduisent le taux de destruction du matériel ou des cartes. Nous ne facturons pas de zénies supplémentaires pour cela.";
	next;
	mes .@n$;
	mes "Il y a une possibilité qu'ils soient détruits même en utilisant un objet spécial. De plus, ^ff0000le niveau d'affinement pourrait être perdu^000000. Avez-vous du matériel à séparer ?";
	next;

	for(set .@i,0; .@i<getarraysize(.@equip_slot); set .@i,.@i+1) {
		if (getequipisequiped(.@equip_slot[.@i]))
			set .@menu$, .@menu$+getequipname(.@equip_slot[.@i])+":";
		else
			set .@menu$, .@menu$+"^999999"+.@equip_name$[.@i]+" (empty)^000000:";
	}

	set .@i, select("Stop the work:"+((.@Jeremy)?"How is it possible?":"")+":"+.@menu$);
	switch(.@i) {
	case 1:
		mes .@n$;
		mes "Quand vous aurez besoin de ce service, revenez me voir ici.";
		close;
	case 2:
		mes .@n$;
		mes "Vous vous demandez ce qui est si spécial. Eh bien, je déteste ne donner qu'un guide, alors laissez-moi vous raconter l'histoire...";
		next;
		mes "^000099Jeremy étire ses épaules et ses mains. Il attend peut-être quelqu'un avec qui parler.^000000";
		next;
		mes .@n$;
		mes "Saviez-vous que la spécialité de Malangdo est les conserves ?";
		next;
		select("I knew that well. Is it that limited?");
		mes .@n$;
		mes "Hehe... Tout le monde l'adore. Mais il y avait quelques problèmes auparavant.";
		next;
		select("Problems? Is there any faulty fish?");
		mes .@n$;
		mes "Non, le poisson ne pose aucun problème. Le problème c'est la grande quantité d'huile de poisson produite après le traitement. C'est un tel déchet industriel.";
		next;
		mes .@n$;
		mes "Cependant, après un processus de raffinage revitalisant, cette huile de poisson est devenue précieuse pour l'entretien du matériel ancien et les lubrifiants industriels.";
		next;
		mes .@n$;
		mes "De plus, cet huile est si utile pour séparer les reliques des armes équipées que nous ne pouvons pas acheter le Lubrifiant Premium et le Lubrifiant Ordinaire avec des Zénies.";
		next;
		mes .@n$;
		mes "Eh bien, n'inquiétez-vous pas de l'argent. Le Lubrifiant Premium est sûrement cher. Si vous payez quelques zénies, je peux vous donner un lubrifiant moins cher.";
		next;
		mes .@n$;
		mes "Je ne suis pas sûr du taux de réussite concernant la qualité. De toute façon, c'est si bon marché, non ?";
		close;
	default:
		set .@equip_num, .@equip_slot[.@i-3];
		if (!getequipisequiped(.@equip_num)) {
			mes .@n$;
			if (.@Jeremy)
				mes "Il n'y a rien dans cette partie ?";
			else
				mes "Il n'y a rien sur cette partie ?";
			close;
		}
		break;
	}

	setarray .@check_equip_card[0], getequipcardid(.@equip_num,0),getequipcardid(.@equip_num,1),getequipcardid(.@equip_num,2),getequipcardid(.@equip_num,3);
	setarray .@equip_card[0], .@check_equip_card[0], .@check_equip_card[1], .@check_equip_card[2], .@check_equip_card[3];
	set .@mvp_list$,
		"|4408|4128|4456|4168|4142"+  //Gloom_Under_Night_Card, Golden_Bug_Card, Nidhogg_Shadow_Card, Dark_Lord_Card, Doppelganger_Card
		"|4134|4137|4386|4407|4357"+  //Dracula_Card, Drake_Card, Detale_Card, Randgris_Card, B_Seyren_Card
		"|4146|4132|4147|4372|4145"+  //Maya_Card, Mistress_Card, Baphomet_Card, Bacsojin_Card, Berzebub_Card
		"|4374|4352|4367|4236|4425"+  //Apocalips_H_Card, B_Ygnizem_Card, B_Shecil_Card, Amon_Ra_Card, Atroce_Card
		"|4359|4123|4144|4135|4143"+  //B_Eremes_Card, Eddga_Card, Osiris_Card, Orc_Load_Card, Orc_Hero_Card
		"|4263|4131|4430|4276|4419"+  //Incant_Samurai_Card, Moonlight_Flower_Card, Ifrit_Card, Lord_Of_Death_Card, Ktullanux_Card
		"|4403|4399|4376|4441|4302"+  //Kiel_Card, Thanatos_Card, Lady_Tanee_Card, Fallen_Bishop_Card, Tao_Gunka_Card
		"|4305|4148|4318|4121|4365"+  //Turtle_General_Card, Pharaoh_Card, Knight_Windstorm_Card, Phreeoni_Card, B_Katrinn_Card
		"|4363|4324|4361|4330|4342|"; //B_Magaleta_Card, Garm_Card, B_Harword_Card, Dark_Snake_Lord_Card, Rsx_0806_Card

	if (.@Jeremy) {
		for ( .@i = 0; .@i < MAX_SLOTS; .@i++ ) {
			if (getiteminfo(.@equip_card[.@i], ITEMINFO_SUBTYPE) == CARD_ENCHANT)
				.@equip_card[.@i] = 0;// Armor Enchant System
		}
		if (!getarraysize(.@equip_card)) {
			mes .@n$;
			mes "La carte n'est pas équipée. Voulez-vous vérifier à nouveau ?";
			close;
		}
		if ((.@equip_card[0] && compare(.@mvp_list$,"|"+.@equip_card[0]+"|")) ||
			(.@equip_card[1] && compare(.@mvp_list$,"|"+.@equip_card[1]+"|")) ||
			(.@equip_card[2] && compare(.@mvp_list$,"|"+.@equip_card[2]+"|")) ||
			(.@equip_card[3] && compare(.@mvp_list$,"|"+.@equip_card[3]+"|")))
			set .@boss_chk,1;
	} else {
		// Official "Richard" script uses a hardcoded list including every possible item.
		//if (!getequipisequiped(.@equip_num)) {
		//	mes "[Richard]";
		//	mes "Désolé, nous n'offrons pas encore cet équipement.";
		//	close;
		//}

		mes "[Richard]";
		mes "Quelle embase souhaitez-vous désolidariser ? À partir de l'embase de gauche, elles sont numérotées 1, 2, 3, 4.";
		next;
		set .@menu$,"";
		for ( .@i = 0; .@i < MAX_SLOTS; .@i++ ) {
			if (.@equip_card[.@i] && getiteminfo(.@equip_card[.@i], ITEMINFO_SUBTYPE) == CARD_NORMAL) // Armor Enchant System
				.@menu$ = .@menu$ + "Socket " + (.@i+1) + " - " + getitemname(.@equip_card[.@i])+":";
			else {
				.@menu$ = .@menu$ + "^777777Socket " + (.@i+1) + " - No card^000000:";
			}
		}
		set .@i, select("Stop the work:"+.@menu$);
		switch(.@i) {
		case 1:
			mes .@n$;
			mes "Quand vous aurez besoin d'un travail, veuillez venir me voir.";
			close;
		default:
			set .@slot, .@i-2;
			if (.@equip_card[.@slot] == 0 || getiteminfo(.@equip_card[.@slot], ITEMINFO_SUBTYPE) == CARD_ENCHANT) {
				mes .@n$;
				mes "Cette embase n'est équipée d'aucune carte. Pourquoi ne pas vérifier à nouveau ?";
				close;
			}
			break;
		}
		if (compare(.@mvp_list$,"|"+.@equip_card[.@slot]+"|"))
			set .@boss_chk,1;
	}
	if (.@boss_chk == 0) {
		mes .@n$;
		if (.@Jeremy)
			mes "À l'exception des cartes, ^ff0000tous les effets enchantés disparaîtront.^000000 Si vous êtes d'accord avec cela, veuillez choisir le type de travail :";
		else
			mes "Veuillez choisir la taxe de travail.";
		next;
		set .@menu$,
			"Next time...:"+
			((Zeny >= 1000000)?"Use 1,000,000z (Do not use special item):":"^999999Use 1,000,000z (Insufficient)^000000:")+
			((countitem(6441))?"Use Premium Lubricant:":"^999999Premium Lubricant (Insufficient)^000000:")+
			((countitem(6440))?"Use Ordinary Lubricant":"^999999Ordinary Lubricant (Insufficient)^000000");
		switch(select(.@menu$)) {
		case 1:
			mes .@n$;
			mes "Quand vous aurez besoin de ce service, revenez me voir ici.";
			close;
		case 2:
			if (Zeny < 1000000) {
				mes .@n$;
				mes "Vous n'avez pas assez de zeny. Veuillez revenir avec des frais suffisants.";
				close;
			}
			mes .@n$;
			mes "C'est un équipement assez ancien. Il y a une forte probabilité de destruction des cartes ou de l'équipement lors du travail. Êtes-vous sûr de vouloir continuer ?";
			next;
			if(select("Next time...:Continue") == 1) {
				mes .@n$;
				mes "Quand vous aurez besoin de ce service, revenez me voir ici.";
				close;
			}
			set .@sf_c_num,150;
			set .@sf_r_num,150;
			set .@sf_w_num,150;
			set Zeny, Zeny - 1000000;
			break;
		case 3:
			if (countitem(6441) == 0) {
				mes .@n$;
				mes "Vous n'avez pas d'Huile Premium.";
				close;
			}
			mes .@n$;
			mes "Si vous utilisez l'Huile Premium, le taux de destruction sera fortement réduit, mais je ne peux pas vous garantir 100 %. Êtes-vous sûr de vouloir continuer ?";
			next;
			if(select("Next time...:Continue") == 1) {
				mes .@n$;
				mes "Quand vous aurez besoin de ce service, revenez me voir ici.";
				close;
			}
			set .@sf_c_num,75;
			set .@sf_r_num,75;
			set .@sf_w_num,75;
			delitem 6441,1; //High_RankLubricant
			break;
		case 4:
			if (countitem(6440) == 0) {
				mes .@n$;
				mes "Vous n'avez pas d'Huile Ordinaire.";
				close;
			}
			mes .@n$;
			mes "Si vous utilisez l'Huile Ordinaire, le taux de destruction sera fortement réduit, mais je ne peux pas vous garantir 100 %. Êtes-vous sûr de vouloir continuer ?";
			next;
			if(select("Next time...:Continue") == 1) {
				mes .@n$;
				mes "Quand vous aurez besoin de ce service, revenez me voir ici.";
				close;
			}
			set .@sf_c_num,75;
			set .@sf_r_num,150;
			set .@sf_w_num,150;
			delitem 6440,1; //General_Lubricant
			break;
		}
	} else if (.@boss_chk == 1) {
		mes .@n$;
		mes "Cet équipement contient une précieuse carte MVP. Cette carte ne peut être désolidarisée avec l'huile. Si vous apportez le super détergent ^0000ffSillit Pong^000000, je pourrai effectuer le travail.";
		next;
		switch(select("Next time...:I have a Sillit Pong.")) {
		case 1:
			mes .@n$;
			mes "Quand vous aurez besoin de ce service, revenez me voir ici.";
			close;
		case 2:
			if (countitem(6443) == 0) {
				mes .@n$;
				mes "Vous n'avez pas de Sillit Pong.";
				close;
			}
			break;
		}
		mes .@n$;
		if (.@Jeremy) {
			mes "À l'exception des cartes, ^ff0000tous les effets enchantés disparaîtront.^000000 Si vous êtes d'accord avec cela, veuillez choisir le type de travail :";
			set .@menu$,"Alright, let's do it!";
		} else {
			mes "Puis-je continuer ?";
			set .@menu$,"I got it. Just do it quickly!";
		}
		next;
		switch(select("Next time...:"+.@menu$)) {
		case 1:
			mes .@n$;
			mes "Quand vous aurez besoin de ce service, revenez me voir ici.";
			close;
		case 2:
			set .@sf_c_num,60;
			set .@sf_r_num,60;
			set .@sf_w_num,60;
			delitem 6443,1; //Sillit_Pong_Bottle
			break;
		}
	}

	set .@equip_id, getequipid(.@equip_num);
	set .@equip_refine, getequiprefinerycnt(.@equip_num);

	// anti-hack
	if (callfunc("F_IsEquipCardHack", .@equip_num, .@check_equip_card[0], .@check_equip_card[1], .@check_equip_card[2], .@check_equip_card[3]))
		close;

	delequip .@equip_num;

	// Chance of retaining refine level.
	if (rand(1,.@sf_r_num) >= 61)
		set .@equip_refine,0;

	if (.@Jeremy) {
		// Chance of retaining equipment.
		if (rand(1,.@sf_w_num) < 61) {
			set .@equip_safe,1;
			getitem2 .@equip_id,1,1,.@equip_refine,0,0,0,0,0;
		}

		// Chance of retaining cards.
		for(set .@i,0; .@i<4; set .@i,.@i+1) {
			if (.@equip_card[.@i]) {
				if (rand(1,.@sf_c_num) < 61)
					getitem .@equip_card[.@i],1;
				else
					set .@card_break,1;
			}
		}
	} else {
		set .@card, .@equip_card[.@slot];
		set .@check_equip_card[.@slot],0;

		// Chance of retaining equipment (all enchantments are preserved).
		if (rand(1,.@sf_w_num) < 61) {
			set .@equip_safe,1;
			getitem2 .@equip_id,1,1,.@equip_refine,0,.@check_equip_card[0],.@check_equip_card[1],.@check_equip_card[2],.@check_equip_card[3];
		}

		// Chance of retaining card.
		if (rand(1,.@sf_c_num) < 61)
			getitem .@card,1;
		else
			set .@card_break,1;
	}

	// Display corresponding effect.
	if (!.@equip_safe && .@card_break)
		specialeffect2 EF_LORD;
	else if (.@equip_safe && .@card_break)
		specialeffect2 EF_SUI_EXPLOSION;
	else if (!.@equip_safe && !.@card_break)
		specialeffect2 EF_FIREPILLAR;
	else
		specialeffect2 EF_MAXPOWER;

	// Output results.
	mes "-- Résultat du décalage de carte --";
	if (.@equip_safe) {
		mes "Aucune fissure n'est apparue lors du processus de séparation des cartes.";
		mes "^0000FFLa séparation de l'équipement est normale.^000000";
	} else {
		mes "Une fissure s'est produite lors du processus de séparation des cartes.";
		mes "L'équipement a été endommagé. ^ff0000Irréparable.^000000";
	}
	mes "-------------------";
	if (!.@card_break) {
		mes "Aucune érosion de surface n'est apparue lors du processus de séparation des cartes.";
		mes "^0000ffLa séparation des cartes a réussi.^000000";
	} else {
		mes "Une érosion de surface s'est produite lors du processus de séparation des cartes.";
		mes "La carte a été endommagée. ^ff0000Irréparable.^000000";
	}
	next;
	mes .@n$;
	mes "Ceci est tout pour les résultats de la séparation des cartes. Veuillez revenir.";
	close;
}
malangdo,215,166,4	duplicate(CardSeparation_mal)	Jeremy#pa0829	553	// Armors :: mal_yong
