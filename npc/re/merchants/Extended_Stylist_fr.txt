alberta_in,55,142,7	script	Stylist#alberta	4_F_02,{
	mes "[Sweet Veronica]";
	mes "Bienvenue au Salon de Beauté Veronica, renaissant comme styliste de classe mondiale. Hhhi.";
	next;
	switch(select("Change your style:End conversation")) {
	case 1:
		close2;
		openstylist();
		end;
	case 2:
		next;
		mes "[Sweet Veronica]";
		mes "Chacun a le droit d'être beau... J'espère que vous trouverez un style qui vous convient.";
		close;
	}
}
