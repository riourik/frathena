	}
	
	else if (Memorial08 == 8) {

		mes "'Cela doit être ce dont Lauds parlait.'";
		mes "'Essayons d'enlever la poussière avec le torchon.'";
		next;
		set Memorial08,9;
		getexp 93750,43750;
		goto L_CleanPlaque;
	}
	
	else if (Memorial08 >= 9) {
		goto L_CleanPlaque;
	}
	
	L_CleanPlaque:
		mes "- Vous voyez un message provenant de la plaque nettoyée. -";
		next;
		mes "- Bien que aucune pierre sculptée ne soit élevée à leur mémoire, -";
		mes "- ni gravés sur une pierre pour témoigner de leurs actions,-";
		mes "- pourtant leur souvenir sera aussi durable que le pays qu'ils ont honoré. -";
		mes "- Daniel Webster -";
		if (Memorial08 == 9) {
			getexp 93750,43750;
		}
		next;
		mes "- Il y a un autre message. -";
		next;
		mes "- Je n'ai peut-être pas suivi la voie que j'avais l'intention de suivre, -";
		mes "- mais je pense avoir fini là où il me fallait être. -";
		mes "- Douglas Adams. -";
		if (Memorial08 == 9) { 
			getexp 93750,43750;
		}
		next;
		mes "- C'est le dernier message. -";
		next;
		mes "-Le véritable héros est remarquablement sobre, extrêmement sans déclamation.-";
		mes "-Ce n'est pas l'envie de surpasser les autres à tout prix,-";
		mes "- mais l'envie de servir les autres à tout prix. -";
		mes "- Arthur Ashe -";
		if (Memorial08 == 9) { 
			getexp 93750,43750;
			set Memorial08,10;
		}
		close;
}
