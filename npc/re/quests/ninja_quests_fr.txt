que_ng,21,72,6	script	Master Armor Craftsman	4_M_RASWORD,{
	mes "[Gyo]";
	mes "Comment définiriez-vous la victoire ?";
	mes "Abattre l'ennemi...";
	mes "Cependant, le plus important est...";
	mes "Ne jamais tomber soi-même !";
	next;
	mes "[Gyo]";
	mes "Pour cela, vous devez mieux vous protéger.";
	mes "Vous ne pouvez pas protéger les 'autres' si vous entrave votre propre protection.";
	next;
	if(select("Combine armor.:End conversation.") == 2) {
		mes "[Gyo]";
		mes "La prudence est aussi une vertu importante en combat.";
		close;
	}
	switch(select("Wolf Armguard[1]:Crescent Armguard[1]:Ninja Scale Armor[1]:Tenebris Latitantes[1]:Quit.")) {
	case 1:
		mes "[Gyo]";
		mes "Wolf Armguard... Une pièce d'armure bestiale.";
		next;
		callsub L_Select,2172;
		if (countitem(2171) > 0 && countitem(6252) > 9) {  //custom translation
			delitem 2171,1; //Fox_Armguard
			delitem 6252,10; //Wolf_Blood
			getitem 2172,1; //Wolf_Armguard
			mes "[Gyo]";
			mes "Sentez-vous la nature du loup ?";
			mes "À la fois sauvage et charmant...";
			close;
		}
		break;
	case 2:
		mes "[Gyo]";
		mes "Crescent Armguard... Une chose de beauté et de force... Sans en être moindre.";
		next;
		callsub L_Select,2173;
		if (countitem(2171) > 0 && countitem(7321) > 99) {  //custom translation
			delitem 2171,1; //Fox_Armguard
			delitem 7321,100; //Fragment_Of_Crystal
			getitem 2173,1; //Crescent_Armguard
			mes "[Gyo]";
			mes "Une telle beauté...";
			mes "Comme si tout le corps déborde de magie.";
			close;
		}
		break;
	case 3:
		mes "[Gyo]";
		mes "Armure d'échelle ninja... On dit qu'un ninja légendaire l'a portée.";
		next;
		callsub L_Select,15054;
		if (countitem(7451) > 29 && countitem(7562) > 29 && countitem(6091) > 29) {  //custom translation
			delitem 7451,30; //Scale_Of_Red_Dragon
			delitem 7562,30; //Ice_Scale
			delitem 6091,30; //Dark_Red_Scale
			getitem 15054,1; //Ninja_Scale_Armor
			mes "[Gyo]";
			mes "C'est vraiment une armure légendaire...";
			mes "Même si je ne crois pas aux légendes...";
			close;
		}
		break;
	case 4:
		mes "[Gyo]";
		mes "Armure du Roi des Ombres... Elle a été forgée pour protéger les autres.";
		next;
		callsub L_Select,15055;
		if (countitem(15056) > 0 && countitem(6089) > 9) {  //custom translation
			delitem 15056,1; //Special_Ninja_Suit_
			delitem 6089,10; //Piece_Of_Darkness
			getitem 15055,1; //Tenebris_Latitantes
			mes "[Gyo]";
			mes "Qu'en pensez-vous ?";
			mes "L'armure n'a-t-elle pas l'air d'en comporter deux ?";
			close;
		}
		break;
	case 5:
		close;
	}
	mes "[Gyo]";
	mes "Vous n'avez pas assez de matériaux.";
	mes "Regardez à nouveau...";
	close;

L_Select:
	.@string1$ = "Check Ingredients.";
	.@string2$ = "Check Stats.";
	while(1) {
		switch(select(.@string1$,.@string2$,"Combine.","Quit.")) {
		case 1:
			.@string1$ = "";
			callsub L_Ingredients, getarg(0);
			break;
		case 2:
			.@string2$ = "";
			callsub L_Stats, getarg(0);
			break;
		case 3:
			callsub L_Combine;
			return;
		case 4:
			mes "[Gyo]";
			mes "Il semble que vous ayez besoin de plus de temps pour décider.";
			close;
		}
	}

L_Ingredients:
	mes "[Gyo]";
	switch (getarg(0)) {
	case 2172:	//Wolf_Armguard
		mes "Pour fabriquer un ^ff0000Brasérier de Loup, il faut 1 Brasérier de Renard et 10 Sang de Loup^000000.";
		mes "Ne sentez-vous pas cela ? Le battement du sang de louve...";
		break;
	case 2173:	//Crescent_Armguard
		mes "Pour fabriquer un ^ff0000Brasérier Croissant, il faut 1 Brasérier de Renard et 100 Fragments de Cristal^000000.";
		mes "Un croissant aussi beau et clair qu'un cristal sera gravé dessus.";
		break;
	case 15054:	//Ninja_Scale_Armor
		mes "Pour fabriquer une Armure d'Écailles de Ninja ^ff0000, vous avez besoin de 30 Écailles de Dragon de Feu, de 30 Écailles de Glace et de 30 Morceaux d'Écaille Rouge Sombre^000000.";
		mes "Vous pouvez sentir ses pouvoirs légendaires.";
		break;
	case 15055:	//Tenebris_Latitantes
		mes "Pour fabriquer une Armure du Roi des Ombres ^ff0000, vous aurez besoin d'un Costume de Ninja Spécial avec un slot et de 10 Morceaux Sombres^000000.";
		mes "Avec cette obscurité, elle efface l'existence même...";
		break;
	}
	next;
	return;

L_Stats:
	mes "[" + getitemname(getarg(0)) + "]";
	switch (getarg(0)) {
	case 2172:	//Wolf_Armguard
		mes "Chance d'avoir pendant 5 sec. ATK+100, FLEE -50 lors d'une attaque au corps à corps.";
		mes "Type de Bouclier, Défense 45, Slot 1";
		mes "Niveau requis 100, Type Ninja uniquement";
		break;
	case 2173:	//Crescent_Armguard
		mes "Diminue le délai post-compétence de 2% pour chaque amélioration.";
		mes "Type de Bouclier, Défense 70, Slot 1";
		mes "Niveau requis 100, Type Ninja uniquement";
		break;
	case 15054:	//Ninja_Scale_Armor
		mes "MHP+15%, MSP-30%";
		mes "Type d'Armure, Défense 90, Slot 1";
		mes "Niveau requis 100, Type Ninja uniquement";
		break;
	case 15055:	//Tenebris_Latitantes
		mes "Chance d'activer Illusion - Ombre Niveau 1 lors d'une attaque au corps à corps.";
		mes "Type d'Armure, Défense 60, Slot 1";
		mes "Niveau requis 100, Type Ninja uniquement";
		break;
	}
	next;
	return;

L_Combine:
	mes "[Gyo]";
	mes "Assurez-vous que ^FF0000vous avez les ingrédients et l'équipement corrects à portée de main^000000.";
	mes "Pas besoin de pleurer par la suite.";
	next;
	switch(select("Combine.:Quit.")) {
	case 1:
		if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2000) {
			mes "- Attendez !! -";
			mes "- Vous ne pouvez pas recevoir d'objets -";
			mes "- car vous êtes trop chargé. -";
			mes "- Veuillez réessayer -";
			mes "- après avoir allégé votre fardeau. -";
			close;
		}
		return;
	case 2:
		mes "[Gyo]";
		mes "Il semble que vous ayez besoin de plus de temps pour décider.";
		close;
	}
	end;
}
