// Channel Warper :: channel
//============================================================
function	script	F_IzludeChannel	{
	mes "[Officiel de Warp Canal]";
	mes "Izlude est un endroit très animé, n'est-ce pas ?";
	mes "Je suis ici pour aider à rendre Izlude plus fluide !";
	mes "Je peux vous envoyer vers une autre copie d'Izlude.";
	mes "Voulez-vous aller vers une autre copie ?";
	next;
	set .@i, select(getarg(0)+":Never mind");
	if (.@i < 6) {
		setarray .@maps$[1],"izlude","izlude_a","izlude_b","izlude_c","izlude_d";
		savepoint .@maps$[.@i],128,98,1,1;
		warp .@maps$[.@i],128,98;
		end;
	}
	close;
}
