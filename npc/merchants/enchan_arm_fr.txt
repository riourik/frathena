prontera,165,60,2	script	Apprentice Craftsman	73,{
	disable_items;
	mes "[Apprentice Craftsman]";
	if (Zeny >= 400000) {
		mes "J'étudie des méthodes pour améliorer une armure afin d'optimiser son potentiel.";
		next;
		mes "[Apprentice Craftsman]";
		mes "L'enchantement est un superbe skill qui imprègne une puissance mystérieuse dans la fente cachée de l'armure.";
		next;
		mes "[Apprentice Craftsman]";
		mes "Cependant, il faut garder à l'esprit que si vous possédez deux armures du même type, l'enchantement sera appliqué dans l'ordre dans lequel elles se trouvent dans votre inventaire.";
		next;
		mes "[Apprentice Craftsman]";
		mes "Dans ce cas, l'enchantement peut être appliqué à un objet que vous n'aviez pas l'intention d'enchanter. Alors ramenez simplement ^5555ffUNE Armure^000000 que vous voulez enchanter pour éviter les problèmes...";
		next;
		mes "[Apprentice Craftsman]";
		mes "Je ne suis pas responsable de ce qui pourrait arriver si vous avez plus d'une du même type dans votre inventaire.";
		next;
		switch(select("Non Slotted Armor.:Slotted Armor.:High Grade Armor.:Maybe next time.")) {
		case 1:
			setarray .@Items[0],2307,2309,2314,2316,2321,2325,2327,2328,2330,2332,2334,2335,2341,2344,2346,2348,2350,2337,2386,2394,2395,2396;
			set .@j,50;
			break;
		case 2:
			setarray .@Items[0],2311,2318,2319,2320,2308,2310,2315,2317,2322,2324,2326,2331,2333,2336,2342,2345,2347,2349,2351;
			set .@j,55;
			set .@k,1;
			break;
		case 3:
			setarray .@Items[0],2364,2365,2391,2374,2375,2376,2377,2378,2379,2380,2381,2382,2387,2388,2389,2390;
			set .@j,60;
			break;
		case 4:
			mes "[Apprentice Craftsman]";
			mes "Veuillez revenir lorsque vous seriez intéressé par l'enchantement de votre armure.";
			close;
		}
		set .@menu$,"";
		for(set .@i,0; .@i<getarraysize(.@Items); set .@i,.@i+1)
			set .@menu$, .@menu$+getitemname(.@Items[.@i])+((.@k)?" [1]":"")+":";
		callsub S_EnchantArmor, .@Items[select(.@menu$)-1], .@j;
		end;
	}
	mes "Je suis chargé de l'Enchantement des Armures. En d'autres termes, j'étudie des méthodes pour améliorer les armures.";
	next;
	mes "[Apprentice Craftsman]";
	mes "Si par hasard vous vouliez enchanter votre armure, apportez-moi 400,000 zeny et l'armure que vous voulez enchanter et vous êtes prêt à partir.";
	close;

S_EnchantArmor:
	set .@itemid,getarg(0);
	set .@failrate,getarg(1);
	mes "[Apprentice Craftsman]";
	if (countitem(.@itemid) == 1) {
		mes "L'enchantement d'une fente coûtera 400,000 zeny. Il y aura une option enchantée aléatoire. Bien sûr, il y a un risque de briser votre armure.";
		next;
		mes "[Apprentice Craftsman]";
		mes "D'abord et avant tout.";
		mes "^ff5555Niveau de Refinement Existant de l'Armure";
		mes "^FF0000 et les Cartes seront PERDUS.^000000";
		mes "Voulez-vous toujours faire un Enchant?";
		next;
		if(select("Hmm... Let me think it over.:Go ahead.") == 1) {
			mes "[Apprentice Craftsman]";
			mes "Bien, je ne peux pas vous en vouloir. Prudence avant tout, hein?";
			mes "Vous avez maintenant une excellente journée.";
			close;
		}
		mes "[Apprentice Craftsman]";
		mes "Quel aventurier énergique! Eh bien, allons-y?";
		close2;
		specialeffect2 EF_MAPPILLAR;
		if (Zeny < 400000) {
			mes "[Apprentice Craftsman]";
			mes "Désolé, mais vous n'avez pas assez de zeny.";
			close;
		}
		progressbar "ffff00",7;
		set Zeny, Zeny-400000;
		delitem .@itemid,1;
		switch (rand(1,.@failrate)) {
			case 1: set .@addpart,4702;break;
			case 2: set .@addpart,4712;break;
			case 3: set .@addpart,4722;break;
			case 4: set .@addpart,4732;break;
			case 5: set .@addpart,4742;break;
			case 6: set .@addpart,4752;break;
			case 7:
			case 8: set .@addpart,4701;break;
			case 9:
			case 10: set .@addpart,4711;break;
			case 11:
			case 12: set .@addpart,4721;break;
			case 13:
			case 14: set .@addpart,4731;break;
			case 15:
			case 16: set .@addpart,4741;break;
			case 17:
			case 18: set .@addpart,4751;break;
			case 19:
			case 20:
			case 21: set .@addpart,4700;break;
			case 22:
			case 23:
			case 24: set .@addpart,4710;break;
			case 25:
			case 26:
			case 27: set .@addpart,4720;break;
			case 28:
			case 29:
			case 30: set .@addpart,4730;break;
			case 31:
			case 32:
			case 33: set .@addpart,4740;break;
			case 34:
			case 35:
			case 36: set .@addpart,4750;break;
			default:
				specialeffect2 EF_PHARMACY_FAIL;
				mes "[Apprentice Craftsman]";
				mes "C'est dommage.";
				mes "L'équipement demandé n'a pas pu être enchanté.";
				close;
		}
		getitem2 .@itemid, 1, 1, 0, 0, 0, 0, 0, .@addpart;
	}
	else {
		mes "Hmm? Il n'y a rien à enchanter!";
		mes "Veuillez revenir avec SEUL un équipement pour l'enchantement.";
		close;
	}
}
