niflheim,201,187,3	script	Niflheim Guide#01nif	849,{
	mes "[Guide de Niflheim]";
	mes "Bienvenue dans... ^8B4513Niflheim^000000...";
	mes "...le Royaume des Morts.";
	mes "Avez-vous besoin d'aide pour vous orienter dans ce royaume... ?";
	if (F_Navi() == 0) {
		next;
		mes "[Guide de Niflheim]";
		mes "Quand vous cliquerez sur le.. ^B9062F[nom du lieu]^000000..., vous recevrez les services de navigation les plus avancés... !";
	}
	while (1) {
		next;
		switch(select("[ Tool Shop ]:[ Pub ]:[ Weapon Shop ]:[ Witch's Castle ]:Remove Marks from Mini-Map:Cancel")) {
		case 1:
			callsub L_Mark, F_Navi("Tool Shop","niflheim,218,196","^0000FF");
			viewpoint 1,218,196,0,0x0A82FF;
			break;
		case 2:
			callsub L_Mark, F_Navi("Pub","niflheim,189,208","^006400");
			viewpoint 1,189,208,1,0xAAFF00;
			break;
		case 3:
			callsub L_Mark, F_Navi("Weapon Shop","niflheim,218,170","^800080");
			viewpoint 1,218,170,2,0xDA70D6;
			break;
		case 4:
			callsub L_Mark, F_Navi("Witch's Castle","niflheim,255,195","^FF1493");
			viewpoint 1,255,195,3,0xFF1493;
			break;
		case 5:
			mes "[Guide de Niflheim]";
			mes "Bien sûr, je vais effacer toutes les marques de votre mini-carte...";
			mes "Y a-t-il autre chose que je puisse faire pour vous... ?";
			viewpoint 2,1,1,0,0xFFFFFF;
			viewpoint 2,1,1,1,0xFFFFFF;
			viewpoint 2,1,1,2,0xFFFFFF;
			viewpoint 2,1,1,3,0xFFFFFF;
			break;
		case 6:
			mes "[Guide de Niflheim]";
			mes "......Au revoir~!..";
			close;
		}
	}
	end;
L_Mark:
	mes "[Guide de Niflheim]";
	mes "Laissez-moi marquer l'emplacement de";
	mes getarg(0)+"^000000...";
	mes "[Ricael]";
	mes "Souhaitez-vous vérifier d'autres emplacements ?";
	return;
}
