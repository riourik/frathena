valkyrie,44,47,5	script	High Wizard#Valkyrie	735,{
	if (ADVJOB == 0 ||  Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Grand Mage]";
			mes "Félicitations.";
			mes "Honneur aux guerriers !";
			close;
		}
		mes "[Grand Mage]";
		mes "Nous, les Grand Mages, avons";
		mes "la responsabilité d'utiliser";
		mes "notre magie destructive";
		mes "à des fins justes.";
		next;
		mes "[Grand Mage]";
		mes "Une vie de formation";
		mes "est requise avant devenir";
		mes "un Grand Mage. Pouvez-vous imaginer";
		mes "ce qui se passerait si notre pouvoir";
		mes "se retrouvait entre les mauvaises mains? !";
		close;
	}
	else if (ADVJOB == Job_High_Wizard && Class == Job_Mage_High && JobLevel > 39) {
		mes "[Grand Mage]";
		mes "C'est le moment.";
		mes "Et Midgard a";
		mes "besoin de ceux qui peuvent";
		mes "manier la magie la plus puissante...";
		next;
		mes "[Grand Mage]";
		mes "Souhaitez-vous";
		mes "devenir un Grand Mage ?";
		next;
		if (select("No.:Yes.") == 1) {
			mes "[Grand Mage]";
			mes "[Grand Prêtre]";
			mes "^FF0000nVenez quand vous le souhaitez.^0";
			mes "^FF0000nHonneur aux guerriers !^0";
			close;
		}
		if (SkillPoint) {
			mes "[Grand Mage]";
			mes "Il est encore possible pour vous d'apprendre des compétences supplémentaires. Veuillez utiliser";
			mes "tous vos points de compétence restants avant de revenir me voir.";
			close;
		}
		jobchange Job_high_Wizard;
		set ADVJOB,0;
		mes "[Grand Mage]";
		mes "Félicitations.";
		mes "En tant que Grand Mage,";
		mes "j'espère que vous utilisez";
		mes "vos pouvoirs pour apporter";
		mes "la paix à ceux qui sont opprimés.";
		close;
	}
	mes "[Grand Mage]";
	mes "Accueillir";
	mes "vers Valhalla,";
	mes "la Salle de l'Honneur.";
	next;
	mes "[Grand Mage]";
	mes "S'il vous plaît faites-vous";
	mes "confortable";
	mes "pendant votre séjour ici.";
	mes "Honneur aux guerriers !";
	close;
}
