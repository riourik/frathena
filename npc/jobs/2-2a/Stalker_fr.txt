valkyrie,53,58,3	script	Stalker#Valkyrie	747,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Stalker]";
			mes "Félicitations.";
			mes "Honneur aux guerriers !";
			close;
		}
		mes "[Stalker]";
		mes "Heh...";
		mes "C'est dur";
		mes "d'être un héros";
		mes "et d'avoir une réputation";
		mes "douteuse,";
		mes "trompeuse,";
		next;
		mes "[Stalker]";
		mes "Mais quand les";
		mes "choses se compliquent";
		mes "mes potes savent qu'ils";
		mes "peuvent compter sur moi.";
		mes "J'ai besoin d'eux et";
		mes "ils ont besoin de moi.";
		close;
	}
	else if (ADVJOB == Job_Stalker && Class == Job_Thief_High && JobLevel > 39) {
		mes "[Stalker]";
		mes "Ce monde a besoin";
		mes "d'plus héros prêts à marcher la ligne";
		mes "entre l'ordre et le chaos.";
		mes "between order and lawlessness.";
		next;
		mes "[Stalker]";
		mes "Es-tu prêt(e)";
		mes "^FF0000pour rejoindre les rangs^000000 des guerriers les plus rusés ?";
		mes "des guerriers les plus rusés de";
		mes "Êtes-vous prêt";
		mes "à devenir un Stalker?";
		next;
		if (select("No.:Yes.") == 1) {
			mes "[Stalker]";
			mes "[Grand Prêtre]";
			mes "n'hésitez pas à revenir.";
			mes "Honneur aux guerriers !";
			close;
		}
		if (SkillPoint) {
			mes "[Stalker]";
			mes "Il est encore possible pour vous d'apprendre des compétences supplémentaires. Veuillez utiliser";
			mes "tous vos points de compétence restants avant de revenir me voir.";
			close;
		}
		jobchange Job_Stalker;
		set ADVJOB,0;
		mes "[Stalker]";
		mes "Félicitations!";
		mes "En tant que Stalker, j'espère que vous poignardez les bonnes personnes";
		mes "dans le dos. Éliminez le mal en utilisant leurs propres";
		mes "méthodes perfides!";
		mes "[Stalker]";
		mes "[Stalker]";
		close;
	}
	mes "[Stalker]";
	mes "Accueillir";
	mes "vers Valhalla,";
	mes "la Salle de l'Honneur.";
	next;
	mes "[Stalker]";
	mes "S'il vous plaît faites-vous";
	mes "confortable";
	mes "pendant votre séjour ici.";
	mes "Honneur aux guerriers !";
	close;
}
