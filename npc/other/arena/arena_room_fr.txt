function	script	Func_Are_Rew	{
	.@item_id = getarg(0);
	.@amount = getarg(1);
	.@arena_point = getarg(2);

	mes "[Givu]";
	mes "Would you like to exchange your arena points with "+ getitemname(.@item_id) +"?";
	mes "You can exchange ^3131FF"+ .@arena_point +" arena points with "+ .@amount +" "+ getitemname(.@item_id) +"^000000.";
	mes "^FF0000Si vous souhaitez annuler, veuillez entrer 0. Sinon, veuillez saisir combien de fois ^3131FFde points d'arène^000000 vous voulez dépenser.^000000";
	next;
	input .@reward;
	if (.@reward <= 0) {
		mes "[Givu]";
		mes "^FF0000Vous avez annulé votre demande.^000000";
		close;
	} else if (.@reward > 1500) {
		mes "[Givu]";
		mes "^FF0000Vous avez dépassé la capacité maximale.^000000";
		close;
	}
	if (arena_point < (.@reward * .@arena_point)) {
		mes "[Givu]";
		mes "Vous n'avez pas assez de points d'arène.";
		mes "Veuillez vérifier le montant total de vos points d'arène.";
		close;
	}
	arena_point = arena_point - (.@reward * .@arena_point);
	getitem .@item_id,.@reward * .@amount;
	mes "[Givu]";
	mes "Merci, revenez s'il vous plaît.";
	close;
}
