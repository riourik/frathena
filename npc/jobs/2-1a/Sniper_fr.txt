valkyrie,44,55,5	script	Sniper#Valkyrie	727,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Tireur d'élite]";
			mes "Félicitations.";
			mes "Honneur aux guerriers !";
			close;
		}
		mes "[Tireur d'élite]";
		mes "Un seul coup.";
		mes "Une seule mort.";
		mes "Ce n'est pas si difficile";
		mes "une fois que vous développez la";
		mes "vue pour ce style de combat.";
		mes "Le monde a besoin";
		close;
	}
	else if (ADVJOB == Job_Sniper && Class == Job_Archer_High && JobLevel > 39) {
		mes "[Tireur d'élite]";
		mes "de Bowmen puissants";
		mes "comme vous. Êtes-vous prêt(e) pour";
		mes "la responsabilité incroyable?";
		mes "[Sniper]";
		next;
		mes "[Tireur d'élite]";
		mes "prendre l'étape suivante et";
		mes "devenir un Sniper?";
		mes "[Sniper]";
		next;
		if (select("No.:Yes.") == 1) {
			mes "[Tireur d'élite]";
			mes "[Grand Prêtre]";
			mes "n'hésitez pas à revenir.";
			mes "Honneur aux guerriers !";
			close;
		}
		if (SkillPoint) {
			mes "[Tireur d'élite]";
			mes "Il est encore possible pour vous d'apprendre des compétences supplémentaires. Veuillez utiliser";
			mes "tous vos points de compétence restants avant de revenir me voir.";
			close;
		}
		jobchange Job_Sniper;
		set ADVJOB,0;
		mes "[Tireur d'élite]";
		mes "Félicitations!";
		mes "En tant que Tireur d'élite, j'espère";
		mes "que les sbires du mal";
		mes "ne seront jamais en sécurité ainsi";
		mes "du moment qu'ils sont dans";
		mes "votre champ de vision !";
		close;
	}
	mes "[Tireur d'élite]";
	mes "Accueillir";
	mes "vers Valhalla,";
	mes "la Salle de l'Honneur.";
	next;
	mes "[Tireur d'élite]";
	mes "S'il vous plaît faites-vous";
	mes "confortable";
	mes "pendant votre séjour ici.";
	mes "Honneur aux guerriers !";
	close;
}
