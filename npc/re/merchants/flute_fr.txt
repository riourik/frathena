-	script	Falcon Flute Trader::fflute	-1,{
	mes "[Marchand de Flûte d'Aigle]";
	if (BaseJob == Job_Hunter) {
		mes "Voulez-vous louer un Falcon ?";
		mes "Avec une Flûte magique, vous pouvez faire appel à votre Faucon de n'importe où !";
		next;
		mes "[Marchand de Flûte d'Aigle]";
		mes "C'est merveilleux, n'est-ce pas ? Ha ha ha !";
		next;
		mes "[Marchand de Flûte d'Aigle]";
		mes "Pourquoi je vends ces Flûtes de Faucon ?";
		mes "C'est parce que je n'ai aucune idée de comment elles fonctionnent.";
		mes "Haha !";
		next;
		mes "[Marchand de Flûte d'Aigle]";
		mes "Bien sûr, vous aurez besoin de la compétence ^ff0000Maîtrise du Faucon^000000.";
		mes "Je n'ai pas la compétence, donc je ne sais pas comment manipuler les Faucons.";
		mes "Pour l'instant, je pourrais peut-être jamais avoir un Faucon.";
		next;
		mes "[Marchand de Flûte d'Aigle]";
		mes "Mais je ne peux pas vous en donner un gratuitement !";
		mes "Je dois gagner de l'argent car j'ai aussi besoin de manger !";
		next;
		mes "[Marchand de Flûte d'Aigle]";
		mes "Si vous avez besoin d'une Flûte de Faucon, je les vendrai pour 12 500 zeny. Ou est-ce trop cher ?";
		next;
		if (eaclass()&EAJL_THIRD) {
			mes "[Marchand de Flûte d'Aigle]";
			mes "Pour les Rangers, en revanche, j'ai un prix spécial de 10 000 zeny !";
			set .@price,10000;
			next;
		} else
			set .@price,12500;
		if(select("Don't buy one:Buy a Falcon Flute") == 1) {
			mes "[Marchand de Flûte d'Aigle]";
			mes "C'est trop dommage. Si vous en avez besoin, revenez !";
			close;
		}
		if (Zeny >= .@price) {
			if (countitem(12848) >= 1) {
				mes "[Marchand de Flûte d'Aigle]";
				mes "Comme je peux le voir, vous possédez déjà une Falcon Flute !";
				mes "Ne vous inquiétez pas, mon jeune ami, une fois que vous aurez acheté la Falcon Flute, vous n'aurez plus besoin d'en faire l'acquisition à nouveau.";
				mes "Puisque la Falcon Flute ne disparaît pas après utilisation, n'est-ce pas génial ?";
				close;
			}
			mes "[Marchand de Flûte d'Aigle]";
			mes "Voici, prenez cette flûte.";
			mes "Avec elle, vous pourrez siffler pour faire venir votre Faucon d'où que vous soyez.";
		  	set Zeny, Zeny - .@price;
			getitem 12848,1; //Falcon_Flute
			next;
			mes "[Marchand de Flûte d'Aigle]";
			mes "Juste un rappel.";
			mes "Vous avez besoin de la compétence ^ff0000Maîtrise du Faucon^000000 pour l'utiliser !";
			close;
		}
		mes "[Marchand de Flûte d'Aigle]";
		mes "Désolé mais vous n'avez pas assez d'argent pour acheter une Falcon Flute, veuillez revenir lorsque vous aurez plus d'argent !";
		close;
	}
	if (BaseJob == Job_Archer) {
		mes "Bonjour, jeune !";
		mes "Actuellement, je ne peux pas t'aider,";
		mes "mais pourquoi ne pas revenir vers moi lorsque tu seras devenu un grand Chasseur et que tu auras appris à manier correctement les Aigles ?";
		close;
	}
	mes "N'est-ce pas une belle journée aujourd'hui ?";
	close;
}

			mes "Vous pouvez revenir ici et me trouver si vous changez d'avis.";
			mes "Est-ce bien ça ?";
			close;
		}
	}
	mes "[Artisan de Flûte Expert]";
	mes "Euh ? Vous n'êtes pas un Chasseur ?";
	mes "Alors cette flûte est totalement inutile pour vous !";
	close;
