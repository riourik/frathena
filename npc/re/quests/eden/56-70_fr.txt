				mes "Ces missions ne conviennent pas à mon niveau. Je devrais chercher d'autres missions.";
				close;
			}
			setquest .@quest_id;
			mes "J'ai accepté la mission avec succès. Je dois revenir confirmer mon travail une fois la mission terminée.";
			next;
			mes "Cette mission n'a pas de limite de temps.";
		}
		close;
	}
	if (getargcount() > 5) {
		if (countitem(getarg(5)) < getarg(6))
			set .@items,1; //incomplete
		else
			set .@items,2; //complete
	}
	if (checkquest(.@quest_id) < 2 && (countitem(getarg(3)) < getarg(4) || .@items == 1)) {
		mes "Vous avez une mission en cours. Souhaitez-vous vérifier les détails ?";
		next;
		if(select("Check the details.:Cancel.") == 1)
			callsub L_Details, .@quest_id;
		close;
	} else {
		mes "J'ai bien avancé sur cette mission. Dois-je rendre compte maintenant ?";
		next;
		if(select("Report the mission.:Do not report it yet.") == 1) {
			delitem getarg(3),getarg(4);
			if (.@items)
				delitem getarg(5),getarg(6);
			getexp getarg(1),getarg(2);
			erasequest .@quest_id;
			mes "Vous avez complété la mission. Obtenez vos récompenses.";
		}
		close;
	}
	end;
