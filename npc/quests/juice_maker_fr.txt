payon_in03,188,146,5	script	Marx Hansen#juice	86,{
	if (checkweight(1201,1) == 0) {
		mes "- Attends un instant ! -";
		mes "- Actuellement, vous portez -";
		mes "- trop d'objets avec vous. -";
		mes "- S'il vous plaît, revenez plus tard -";
		mes "- après avoir mis certains objets dans le stockage kafra. -";
		close;
	}
	if (MISC_QUEST&1) {
		mes "[Marchand Marx Hansen]";
		mes "Accueillir.";
		mes "Es-tu venu à";
		mes "transformer les fruits également ?";
		next;
		switch(select("Make Juice.:Talk and get information about fruit processing.:Cancel.")) {
		case 1:
			mes "[Marchand Marx Hansen]";
			mes "Quel genre de jus de fruits aimeriez-vous préparer ?";
			next;
			switch(select("Apple Juice:Banana Juice:Carrot Juice:Grape Juice:Cancel")) {
			case 1:
				set .@fruit,512;
				set .@juice,531;
				break;
			case 2:
				set .@fruit,513;
				set .@juice,532;
				break;
			case 3:
				set .@fruit,515;
				set .@juice,534;
				break;
			case 4:
				set .@fruit,514;
				set .@juice,533;
				break;
			case 5:
				mes "[Marchand Marx Hansen]";
				mes "Eh bien alors...";
				mes "A la prochaine fois.";
				close;
			}
			if (countitem(.@fruit) == 0 || countitem(713) == 0 || Zeny < 3) {
				mes "[Marchand Marx Hansen]";
				mes "Oh non...";
				mes "You don't have all the necessary materials. To make "+getitemname(.@juice)+", I need 1 "+getitemname(.@fruit)+" and 1 Empty Bottle. I will also need a 3 zeny fee.";
				next;
				mes "[Marchand Marx Hansen]";
				mes "Lorsque vous aurez tout préparé, je mixerai les fruits pour vous offrir un délicieux jus.";
				close;
			}

			mes "[Marchand Marx Hansen]";
			mes "Combien en voudriez-vous ?";
			next;
			switch(select("As many as I can.:I want a certain amount.:Cancel.")) {
			case 1:
				set .@make,countitem(.@fruit);
				if (countitem(713) < .@make) set .@make, countitem(713);
				if (Zeny/3 < .@make) set .@make, Zeny/3;
				break;
			case 2:
				mes "[Marchand Marx Hansen]";
				mes "Choose a number less than 100. If you don't want to, put '0'. You can make up to " + countitem(.@fruit) + " bottles of juice.";
				next;
				while(1) {
					input .@input;
					if (.@input == 0) {
						mes "[Marchand Marx Hansen]";
						mes "Eh bien alors...";
						mes "Reviens.";
						close;
					}
					else if (.@input > 100) {
						mes "[Marchand Marx Hansen]";
						mes "Plus de 100 bouteilles est impossible. Choisissez un montant différent.";
						next;
					}
					else {
						break;
					}
				}
				set .@make,.@input;
				break;
			case 3:
				mes "[Marchand Marx Hansen]";
				mes "Eh bien alors...";
				mes "Reviens.";
				close;
			}

			set .@total_zeny,3 * .@make;

			if (countitem(.@fruit) < .@make || countitem(713) < .@make || Zeny < .@total_zeny) {
				mes "[Marchand Marx Hansen]";
				mes "Oh non...";
				mes "Vous n'avez pas tout le matériel nécessaire. Je ne peux pas aider une situation comme celle-ci. Je suppose que vous rassemblez ce dont vous avez besoin.";
				close;
			}
			delitem .@fruit,.@make;
			delitem 713,.@make;
			set Zeny, Zeny-.@total_zeny;
			getitem .@juice,.@make;

			mes "[Marchand Marx Hansen]";
			mes "Voici! Jus frais et délicieux comme promis. Cela devrait être très rafraîchissant et savoureux.";
			next;
			mes "[Marchand Marx Hansen]";
			mes "Eh bien alors...";
			mes "Reviens.";
			close;
		case 2:
			mes "[Marchand Marx Hansen]";
			mes "Avant que les humains ne soient capables de développer des sociétés aussi vastes, ils cueillaient les fruits des arbres pour survivre. Les fruits ont peut-être été une bénédiction de la nature qui nous a permis d'exister dans le monde.";
			next;
			mes "[Marchand Marx Hansen]";
			mes "Depuis que la vie est devenue si prospère, la jeune génération semble ne plus manger de fruits. J’ai donc commencé à réfléchir à des moyens de rendre les fruits plus faciles à manger.";
			next;
			mes "[Marchand Marx Hansen]";
			mes "J'ai réalisé que lorsqu'on prépare des jus de fruits, ils sont plus pratiques à manger et ont un bien meilleur goût.";
			next;
			mes "# Informations sur les jus de fruits #";
			mes "^CC4E5C- Jus de pomme -^000000";
			mes "Pomme x 1 pièce, Bouteille vide x 1 pièce, 3 zeny.";
			mes "^E3CF57- Jus de banane -^000000";
			mes "Banane x 1 pièce, Bouteille vide x 1 pièce, 3 zeny.";
			mes "^ED9121- Jus de carotte -^000000";
			mes "Carotte x 1 pièce, Bouteille vide x 1 pièce, 3 zeny.";
			mes "^CC00FF- Jus de raisin -^000000";
			mes "Raisin x 1 ch., Bouteille vide x 1 ch., 3 zeny.";
			close;
		case 3:
			mes "[Marchand Marx Hansen]";
			mes "Hé!";
			mes "Si vous visitez";
			mes "quelqu'un, parle-lui !";
			close;
		}
	}
	else {
		mes "[Marchand Marx Hansen]";
		mes "Accueillir.";
		mes "Comment puis-je vous aider ?";
		next;
		if (select("Talk:Cancel") == 1) {
			mes "[Marchand Marx Hansen]";
			mes "Avant que les humains ne soient capables de développer des sociétés aussi vastes, ils cueillaient les fruits des arbres pour survivre. Les fruits ont peut-être été une bénédiction de la nature qui nous a permis d'exister dans le monde.";
			next;
			mes "[Marchand Marx Hansen]";
			mes "Depuis que la vie est devenue si prospère, la jeune génération semble ne plus manger de fruits. J’ai donc commencé à réfléchir à des moyens de rendre les fruits plus faciles à manger.";
			next;
			mes "[Marchand Marx Hansen]";
			mes "J'ai réalisé que lorsqu'on prépare des jus de fruits, ils sont plus pratiques à manger et ont un bien meilleur goût.";
			close;
		}
		mes "[Marchand Marx Hansen]";
		mes "Hé!";
		mes "Si vous visitez";
		mes "quelqu'un, parle-lui !";
		close;
	}
}
