	case -1:
		.@party_id = getcharid(1);
		if (.@party_id < 1) {
			mes "...Ça ne sent pas bon ici.";
			mes "Cela pourrait être dangereux seul, donc il vaudrait mieux former un groupe.";
			close3;

			.@string$ = "Open door.";
		switch( select( .@string$, "Go in.", "Quit." ) ) {
		case 1:
			if (instance_create("Last room") < 1) {
				mes "Party: " + getpartyname( getcharid(1) ) + "";
				mes "Leader: " + getpartyleader( getcharid(1) ) + "";
				mes "^0000ffDernière pièce ^000000- Création échouée";
				close3;

			mes "[Mark]";
			mes "Je ouvre la porte.";
			mes "Entrez lorsque la porte sera complètement ouverte.";
			close3;
		case 2:
			switch( instance_enter("Last room") ) {
			case IE_NOMEMBER:
				mes "[Mark]";
				mes "...Ça fait très suspect !";
				mes "C'est un peu dangereux d'être seul. Il est recommandé de former une équipe !";
				close3;
			case IE_NOINSTANCE:
				mes "[Mark]";
				mes "La porte ne semble pas être entièrement ouverte.";
				close3;
			case IE_OTHER:
				mes "[Mark]";
				mes "Euh... qu'est-ce qui se passe ?";
				close3;
			case IE_OK:
				mapannounce "un_myst", "" + strcharinfo(0) + " of the party, " + getpartyname(.@party_id) + ", is entering Last room.",bc_map,"0x00FF99";
				setquest 11379;// Final Room
				if (isbegin_quest(11380) == 0)
					setquest 11380;// Final Room
				end;
			}
			end;
		case 3:
			close3;
		}
		end;
	case 0:
	case 1:
		mes "[Mark]";
		cutin "bu_mark1.bmp",0;
		mes "...Vous avez l'air fatigué.";
		mes "Depuis que vous y êtes allé, vous devriez prendre une journée pour vous reposer.";
		close3;
	case 2:
		mes "[Mark]";
		cutin "bu_mark1.bmp",0;
		mes "Je ne peux pas croire que vous vouliez entrer dans un endroit comme celui-ci à nouveau...";
		erasequest 11379;// Final Room
		close3;
	}
}
