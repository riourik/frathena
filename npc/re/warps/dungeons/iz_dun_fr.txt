iz_dun04,130,234,5	script	Gatekeeper#iz_dun	413,{
	mes "[Gardien]";
	mes "Arrêtez !";
	mes "Que faites-vous ici, Aventurier ?";
	mes "Nous avons découvert un nouveau niveau de cette cité sous-marine.";
	next;
	mes "[Gardien]";
	mes "Pensez-vous relever le défi ?";
	mes "Les monstres d'ici sont entre les niveaux 110 et 124.";
	mes "Et vous devez faire attention au Kraken !";
	next;
	switch(select("Release the Kraken!:I'll stay here.")) {
	case 1:
		mes "[Gardien]";
		mes "Soyez prudent(e) et bonne chance.";
		mes "Si vous voyez le Kraken, assurez-vous d'avoir vos amis pour vous aider ou faites comme nous et fuyez.";
		close2;
		warp "iz_dun05",141,187;
		end;
	case 2:
		mes "[Gardien]";
		mes "Il est toujours préférable de se préparer avant d'explorer une nouvelle zone.";
		mes "Revenez lorsque vous serez prêt(e) à partir.";
		close;
	}
}
