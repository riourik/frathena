yuno,179,174,4	script	Gomer	851,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 1000) {
		mes "[Gomer]";
		mes "Vous transportez trop de choses. Déposez une partie de vos bagages et revenez vers moi.";
		close;
	}
	if (countitem(1549)) {
		mes "[Gomer]";
		mes "Alors, comment vous porte votre Pilebuncker ?";
		close;
	}
	if (Class != Job_Mechanic && Class != Job_Mechanic_T && Class != Job_Baby_Mechanic) {
		mes "[Gomer]";
		mes "Je suis un Mécanicien retraité et je ne pense pas avoir affaire à vous.";
		close;
	}
	mes "[Gomer]";
	mes "Donc ? Qu'est-ce que ça fait d'être sur un Madogear ?";
	mes "C'est quelque chose qui n'est pas autorisé pour ceux qui ont échoué à trouver leur propre voie.";
	next;
	mes "[Gomer]";
	mes "Moi ? Je me suis retiré il y a longtemps. À cause d'un accident... j'ai été grièvement blessé et je ne pourrai plus jamais rouler. C'est ce qui me chagrine.";
	next;
	mes "[Gomer]";
	mes "Alors, j'ai décidé de rester ici en forgeant des engrenages pour les Madogear.";
	mes "Euh ? Je ne dis pas que je n'ai aucun grief contre ce travail.";
	next;
	mes "[Gomer]";
	mes "Être forcé à faire quelque chose qu'on n'aime pas, il n'y a rien de plus misérable que cela.";
	next;
	mes "[Gomer]";
	mes "Il y a des gens qui sont heureux d'être simplement des accessoires, vous savez, comme être le mur contre lequel quelqu'un peut s'appuyer.";
	next;
	mes "[Gomer]";
	mes "Cela me fait du bien et je suis sur le point de fabriquer un nouvel engrenage. Je veux que vous m'aidiez. Qu'en pensez-vous ?";
	next;
	switch(select("I am not that interested.:What are you going to make?:Where do you get the materials?:I brought the materials.")) {
	case 1:
		mes "[Gomer]";
		mes "Rien que je ne puisse faire";
		mes "à ce sujet alors.";
		mes "À la prochaine !";
		close;
	case 2:
		mes "[Gomer]";
		mes "Euh, une énorme tige métallique capable d'écraser un adversaire en un seul coup !";
		next;
		mes "[Gomer]";
		mes "Elle ne fait pas très chic, mais j'ai conçu un mécanisme à ressort pour attaquer avec une grande force.";
		next;
		mes "[Gomer]";
		mes "Il est dissimulé, donc quand tu trouveras l'occasion, un seul tir pourrait mettre un adversaire dans un état critique !";
		mes "J'ai hâte de le voir en action !";
		next;
		mes "[Gomer]";
		mes "Si vous imaginez à quel point ce genre d'arme peut être efficace, je vais la rendre réalité.";
		next;
		mes "[Gomer]";
		mes "Apportez-moi 200 Acier, 30 Tuyaux Flexibles et 1 Brocca. Je pourrais fabriquer ce que j'ai décrit avec ces matériaux.";
		close;
	case 3:
		mes "[Gomer]";
		mes "Vous devriez pouvoir trouver de l'acier auprès de nombreux monstres comme les Kobolds.";
		next;
		mes "[Gomer]";
		mes "Les Tuyaux Flexibles sont plus rares, tombés des Metalings autour d'Einbroch et de Lighthalzen.";
		next;
		mes "[Gomer]";
		mes "Enfin, trouver une Brocca est très difficile, mais vous devriez pouvoir y arriver. Haha, allez la chercher !";
		close;
	case 4:
		if (countitem(999) < 200 || countitem(7325) < 30 || countitem(1415) < 1) {
			mes "[Gomer]";
			mes "Vous n'avez pas apporté tous les matériaux que je vous ai demandés.";
			next;
			mes "[Gomer]";
			mes "J'ai l'habitude d'abandonner ce que j'étais en train de fabriquer pour essayer de créer quelque chose de nouveau lorsque l'inspiration me vient.";
			mes "Vous feriez mieux de vous dépêcher !";
			close;
		} else {
			mes "[Gomer]";
			mes "Oh, je t'attendais. Comme tu n'es pas venu à l'heure, je me demandais si tu avais oublié. Attends un peu là.";
			specialeffect EF_REPAIRWEAPON;
			next;
			mes "[Gomer]";
			mes "C'est fini. Qu'en penses-tu ? Pas mal, hein ? Bien que je l'aie fait moi-même, je dois dire qu'il est plutôt bien réalisé.";
			next;
			mes "[Gomer]";
			mes "Combinaison de poids, de volume et d'apparence avec sa couleur et son éclat. Ça me prend au cœur !";
			next;
			mes "[Gomer]";
			mes "^FFFFFF...........";
			next;
			mes "[Gomer]";
			mes "Euh, avec quelques petites améliorations, cela pourrait être produit en série et livré. Je suis sûr que je peux en faire d'autres, donc je te donne celui-ci.";
			next;
			mes "[Gomer]";
			mes "Oh, et il s'avère que je n'avais pas besoin de tout ce Fer après tout. Alors voici, reprends la plus grande partie.";
			delitem 999,50; //Steel
			delitem 7325,30; //Tube
			delitem 1415,1; //Skewer
			getitem 1549,1; //Pilebuncker
			getitem 1360,1; //Two_Handed_Axe
			close;
		}
	}
}
