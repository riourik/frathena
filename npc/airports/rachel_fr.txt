ra_fild12,295,208,0	script	toairplane#rachel	45,1,1,{
	end;
OnTouch:
	mes "Pour utiliser le dirigeable, vous devez payer 1 200 zeny ou un billet de dirigeable gratuit.";
	mes "Souhaitez-vous utiliser le service ?";
	next;
	if (select("Yes:No") == 1) {
		if (countitem(7311) > 0) {
			delitem 7311,1; //Free_Flying_Ship_Ticket
			warp "airplane_01",245,60;
			end;
		}
		if (Zeny >= 1200) {
			set Zeny, Zeny - 1200;
			warp "airplane_01",245,60;
			end;
		}
		mes "Je suis désolé, mais vous n'avez pas assez d'argent.";
		mes "N'oubliez pas que vous devez payer 1 200 zeny pour utiliser le service.";
		close;
	}
	mes "Merci, revenez s'il vous plaît.";
	close;
}
