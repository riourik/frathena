prontera,246,62,3	script	Training Instructor#sol1	8W_SOLDIER,{
	mes "[Instructeur d'entraînement]";
	mes "Si vous souhaitez vous entraîner dans une zone plus vaste, je vous laisserai entrer.";
	next;
	mes "[Instructeur d'entraînement]";
	mes "Le terrain d'entraînement est divisé en deux zones principales.";
	next;
	mes "[Instructeur d'entraînement]";
	mes "Zone supérieure où vous pouvez vérifier la force appliquée en fonction de la taille du monstre.";
	next;
	mes "[Instructeur d'entraînement]";
	mes "Zone inférieure où vous pouvez vérifier la force appliquée selon l'attribut et la forme du monstre.";
	next;
	mes "[Instructeur d'entraînement]";
	mes "Où souhaitez-vous entrer ?";
	next;
	switch( select( "Upper area", "Lower area", "Do not enter" ) ) {
	case 1:
		mes "[Instructeur d'entraînement]";
		mes "Alors je vais vous guider.";
		close2;
		warp "tra_fild",112,167;
		end;
	case 2:
		mes "[Instructeur d'entraînement]";
		mes "Alors je vais vous guider.";
		close2;
		warp "tra_fild",111,74;
		end;
	case 3:
		mes "[Instructeur d'entraînement]";
		mes "Si vous avez besoin d'aide, venez vers moi.";
		close;
	}
}
