	mes "[Officier des relations publiques de Moscou]";
	mes "Moscovie! Connaissez-vous la Moscovie ?";
	mes "le paradis s'étendant sur les mers sans fin...";
	mes "Bienvenue en Moscovie";
	mes "C'est aventureux et mystique.";
	next;
	switch(select("About Moscovia...:Go to Moscovia.:Cancel")) {
		case 1:
			mes "[Officier des relations publiques de Moscou]";
			mes "La Moscovie est un beau royaume";
			mes "sur une île située au nord de Rune-";
			mes "Midgarts.";
			next;
			mes "[Officier des relations publiques de Moscou]";
			mes "Je suis sûr que tu le seras";
			mes "absolument fasciné";
			mes "par les magnifiques paysages de la Moscovie";
			mes "et un palais magnifique.";
			next;
			mes "[Officier des relations publiques de Moscou]";
			mes "Maintenant que notre long hiver est terminé";
			mes "passé,";
			mes "Je suis heureux de pouvoir maintenant te montrer";
			mes "notre magnifique ville natale.";
			next;
			mes "[Officier des relations publiques de Moscou]";
			mes "Si vous avez envie de visiter la Moscovie";
			mes "saisissez votre chance maintenant !";
			mes "Je vais t'aider à faire un bon voyage";
			mes "en Moscovie!";
			close;

		case 2:
			mes "[Officier des relations publiques de Moscou]";
			mes "Ok, commençons maintenant.";
			mes "Tu devrais me payer 10 000 zeny";
			mes "aller en Moscovie.";
			mes "Mais quand tu reviens,";
			mes "vous n'êtes pas obligé de payer.";
			next;
			mes "[Officier des relations publiques de Moscou]";
			mes "Pouvons-nous partir maintenant ?";
			next;
			if (select("Let's go!:Cancel") == 2) {
				mes "[Officier des relations publiques de Moscou]";
				mes "Si tu es trop occupé maintenant,";
				mes "s'il te plaît, répète-moi quand tu veux.";
				mes "Je suis toujours prêt à guider n'importe qui vers la Moscovie.";
				close;
			}
			if (Zeny < 10000) {
				mes "[Officier des relations publiques de Moscou]";
				mes "Je suis désolé mais tu n'as pas";
				mes "assez zeny maintenant";
				mes "Il te faut 10 000 zeny";
				mes "aller en Moscovie";
				mes "Merci.";
				close;
			} else {
				mes "[Officier des relations publiques de Moscou]";
				mes "Ok, on part maintenant.";
				close2;
				set Zeny,Zeny - 10000;
				warp "moscovia",163,55;
				end;
			}
		case 3:
			mes "[Officier des relations publiques de Moscou]";
			mes "Si tu es trop occupé maintenant,";
			mes "s'il te plaît, répète-moi quand tu veux.";
			mes "Je suis toujours prêt à guider n'importe qui vers la Moscovie.";
			close;
	}
}
