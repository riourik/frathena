tra_fild01,204,110,2	script	Training Ground Warehouse Soldier#fild01_1	8W_SOLDIER,{
	mes "[Soldat de l'Entrepôt du Terrain d'Entraînement]";
	mes "Souhaitez-vous utiliser le Stockage ? 1000 Zeny sont nécessaires pour utiliser le Stockage du terrain d'entraînement. Veuillez sélectionner le Stockage que vous souhaitez utiliser.";
	next;
	// switch( select( " Storage 1", " Storage 2", " Storage 3", " Guild Storage", " Cancel" ) ) {
	switch( select( " Storage 1", "", "", " Guild Storage", " Cancel" ) ) {
	case 1:
		.@storage_num = 1;
		break;
	case 2:
		.@storage_num = 2;
		break;
	case 3:
		.@storage_num = 3;
		break;
	case 4:
		// copied from npc/re/merchants/guild_warehouse.txt
		.@guildid = getcharid( 2 );

		if( .@guildid == 0 ){
			mes "[Soldat de l'Entrepôt du Terrain d'Entraînement]";
			mes "Le stockage de la guilde n'est disponible que pour les membres de la guilde.";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "[Soldat de l'Entrepôt du Terrain d'Entraînement]";
			mes "Il semble que votre guilde ne puisse pas encore utiliser l'entrepôt de guilde ou que vous n'avez pas les permissions d'accès requises.";
			mes "Veuillez revenir après avoir vérifié la compétence de guilde et les permissions d'accès à l'entrepôt.";
			close;
		}

		if( Zeny < 1000 ){
			mes "[Soldat de l'Entrepôt du Terrain d'Entraînement]";
			mes "Je crains que vous n'ayez pas assez d'argent pour régler les frais.";
			mes "Les frais s'élèvent à 1000 Zeny.";
			close;
		}

		mes "[Soldat de l'Entrepôt du Terrain d'Entraînement]";
		mes "Je vais alors ouvrir le stockage de la guilde pour vous. Passez un moment inoubliable !";
		close2;

		if( Zeny < 1000 ){
			// Cheat prevention
			end;
		}

		if( guildopenstorage() == GSTORAGE_OPEN ){
			Zeny -= 1000;
			end;
		}else{
			mes "[Soldat de l'Entrepôt du Terrain d'Entraînement]";
			mes "Il semble qu'une autre personne utilise actuellement l'entrepôt.";
			mes "Veuillez réessayer dans quelques instants.";
			close;
		}
	case 5:
		end;
	}
	// TODO (custom text)
	if (!callfunc("F_CanOpenStorage")) {
		mes "[Soldat de l'Entrepôt du Terrain d'Entraînement]";
		mes "Je suis désolé, mais toi";
		mes "j'ai besoin du statut d'apprenti";
		mes "Niveau de compétence basique 6 pour";
		mes "utiliser le service de stockage.";
		close2;
		cutin "",255;
		end;
	}

	if (Zeny < 1000) {
		mes "[Soldat de l'Entrepôt du Terrain d'Entraînement]";
		mes "^4d4dff1000 Zeny sont nécessaires pour utiliser le Stockage des terrains d'entraînement.^000000";
		close;
	}
	Zeny -= 1000;
	openstorage; // TODO: open correct storage 1-3
	end;
}

tra_fild01,142,113,6	script	Cleaning Robot#fild01	EP17_2_OMEGA_CLEANER,{
OnStart:
	.@num = atoi( charat(strnpcinfo(2), 5) );

	if (.is_moving[.@num])
		end;
	.is_moving[.@num] = true;
	stopnpctimer;

	setarray .@emotions[0], ET_SWEAT, ET_PROFUSELY_SWEAT, ET_HUNGRY, ET_OHNO;
	.@r = rand( getarraysize(.@emotions) );
	emotion .@emotions[.@r];

	setarray .@speed[0], 300, 400, 500, 600;
	.@r = rand( getarraysize(.@speed) );
	npcspeed .@speed[.@r];

	getmapxy .@map$,.@x,.@y, BL_NPC;
	switch( .@num ) {
	case 1:
		if (.@x <= 126)
			.sens[.@num] = 1;
		else if (.@x >= 198)
			.sens[.@num] = -1;
		.@x += 8 * .sens[.@num];
		break;
	case 2:
		if (.@x <= 21)
			.sens[.@num] = 1;
		else if (.@x >= 85)
			.sens[.@num] = -1;
		.@x += 8 * .sens[.@num];
		break;
	case 3:
		if (.@y <= 33)
			.sens[.@num] = 1;
		else if (.@y >= 89)
			.sens[.@num] = -1;
		.@y += 8 * .sens[.@num];
		break;
	}
	unitwalk getnpcid(0),.@x,.@y, strnpcinfo(0) + "::OnMoveEnd";
	end;
OnMoveEnd:
	.@num = atoi( charat(strnpcinfo(2), 5) );
	.is_moving[.@num] = false;
	initnpctimer;
	end;
OnTimer30000:
	donpcevent strnpcinfo(0) + "::OnStart";
	end;

OnInit:
	.@num = atoi( charat(strnpcinfo(2), 5) );
	.sens[.@num] = 1;
	initnpctimer;
	end;
}
