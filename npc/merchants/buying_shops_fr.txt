// Black Marketeer (Buy Licenses - Non-Merchant Classes) :: qskill_buy_store
//============================================================ 
que_job01,68,84,1	script	Black Marketeer#Buying	881,{
	if (checkweight(1201,1) == 0 || MaxWeight - Weight < 2400) {
		mes "[M. Jass]";
		mes "Tu es trop cupide, même par rapport à quelqu'un comme moi !";
		mes "Pourquoi ne pas alléger ton sac d'abord ?";
		close;
	}
	if (getskilllv("ALL_BUYING_STORE") == 1) {
		mes "[M. Jass]";
		mes "Hey, tu as déjà fait un contrat avec Hugh.";
		mes "Je n'ai aucun commerce à faire avec toi.";
		close;
	}
	mes "[M. Jass]";
	mes "Tu dois avoir besoin de quelque chose d'important pour venir me trouver.";
	mes "Que veux-tu?";
	next;
	switch(select("Bulk Buyer Shop License:Who are you?:Nothing, nothing!")) {
	case 1:
		mes "[M. Jass]";
		mes "Je savais que c'était ça!";
		mes "Bien sûr, je peux le faire pour toi.";
		mes "Le mien ressemble exactement à l'original que les commerçants obtiennent de ce salopard de M. Hugh !";
		next;
		mes "[M. Jass]";
		mes "Et mon permis est meilleur car tu n'as pas besoin d'apprendre de nouvelles compétences.";
		mes "Combien en veux-tu ?";
		next;
		mes "[M. Jass]";
		mes "Sache que je ne peux en fabriquer qu'en petites quantités, jusqu'à 10 à la fois.";
		mes "Cela te coûtera 500 zeny pour chacun.";
		next;
		while(1) {
			input .@input;
			mes "[M. Jass]";
			if (.@input == 0) {
				mes "N'as-tu pas besoin de ces licences ?";
				close;
			} else if (.@input > 10) {
				mes "Je ne peux en fabriquer que jusqu'à 10 à la fois, tu sais.";
				next;
			} else {
				mes "It'll cost "+(.@input*500)+" zeny.";
				if (Zeny < (.@input*500)) {
					mes "mais tu n'as pas assez d'argent.";
					next;
					mes "[M. Jass]";
					mes "Ne connais-tu donc pas les bases des affaires ? Tout a un prix.";
					mes "Si tu veux quelque chose, il faut le payer.";
				} else {
					mes "Ha... Ha ha ha !";
					mes "Mr Hugh, je vais prendre en charge votre licence commerciale. Vous verrez !";
					mes "*Rire Rire*";
					getitem 12548,.@input; //Buy_Market_Permit2
					set Zeny,Zeny-(.@input*500);
				}
				close;
			}
		}
	case 2:
		mes "[M. Jass]";
		mes "Je suis parti de ma ville natale il y a longtemps.";
		mes "Il est inutile de demander qui je suis car tout ce qu'il me reste maintenant, c'est ma haine.";
		next;
		mes "[M. Jass]";
		mes "...";
		mes "Hugh est un commerçant corrompu sans éthique commerciale.";
		mes "Mon seul but dans la vie est de détruire Hugh.";
		next;
		mes "[M. Jass]";
		mes "Aaah... j'ai trop bu... (*Hic*)";
		mes "C'est juste l'alcool qui parle.";
		mes "Veuillez ignorer tout ce que je vous ai dit.";
		close;
	case 3:
		mes "[M. Jass]";
		mes "D'accord, d'accord ! Il n'y a pas besoin de crier.";
		mes "Laissez-moi tranquille si vous n'avez rien à me dire.";
		close;
	}
}

		mes "- Vous ne pouvez ni converser ni faire la quête car vous transportez trop d'objets. -";
		close;
	}
	mes "[M. Hugh]";
	if (BaseClass == Job_Merchant && getskilllv("MC_VENDING") >= 1) {
		if (getskilllv("ALL_BUYING_STORE") == 1) {
			mes "Je suis Hugh de l'équipe des Achats.";
			mes "Comment puis-je vous aider aujourd'hui ?";

					mes "Veuillez entrer un montant de 50 ou moins.";
					next;

					mes "It'll cost "+(.@input*200)+" zeny for "+.@input+" licenses.";
					if (Zeny < (.@input*200)) {
						mes "mais vous ne semblez pas avoir assez d'argent.";

						mes "Merci pour votre mécénat.";

					}
					close;
				}

			mes "Vous avez raison, mais réfléchissez-y:";
			mes "n'avez-vous pas eu du mal à acheter en gros?";
			mes "Vous auriez dû trouver et parler à chacun qui possède l'article que vous souhaitez.";
			mes "Plutôt inconvenant, n'est-ce pas?";
			next;
			mes "[M. Hugh]";
			mes "Comme l'achat en gros est une question importante pour nous, les Marchands,";
			mes "j'ai proposé un plan innovant à notre guilde, basé sur mes 10 ans d’expérience dans les achats.";
			next;
			mes "[M. Hugh]";
			mes "'Laissez les individus ouvrir
un magasin d'acheteur en gros !'";
			mes "Voici le titre de ma proposition.";
			mes "Vous verrez, quand vous lirez cela...";
			mes "[M. Hugh]";
			next;
			mes "[M. Hugh]";
			mes "...";
			mes "[M. Hugh]";
			next;
			select("Alright, what's your point?");
			mes "[M. Hugh]";
			mes "vous pouvez acheter certains objets en gros par le biais de la Vente.";
			mes "[M. Hugh]";
			next;
			mes "[M. Hugh]";
			mes "[M. Hugh]";
			next;
			mes "[M. Hugh]";
			mes "[M. Hugh]";
			next;

			mes "Notre président était si content d'apprendre que nous allions faire des bénéfices importants!";
			mes "Our president was so happy to hear that we're going to make big bucks!";
			next;
			mes "["+strcharinfo(0)+"]";
			mes "Veuillez en venir au fait !";
			next;
			mes "[M. Hugh]";
			mes "Ne sois pas si impatient, d'accord ?";
			mes "Mon propos est que nous pouvons vous laisser ouvrir le Bulk Buyer Shop si vous avez appris le Vending.";

			mes "[M. Hugh]";
			mes "Bien sûr, nous facturons un frais de dossier unique de 10 000 zeny.";
			mes "Vous devrez avoir une ^4A4AFFLicence du Bulk Buyer Shop^000000 chaque fois que vous ouvrirez le magasin.";
