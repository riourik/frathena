izlude,127,175,3	script	Hypnotist#novice	124,{
	mes "[Hypnotiste]";
	mes "Salutations, aventurier.";
	mes "Je suis membre du Hypnotiste";
	mes "Académie envoyée ici à Izlude";
	mes "pour fournir des services de Réinitialisation de Compétences.";
	mes "à certains personnages de Classe Primaire pour un prix vraiment excellent : gratuit !";
	next;
	mes "[Hypnotiste]";
	mes "Bien que je propose une réinitialisation illimitée";
	mes "réinitialisations de stats/compétences pour le moment, j'ai";
	mes "deux conditions qui doivent être";
	mes "remplies. Tout d'abord, vous devez être";
	mes "inférieur à ^FF0000Niveau de Base 50^000000.";
	next;
	mes "[Hypnotiste]";
	mes "Deuxièmement, vous devez être un";
	mes "^FF0000Guerrier, Religieux, Mage, Voleur, Archer, Commerçant, Garçon de Taekwon, Fille de Taekwon, Tireur d'élite et Ninja^000000";
	mes "Personnage de Métier pour être éligible.";
	mes "Garçon Taekwon, ou Fille Taekwon,";
	mes "Gunslinger et Ninja^000000";
	mes "[Hypnotiste]";
	mes "Les Réinitialisations de Compétences permettent aux aventuriers";
	next;
	switch(select("^FF0000Stat/Skill Reset?:Reset stats:Reset skills^000000:Cancel")) {
	case 1:
		mes "[Hypnotiste]";
		mes "Les Réinitialisations Stat/Compétences permettent aux aventuriers";
		mes "pour redistribuer leurs Points de Compétence";
		mes "de Compétence s'ils ne sont pas contents";
		mes "avec leurs compétences actuelles.";
		next;
		mes "[Hypnotiste]";
		mes "Avant de procéder à";
		mes "une Réinitialisation Stat/Compétence, vous devez";
		mes "réduire le poids total";
		mes "de tous les objets que vous portez sur votre";
		mes "personnage à 0. Vous pouvez déposer";
		mes "des objets supplémentaires dans le Dépôt de Kafra.";
		next;
		mes "[Hypnotiste]";
		mes "Ah, il est aussi important";
		mes "de retirer votre ^FF0000Caddie^000000";
		mes "si vous en avez un équipé.";
		mes "Sinon, l'hypnose ne";
		mes "fonctionnera pas, ou se retournera contre vous...";
		close;
	case 2:
		set .@str$,"Stat";
	case 3:
		if (.@str$ == "")
			set .@str$,"Skill";
		mes "[Hypnotiste]";
		mes "Êtes-vous sûr que vous";
		mes "voulez procéder avec";
		mes "my ^FF0000"+.@str$+" Reset^000000 service?";
		next;
		if(select("Yes:Cancel") == 1)
			break;
	case 4:
		mes "[Hypnotiste]";
		mes "^FF0000Merci, et bonne";
		mes "^FF0000chance sur vos aventures.";
		mes "^FF0000Voyagez en sécurité~";
		close;
	}
	if ((Class >= Job_Swordman && Class <= Job_Thief) || Class == Job_Taekwon || Class == Job_Gunslinger || Class == Job_Ninja) {
		if (checkcart()) {
			mes "[Hypnotiste]";
			mes "chariot avant de procéder à la réinitialisation des compétences.";
			mes "Merci de coopérer~";
			mes "with the "+.@str$+" Reset service.";
			mes "Si vous êtes ici pour mon service de Réinitialisation des Compétences, veuillez";
			close;
		}
		if (Weight != 0) {
			mes "[Hypnotiste]";
			mes "If you're here for my "+.@str$;
			mes "[Hypnotiste]";
			mes "Pourquoi ne pas placer vos";
			mes "reset your "+.@str$+"s until the";
			mes "^FF0000poids de vos objets transportés dans";
			mes "votre Inventaire est réduit à 0^000000.";
			next;
			mes "[Hypnotiste]";
			mes "Pourquoi ne pas placer votre";
			mes "choses dans le Stockage Kafra";
			mes "pour l'instant ? De cette façon, vous pourrez";
			mes "garder en sécurité tous vos biens.";
			close;
		}
		if (BaseLevel >= 50 && (Class == Job_Taekwon || Class == Job_Gunslinger || Class == Job_Ninja)) {
			mes "[Hypnotiste]";
			mes "Désolé, mais les personnages";
			mes "ayant un Niveau de Base supérieur à";
			mes "que 50 sont éligibles pour la";
			mes .@str$+" Reset service I provide.";
			close;
		}
		mes "[Hypnotiste]";
		mes "Merci d'avoir utilisé";
		mes "my "+.@str$+" Redistribution";
		mes ". Oh, et bonne chance dans vos";
		mes "voyages, aventurier.";
		mes "[Hypnotiste]";
		if (.@str$ == "Stat")
			resetstatus;
		else
			resetskill;
		close;
	} else {
		mes "[Hypnotiste]";
		mes "Classe Professionnelle ne correspond pas";
		mes "aux exigences du service de Réinitialisation des Compétences";
		mes "for the "+.@str$+" Reset service";
		mes "professions suivantes...";
		mes "offer "+.@str$+" Resets to the";
		mes "^FF0000Guerrier, Acolyte, Sorcier, Voleur, Artilleur, Marchand, Garçon de Taekwon, Fille de Taekwon, Tireur d'élite et Ninja^000000.";
		next;
		mes "[Hypnotiste]";
		mes "^FF0000Guerrier, Acolyte,";
		mes "Mage, Voleur, Archer,";
		mes "Commerçant, Taekwon";
		mes "Garçon, Fille Taekwon,";
		mes "Gunslinger et Ninja^000000.";
		close;
	}
}
