yuno_in01,124,164,6	script	Guild Leader	108,{
	mes "[Keegan]";
	mes "Qu'est-ce qui vous amène ici ? Avez-vous quelque chose à dire ?";
	next;
	menu "I want to be a Treasure Hunter",-,"Take a New Quest.",N_NQuest,"Take me to the Official Shop.",N_OShop,"Never mind.",N_NVM;
	if(Event_THQS==1) goto N_AlreadyReg;
	mes "[Keegan]";
	mes "Well "+strcharinfo(0)+" if you want to be a Treasure Hunter first you must.";
	mes " ";
	mes "^FF00001.^000000 Vous serez transformé en ^FF000025,000z^000000 pour une formation de base et votre preuve d'appartenance au groupe.";
	mes " ";
	mes "Euh... C'est tout ce que vous avez à faire. Hahaha.";
	next;
	menu "Pay ^FF000025,000z^000000.",-,"Thats way to high!!",N_HighPrice;
	if (Zeny < 25000) goto N_NoZeny;
	set Zeny,Zeny-25000;
	//getitem 7950, 1;
	//getitem 7951, 1;
	set Event_THQS,1;
	mes "[Keegan]";
	mes "Félicitations!";
	emotion ET_CONGRATULATION;
	next;
	mes "[Keegan]";
	mes "Bienvenue dans la Guilde des Chasseurs de Trésors.";
	mes "Vous pouvez maintenant prendre une quête de chasseur de trésor comme vous le souhaitez.";
	close;
N_NoZeny:
	mes "[Keegan]";
	mes "Hmmm you don't seem to have ^FF000025,000z^000000 "+strcharinfo(0)+". Please come back when you do.";
	close;
N_HighPrice:
	mes "[Keegan]";
	mes "Bien sûr, si vous ne pouvez pas vous permettre un petit ^FF000025,000z^000000, vous ne devriez pas être encore un chasseur de trésors.";
	close;
N_AlreadyReg:
	mes "[Keegan]";
	mes "Haha... Vous êtes déjà membre.";
	close;
N_NQuest:
	if (Event_THQS!=1) goto A_NeedReg;
	mes "[Keegan]";
	mes "Well "+strcharinfo(0)+" if you wish to take a quest go talk to the fellow Treasure Hunter in the other room there.";
	close;
A_NeedReg:
	mes "[Keegan]";
	mes "Désolé, vous devez être un membre de la Guilde des Chasseurs de Trésors pour prendre une quête.";
	mes " ";
	mes "Nous avons également maintenant un magasin réservé aux membres qui vous permettra d'utiliser nos boutiques à tout moment que vous le souhaitez. Ils se trouvent tous au rez-de-chaussée.";
	mes " ";
	mes "Nous avons aussi un autre magasin exclusif pour les membres, dans lequel nous utilisons de petits diamants métalliques appelés Jetons Chasseurs de Trésors en échange de biens rares.";
	mes " ";
	mes "Vous aurez même accès à notre donjon personnel exclusif.";
	close;
N_OShop:
	if (Event_THQS!=1) goto N_NeedMem;
	mes "[Keegan]";
	mes "Yes "+strcharinfo(0)+" I will take you to our shop right away!";
	next;
	warp "prt_in",166,171;
	close;
	end;
N_NeedMem:
	mes "[Keegan]";
	mes "Désolé, seuls les membres peuvent visiter notre magasin privé.";
	close;
N_NVM:
	mes "[Keegan]";
	mes "D'accord, revenez quand vous aurez le temps de vous y consacrer.";
	close;
}
