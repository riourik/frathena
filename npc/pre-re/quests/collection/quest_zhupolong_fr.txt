lou_dun02,161,264,4	script	Dragon Hunter	732,{

	if (checkquest(60182) != -1) {
		if (checkquest(60182,HUNTING) != 2 ) {
			mes "[Dragon Hunter]";
			mes "Avez-vous terminé la chasse des 50 Zhu Po Long?";
			next;
			switch(select("Yes:No:I want to quit")) {
			case 1:
				mes "[Dragon Hunter]";
				mes "Hm, je ne pense pas que vous ayez encore exterminé 50 d'entre eux.";
				close;
			case 2:
				mes "[Dragon Hunter]";
				mes "Rappelez-vous, j'ai besoin d'aide pour chasser les Zhu Po Long. Allez donc en tuer 50.";
				close;
			case 3:
				mes "[Dragon Hunter]";
				mes "Êtes-vous sûr de vouloir arrêter la chasse?";
				mes "Tout progrès que vous avez fait sera effacé";
				next;
				switch(select("Yes:No")) {
				case 1:
					mes "[Dragon Hunter]";
					mes "D'accord, revenez ici si vous changez d'avis.";
					erasequest 60182;
					close;
				case 2:
					mes "[Dragon Hunter]";
					mes "Veuillez tuer 50 Zhu Po Long afin que cet endroit devienne un lieu où les joueurs peuvent se rassembler et gagner des Expériences.";
					close;
				}
			}
		}
		mes "[Dragon Hunter]";
		mes "Impressionnant, vous l'avez fait rapidement.";
		getexp 151300,60520;
		erasequest 60182;
		close;
	}
	if ((BaseLevel > 49) && (BaseLevel < 91)) {
		mes "[Dragon Hunter]";
		mes "Oh super!";
		mes "Vous êtes ici pour m'aider, n'est-ce pas?";
		mes "Sinon, je ne sais pas quoi faire avec les monstres dans cet endroit!";
		next;
		switch(select("Sure:No")) {
		case 1:
			mes "[Dragon Hunter]";
			mes "Super! Je savais dès que je vous ai vu que je pouvais compter sur vous!";
			mes "J'ai besoin de réduire la Quantité de monstres qui sont actuellement en train de dévaster cette carte!";
			mes "Il y a trop de Zhu Po Long dans le coin, ce qui rend impossible pour les joueurs de faible niveau d'acquérir une bonne expérience ici.";
			next;
			mes "[Dragon Hunter]";
			mes "Alors, voudriez-vous encore m'aider ?";
			next;
			switch(select("Sure:Cancel")) {
			case 1:
				if (BaseLevel > 90) {
					mes "[Dragon Hunter]";
					mes "Merci beaucoup pour votre aide précieuse avec les Zhu Po Long!";
					mes "Actuellement la Situation est stable et je n'ai plus besoin de votre aide pour les tuer.";
					close;
				}
				mes "[Dragon Hunter]";
				mes "Tuez 50 de ces Zhu Po Long et faites-moi savoir quand vous avez terminé. Je vous récompenserai pour vos efforts.";
				setquest 60182;
				close;
			case 2:
				mes "[Dragon Hunter]";
				mes "^FF0000Si tu changes d'avis, reviens s'il te plaît.";
				close;
			}
		case 2:
			mes "[Dragon Hunter]";
			mes "^FF0000Si tu changes d'avis, reviens s'il te plaît.";
			close;
		}
	}
	mes "[Dragon Hunter]";
	mes "Oh là là, Oh là là!";
	next;
	mes "[Dragon Hunter]";
	mes "^FF0000Désolé, je ne peux pas parler maintenant, je suis dans un monde de douleur ici, ces fichus Zhu Po Long vont me tuer.^[FF0000]";
	close;
}
