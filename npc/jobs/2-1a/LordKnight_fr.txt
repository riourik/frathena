valkyrie,44,39,5	script	Lord Knight#Valkyrie	56,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Seigneur Chevalier]";
			mes "Félicitations.";
			mes "Honneur aux guerriers !";
			close;
		}
		mes "[Seigneur Chevalier]";
		mes "Nous, Chevaliers, avons une";
		mes "responsabilité incroyable...";
		mes "Servir et protéger.";
		next;
		mes "[Seigneur Chevalier]";
		mes "Même au prix de nos vies";
		mes ",";
		mes "nous devons veiller à la";
		mes "santé et au bien-être de nos camarades.";
		close;
	}
	else if (ADVJOB == Job_Lord_Knight && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[Seigneur Chevalier]";
		mes "Votre heure est arrivée !";
		mes "Le monde a toujours besoin de vous.";
		mes "Veuillez continuer votre vie";
		mes "en tant qu'héros avec une nouvelle apparence.";
		next;
		mes "[Seigneur Chevalier]";
		mes "Aimeriez-vous";
		mes "devenir un Lord Knight ?";
		next;
		if (select("No.:Yes.") == 1) {
			mes "[Seigneur Chevalier]";
			mes "[Grand Prêtre]";
			mes "n'hésitez pas à revenir.";
			mes "Honneur aux guerriers !";
			close;
		}
		if (SkillPoint) {
			mes "[Seigneur Chevalier]";
			mes "Il est encore possible pour vous d'apprendre des compétences supplémentaires. Veuillez utiliser";
			mes "tous vos points de compétence restants avant de revenir me voir.";
			close;
		}
		jobchange Job_Lord_Knight;
		set ADVJOB,0;
		mes "[Seigneur Chevalier]";
		mes "Félicitations!";
		mes "En tant que Seigneur Chevalier,";
		mes "j'espère que vous serez";
		mes "en première ligne de bataille,";
		mes "et mènera vos alliés vers la victoire !";
		close;
	}
	mes "[Seigneur Chevalier]";
	mes "Accueillir";
	mes "vers Valhalla,";
	mes "la Salle de l'Honneur.";
	next;
	mes "[Seigneur Chevalier]";
	mes "S'il vous plaît faites-vous";
	mes "confortable";
	mes "pendant votre séjour ici.";
	mes "Honneur aux guerriers !";
	close;
}
