prt_castle,80,170,4	script	King of Prontera	108,{
	mes "[Roi de Prontera]";
	mes "Bonjour, pouvez-vous m'aider ?";
	next;
	if(select("Sure, what can I do?:Sorry, I'm busy.")==2) {
		mes "[Roi de Prontera]";
		mes "D'accord, ça va, je suis sûr qu'un autre aidera.";
		close; }
	mes "[Roi de Prontera]";
	mes "Merci. Eh bien, ma fille à côté est un peu turbulente...";
	mes "Si vous lui apportez ces articles, je suis sûr qu'elle se calmera.";
	mes "Je vous donnerai mon diadème ou mon épée en récompense.";
	next;
	setarray .@Items1[0],5007,742,30,741,50,753,20,732,1,710,20,629,20;
	setarray .@Items2[0],1137,2254,1,2282,1,714,1;
	switch(select("What do you need?:I have the items!:Forget it.")) {
		case 1:
			for(set .@i,1; .@i<3; set .@i,.@i+1) {
				mes "[Roi de Prontera]";
				mes "Si vous pouviez m'apporter :";
				for(set .@j,1; .@j<getarraysize(getd(".@Items"+.@i)); set .@j,.@j+2)
					mes " ~ ^0055FF"+getd(".@Items"+.@i+"["+(.@j+1)+"]")+"^000000x "+getitemname(getd(".@Items"+.@i+"["+.@j+"]"));
				mes "I'll give my "+((.@i == 2 )?"sword.":"crown.");
				if (.@i==2) close;
				next; }
		case 2:
			mes "[Roi de Prontera]";
			mes "Quel article voulez-vous ?";
			next;
			set .@i, select("Crown:Sword");
			for(set .@j,1; .@j<getarraysize(getd(".@Items"+.@i)); set .@j,.@j+2)
				if (countitem(getd(".@Items"+.@i+"["+.@j+"]")) < getd(".@Items"+.@i+"["+(.@j+1)+"]")) set .@nr,1;
			mes "[Roi de Prontera]";
			if (.@nr) { 
				mes "Vous manquez quelque chose.";
				mes "Revenez après avoir obtenu TOUS les articles.";
				close; }
			for(set .@j,1; .@j<getarraysize(getd(".@Items"+.@i)); set .@j,.@j+2)
				delitem getd(".@Items"+.@i+"["+.@j+"]"), getd(".@Items"+.@i+"["+(.@j+1)+"]");
			getitem getd(".@Items"+.@i+"[0]"),1;
			mes "MERCI BEAUCOUP !!";
			mes "Ça va sûrement la calmer.";
			mes "As promised, here is my "+((.@i==1)?"crown.":"sword.");
			close;
		case 3:
			mes "[Roi de Prontera]";
			mes "D'accord, ça va, je suis sûr qu'un autre aidera.";
			close;
	}
}
