rachel,138,146,5	script	Rachel Guide	934,{
	mes "[Rachel Guide]";
	mes "Bienvenue à la capitale de";
	mes "Arunafeltz, Rachel où une douce brise de la déesse Freya souffle.";
	mes "Si c'est votre première utilisation des services du guide, je peux vous indiquer l'emplacement de tout édifice dans Rachel.";
	mes "[Rachel Guide]";
	mes "Cheshrumnir est un lieu sacré où le pape, incarnation de la déesse Freya, réside.";
	mes "don't you check the \"Notice\" menu first?";
	set .@loop1,1;
	while(.@loop1) {
		next;
		switch(select("Village Guide:Remove Marks from Mini-Map:Notice:Cancel")) {
		case 1:
			mes "[Rachel Guide]";
			mes "[Rachel Guide]";
			mes "Alors où voulez-vous aller?";
			if (.@compass_check == 0) {
				mes "Voulez-vous que je marque les emplacements";
				mes "sur votre Mini-Mappe ?";
				mes "^FF0000 sur votre Mini-Map ?";
				next;
				if (select("Yes:No") == 1) set .@compass_check,1;
			}
			set .@loop2,1;
			while(.@loop2) {
				if (.@wait_button_chk == 0) set .@wait_button_chk,1;
				else next;

				switch(select("Cheshrumnir:Inn:Weapon Shop:Tool Shop:Airport:Cancel")) {
				case 1:
					mes "[Rachel Guide]";
					mes "L'édifice à ma gauche est l'auberge de Rachel.";
					mes "[Rachel Guide]";
					if (.@compass_check == 1) viewpoint 1,150,249,1,0xFF0000;
					break;
				case 2:
					mes "[Rachel Guide]";
					mes "Le magasin d'armes se trouve près de la porte occidentale.";
					mes "[Rachel Guide]";
					if (.@compass_check == 1) viewpoint 1,115,149,2,0xFF00FF;
					break;
				case 3:
					mes "[Rachel Guide]";
					mes "Il est situé près de la porte ouest.";
					mes "The weapon shop is located nearby the western gate.";
					if (.@compass_check == 1) viewpoint 1,42,87,3,0x99FFFF;
					break;
				case 4:
					mes "[Rachel Guide]";
					mes "Rachel tool shop sells the best quality potions.";
					mes "It's located nearby the western gate.";
					if (.@compass_check == 1) viewpoint 1,83,78,4,0x0000FF;
					break;
				case 5:
					mes "[Rachel Guide]";
					mes "L'aéroport se trouve à l'extérieur de la porte est.";
					if (.@compass_check == 1) viewpoint 1,273,125,5,0x00FF00;
					break;
				case 6:
					mes "[Rachel Guide]";
					mes "Si vous aimez vous débarrasser de tous les marqueurs de lieu sur votre Minimaps,";
					mes "just ask me again, and choose \"Remove Marks from Mini-Map\" menu.";
					set .@loop2,0;
					break;
				}
			}
			break;

		case 2:
			viewpoint 2,150,249,1,0xFF0000;
			viewpoint 2,115,149,2,0xFF00FF;
			viewpoint 2,42,87,3,0x99FFFF;
			viewpoint 2,83,78,4,0x0000FF;
			viewpoint 2,273,125,5,0x00FF00;
			mes "[Rachel Guide]";
			mes "D'accord, ils sont partis maintenant. Si vous avez plus de lieux à demander, faites-le-moi savoir.";
			break;
		case 3:
			mes "[Rachel Guide]";
			mes "Quand vous utilisez le menu ''Village Guide'',";
			mes "assurez-vous que les emplacements des bâtiments seront marqués sur votre minimap située en haut à droite de l'écran.";
			mes "Si vous ne pouvez pas voir votre minimap, utilisez la touche raccourcie ''ctrl+tab'' ou appuyez sur le bouton ''Carte'' dans vos fenêtres d'informations de base, d'accord?";
			mes "Et vous pouvez également zoomer out sur votre minimap en utilisant le bouton ''-'', au cas où vous ne pourriez pas voir tout le village.";
			break;
		case 4:
			mes "[Rachel Guide]";
			mes "Bonne chance pour votre voyage dans Arunafeltz.";
			mes "à Arunafeltz.";
			close2;
			set .@loop1,0;
			break;
		}
	}
	close;
}
