//=   arg(3): y1 coordinate for warp to treasure room

function	script	F_GldManager	{
	.@npc_name$ = getarg(0);
	.@map_name$ = getarg(1);
	.@gid = getcastledata(.@map_name$,1);
	.@gm_name$ = getguildmaster(.@gid);

	mes "[ "+ .@npc_name$ +" ]";
	if (.@gid == 0){
		mes "Je attends mon maître. Héroïque aventurier, suivez votre destinée !";
		return;
	}
	if (getcharid(2) != .@gid){
		mes "I am here to follow ^5533FF" + getguildmaster(.@gid) + "^000000's command! Hey! Your not even a part of the guild!!";
		mes "Où sont les gardiens ? Détruisons ces intrus !";
		return;
	}
	if (is_guild_leader() == false){
		mes "You're not ^5533FF" + getguildmaster(.@gid) + "^000000! I am here to follow ^5533FF" + getguildmaster(.@gid) + "^000000's command only";
		return 0;
	}

	mes "Welcome Master ^5533FF" + getguildmaster(.@gid) + "^000000 ! I will assist you in any way I can!";
	next;

	menu "Kafra Staff Employment / Dismissal",M_Kaf, "Enter Treasure Room",M_Treas, "Cancel",M_End;

	//===========================
	M_Kaf:
	//======
		mes "[ "+ .@npc_name$ +" ]";
		if (GetCastleData(.@map_name$,9) == 1) goto L_Dismiss;
		if (getgdskilllv(.@gid,10001) == 0){
			mes "Maître, vous n'avez pas de contrat avec la compagnie Kafra.";
			mes "Pour engager un Kafra, il faut d’abord apprendre le métier ^5533FFContrat Avec Kafra^000000.";
			return;
		}

		L_Hire:
			mes "Voulez-vous employer les services d'un Kafra ? Vous aurez besoin de ^5533FF10 000 Zeny^000000 pour cela...";
			next;
			menu "Employ Kafra.",-,"Cancel",sM_KafEnd;

				mes "[ "+ .@npc_name$ +" ]";
				if (Zeny < 10000){
					mes "Maître, vous n'avez pas assez d’argent pour engager un Kafra. L'embauche a été annulée.";
					return;
				}
				set Zeny,Zeny-10000;
				enablenpc "Kafra Staff#"+getarg(4);
				SetCastleData .@map_name$,9,1;
				mes "Vous avez créé un contrat avec la compagnie Kafra Staff.";
				next;
				cutin "kafra_01",2;
				mes "[ Kafra Staff ]";
				mes "Comment allez-vous ? Je suis là pour vous servir ! Je ferai de mon mieux pour vous aider.";
				next;
				cutin "kafra_01",255;
				mes "[ "+ .@npc_name$ +" ]";
				//mes "Votre contrat d'emploi est valable ^5533FF1 mois^000000. Après cette période, il faudra créer un nouveau contrat.";
				mes "Je pense que le Kafra Staff serait bénéfique pour les membres de notre guilde.";
				return;

			sM_KafEnd:
				mes "[ "+ .@npc_name$ +" ]";
				mes "Comme vous l'avez dit Maître. Mais je suggère fortement d'avoir un Kafra dès maintenant !";
				return;

		L_Dismiss:
			mes "Voulez-vous mettre fin au contrat actuel avec le Kafra ?";
			next;
			menu "Dismissal",-,"Cancel",sM_KafEnd2;

				cutin "kafra_01",2;
				mes "[ Kafra Staff ]";
				mes "Ai-je fait quelque chose de mal ? Si tel est le cas, pourriez-vous me pardonner ?";
				next;
				menu "Dismiss",-,"Cancel",ssM_KafEnd2;

					mes "[ Kafra Staff ]";
					mes "C'est dommage que je ne puisse plus servir votre guilde...";
					next;
					disablenpc "Kafra Staff#"+getarg(4);
					SetCastleData .@map_name$,9,0;
					cutin "kafra_01",255;
					mes "[ "+ .@npc_name$ +" ]";
					mes "Le Kafra a été licencié. Mais... il faudrait vraiment avoir un Kafra rapidement !";
					return;
				ssM_KafEnd2:
					mes "[ Kafra Staff ]";
					mes "Merci Maître, je ferai de mon mieux ! ^^. ";
					cutin "kafra_01",255;
					return;
			sM_KafEnd2:
				mes "[ "+ .@npc_name$ +" ]";
				mes "Maître, je pense que vous devriez conserver le poste actuel de Kafra car elle fait déjà de son mieux pour nous servir.";
				return;

	//=========================
	M_Treas:
	//========
		mes "[ "+ .@npc_name$ +" ]";
		mes "Voudriez-vous aller dans notre salle du trésor ? Seul vous, le Maître de Guilde, pouvez entrer dans cette pièce.";
		next;
		menu "Enter Treasure room.",-,"Cancel",sM_TresEnd;

			mes "[ "+ .@npc_name$ +" ]";
			mes "Suivez-moi par le passage secret.";
			mes "Vous devez actionner la poignée secrète pour sortir.";
			next;
			warp .@map_name$,getarg(2),getarg(3);
			return;
		sM_TresEnd:
			mes "[ "+ .@npc_name$ +" ]";
			mes "Les marchandises sont produites quotidiennement.";
			mes "Vous devriez les obtenir dès que possible car elles peuvent disparaître si vous ne les prenez pas au bon moment.";
			return;


	//==========================
	M_End:
	//=======
		mes "[ "+ .@npc_name$ +" ]";
		mes "Comme vous le souhaitez, maître.";
		return;
}
