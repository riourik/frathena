aldeba_in,96,181,4	script	Kafra Service	113,{
	cutin "kafra_05",2;
	mes "[Kafra Leilah]";
	mes "Hum...?";
	mes "Oh, bienvenue à";
	mes "la Corporation Kafra";
	mes "les bureaux principaux. Avez-vous";
	mes "besoin de quelque chose?";
	next;
	switch(select("Save:Use Storage:Rent a Pushcart:Cancel")) {
	case 1:
		mes "[Kafra Leilah]";
		mes "Votre point de renaissance a été enregistré ici, au sein des";
		mes "bureaux principaux de la Corporation Kafra. Merci.";
		mes "[Kafra Leilah]";
		mes "Veuillez vous servir des services Kafra disponibles dans toute";
		next;
		savepoint "aldeba_in",96,179,1,1;
		mes "[Kafra Leilah]";
		mes "Corporation Kafra.";
		mes "[Kafra Leilah]";
		mes "Excusez-moi, mais il semble que vous n'ayez pas les";
		mes "20 zeny nécessaires pour payer.";
		mes "visiting the Kafra Headquarters.";
		close2;
		break;
	case 2:
		if (Zeny < 20) {
			mes "[Kafra Leilah]";
			mes "Excuse me, but it";
			mes "seems that you don't";
			mes "have the 20 zeny to pay";
			mes "La taxe d'accès au Dépôt...";
			close3;
		}
		mes "[Kafra Leilah]";
		mes "Bien que cette installation soit";
		mes "réservée exclusivement à";
		mes "la formation des Employés Kafra";
		mes "et aux fonctions administratives,";
		mes "je vais accéder à votre Dépôt pour vous.";
		next;
		set Zeny, Zeny-20;
		set RESRVPTS, RESRVPTS + (20/5);
		mes "[Kafra Leilah]";
		mes "À l'avenir, demandez un Employé Kafra de";
		mes "service si vous souhaitez utiliser";
		mes "l'un des services Kafra.";
		mes "any of the Kafra Services.";
		mes "Merci pour votre mécénat.";
		callfunc("F_CheckKafCode");    //check your storage password, if set
		close2;
		openstorage;
		break;
	case 3:
		mes "[Kafra Leilah]";
		mes "Je suis désolée, mais je ne suis pas en service. Je vous aiderais si j'en avais la possibilité,";
		mes "mais en réalité, je n'ai actuellement aucun chariot disponible.";
		mes "if I could, but actually don't";
		mes "Pourquoi ne demandez-vous pas à un autre Employé Kafra de vous apporter assistance?";
		mes "Why don't you ask another Kafra";
		mes "[Kafra Leilah]";
		next;
		mes "[Kafra Leilah]";
		mes "Les employés de Kafra sont";
		mes "stationnés un peu partout dans";
		mes "le continent de Midgard,";
		mes "et vous devriez pouvoir en trouver";
		mes "plusieurs à l'extérieur d’Al De Baran.";
		close2;
		break;
	case 4:
		mes "[Kafra Leilah]";
		mes "^666666*Whew...*^000000";
		mes "Parfait, car je suis";
		mes "en fait en pause";
		mes "pour l'instant. De choisir";
		mes "''Annuler'' était une bonne";
		mes "décision de votre part.";
		close2;
		break;
	}
	cutin "",255;
	end;
}
