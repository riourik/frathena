		mes "Nous avons de nombreux articles, alors jetez-y un œil et achetez délibérément.";
		close2;
		readbook 11010,1;
		end;
	}
	set .@menu$,"";
	if (.@menu1$ != "") set .@menu$, .@menu1$;
	else for(set .@i,0; .@i<getarraysize(.@Items); set .@i,.@i+2)
		set .@menu$, .@menu$+getitemname(.@Items[.@i])+":";
	set .@i, (select(.@menu$)-1)*2;
	mes "[Erundek]";
	mes "You chose ^3131FF"+getitemname(.@Items[.@i])+"^000000.";
	switch(.@Items[.@i]) {
		case 2720: mes "Cet article est uniquement destiné aux Master Class Swordman et Taekwon."; break;
		case 2721: mes "Cet objet est réservé à la classe Voleur."; break;
		case 2722: mes "Cet objet est réservé à la classe Acolyte."; break;
		case 2723: mes "Cet objet est réservé à la classe de magicien."; break;
		case 2724: mes "Cet article est uniquement destiné à la classe Archer."; break;
		case 2725: mes "Cet objet est réservé à la classe marchande."; break;
		case 2733: mes "Cet article est uniquement destiné à Gunslinger."; break;
		default: break;
	}
	mes "You can exchange for this item with ^FF0000"+.@Items[.@i+1]+" "+getitemname(7828)+" or "+.@Items[.@i+1]+" "+getitemname(7829)+"^000000.";
	mes "Vous souhaitez échanger ?";
	next;
	switch(select("Do not exchange:Exchange")) {
	case 1:
		mes "[Erundek]";
		mes "Avez-vous besoin de plus de temps pour vérifier les articles ?";
		break;
	case 2:
		mes "[Erundek]";
		mes "Quel badge souhaitez-vous échanger ?";
		mes "You need ^3131FF"+.@Items[.@i+1]+" Badges^000000 to exchange.";
		next;
		if (.@item[0] < 12269 || .@item[0] > 12273 ) {
			mes "[Erundek]";
			mes "N'oubliez pas que les objets de récompense du champ de bataille sont ^FF0000Character Bound^000000. Êtes-vous sûr de vouloir cet article ?";
			next;
		}
		set .@j, select("Bravery Badge:Valor Badge:Cancel");
		mes "[Erundek]";
		if (.@j == 3) {
			mes "Vous avez annulé l'échange.";
			break;
		}
		set .@cost, ((.@j==1)?7828:7829);
		if (countitem(.@cost) >= .@Items[.@i+1]) {
			mes "Merci pour l'échange.";
			delitem .@cost, .@Items[.@i+1];
			getitem .@Items[.@i],1;
		}
		else mes "You do not have enough "+getitemname(.@cost)+"s.";
		break;
	}
	close;
}

// Badge Reward Function
//============================================================
function	script	F_BG_Badge	{
	.@team$ = getarg(1);
	.@bg_type$ = getarg(2);

	if (.@bg_type$ == "Tierra") {
		set .@badge,7828; //BF_Badge1
		set .@amount_win,3;
		set .@amount_lose,1;
	} else if (.@bg_type$ == "Flavius") {
		set .@badge,7829; //BF_Badge2
		set .@amount_win,9;
		set .@amount_lose,3;
	} else
		return;

	if (getarg(0) == 1) {
		set .@amount, .@amount_win;
		if (.@team$ == "Guillaume") {
			mes "[Axl Rose]";
			mes "Bienheureux Guillaume !";
		} else if (.@team$ == "Croix") {
			mes "[Swanderie]";
			mes "Bienheureuse Croix !";
		}
		mes "Profitons de notre glorieuse victoire !";
		mes strcharinfo(0)+", it's a sign reflecting victory.";
	} else {
		set .@amount, .@amount_lose;
		if (.@team$ == "Guillaume") {
			mes "[Axl Rose]";
			mes "Vous avez perdu, mais vous êtes dévoué à cette bataille.";
			mes "C'est une récompense pour votre grand dévouement de la part de Guillaume Marollo !";
			mes "Prenez simplement cette défaite comme une leçon, et la prochaine fois, vous gagnerez certainement.";
		} else if (.@team$ == "Croix") {
			mes "[Swanderie]";
			mes "Oh, "+strcharinfo(0)+" Don't be sad.";
			mes "Même si nous n'avons pas gagné, nous avons fait de notre mieux.";
			mes "C'est un cadeau royal de Croix, et n'oubliez pas cette bataille. Nous gagnerons le prochain.";
		}
	}
	close2;

	// If the VIP system is enabled, VIP players receive an extra 2 badges.
	if (VIP_SCRIPT && vip_status(VIP_STATUS_ACTIVE))
		set .@amount, .@amount+2;

	set .@medal_gap, 500 - countitem(.@badge);
	if (.@medal_gap >= .@amount)
		getitem .@badge,.@amount;
	else
		getitem .@badge,.@medal_gap;
	return;
}
