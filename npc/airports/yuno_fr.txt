y_airport,143,43,5	script	Airport Staff#y_air1a::y_airport1	90,{
	mes "[Personnel de l'aéroport]";
	mes "Bienvenue à l'aéroport Juno où nous proposons des vols intérieurs vers Einbroch, Lighthalzen et Hugel,";
	mes "et vols internationaux vers Izlude et Rachel.";
	mes "Comment puis-je être utile ?";
	next;
	if (select("Board the Airship.:Cancel.") == 1) {
		mes "[Personnel de l'aéroport]";
		mes "Les frais d'embarquement pour tous";
		mes "les vols coûtent 1 200 zeny. Si tu";
		mes "utilisez un billet gratuit pour un dirigeable,";
		mes "les frais d'embarquement seront supprimés. Alors, vous souhaitez partir ?";
		next;
		if (select("Yes:No") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "y_airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "y_airport",148,51;
				end;
			}
			mes "[Personnel de l'aéroport]";
			mes "Je suis désolé, mais tu ne le fais pas";
			mes "j'ai 1 200 zeny à payer";
			mes "pour les frais d'embarquement.";
			close;
		}
	}
	mes "[Personnel de l'aéroport]";
	mes "Merci et";
	mes "passe une bonne journée.";
	close;
}

	mes "Montez à bord du dirigeable au départ pour atteindre votre destination prévue.";

		mes "[Personnel de l'aéroport]";
		mes "Une fois dans le terminal principal, vous devez payer à nouveau les frais.";
		mes "monter à bord d'un dirigeable au départ. Vous ne devez quitter que si votre intention est";
		mes "la destination est Junon. Passer à";
		mes "sortie vers le terminal principal ?";
		next;
		if (select("Yes:No") == 1) {
			warp "y_airport",142,40;
			end;
		}
	}
	mes "[Personnel de l'aéroport]";
	mes "Très bien, merci";
	mes "pour votre mécénat";
	mes "et j'espère que tu as";
	mes "un vol agréable~";
	close;
}
