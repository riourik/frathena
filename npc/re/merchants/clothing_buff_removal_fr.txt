//===== rAthena Script =======================================
//= Clothing Effect Removal Service
//===== Description: =========================================
//= Service to remove clothing effects like Xmas, Wedding, ...
//===== Changelog: ===========================================
//= 1.0 Initial version. [Balferian, Everade, Lemongrass]
//============================================================

prontera,80,106,5	script	Clothing Effect Removal Service	4_EP16_COOK2,{
	mes "[Derek Shot]";
	mes "Êtes-vous incapable de participer au combat à cause des effets du vêtement ? Cela doit être ennuyeux.";
	next;
	mes "[Derek Shot]";
	mes "Bien qu'il ait l'air bien sur vous, il peut aussi devenir vraiment ennuyeux. Je peux m'en débarrasser avec un seul geste de la main !! Je vais le ranger soigneusement. Si vous êtes prêt(e), choisissez votre option !";
	next;
	switch( select( "Keep it", "Release it immediately" ) ){
		case 1:
			mes "[Derek Shot]";
			mes "Revenez vers moi si vous changez d'avis.";
			close;
		case 2:
			sleep2 500;
			specialeffect2 EF_SPHERE;
			sleep2 500;
			specialeffect2 EF_SPHERE;
			sleep2 500;
			specialeffect2 EF_SPHERE;
			mes "[Derek Shot]";
			mes "La suppression est terminée. Assurez-vous de pouvoir bouger librement maintenant.";
			atcommand "@changedress";
			close;
	}
}
