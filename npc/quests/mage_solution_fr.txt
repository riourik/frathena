pay_arche,122,100,0	script	Dollshoi	88,{
	mes "[Mage Guildman]";
	mes "Vous voulez une solution ? Hmm, donne-moi 50 Zeny et au moins 1 tube à essai vide.";
	next;
	if (select("Alright, Deal.:Nah, forget it.") == 1) {
		mes "[Mage Guildman]";
		if (Zeny < 50) {
			mes "Hé! Vous n'avez pas assez d'argent pour couvrir mes frais de 50 Zeny.";
			close;
		}
		if (countitem(1092) == 0) {
			mes "Vous ne pouvez pas transporter de solutions sans bouteille ! Apportez-moi un tube à essai vide.";
			close;
		}
		delitem 1092,1; //Empty_Cylinder
		set Zeny, Zeny-50;
		getitem 1089,1; //Payon_Potion
	}
	close;
}
