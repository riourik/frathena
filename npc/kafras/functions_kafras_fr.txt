//=== Main Function ========================================================
//=   arg(0): Used to determine which welcome message to show.
//=   arg(1): Used to determine which menu to display.
//=   arg(2): Used to determine if the info menu is shown in F_KafInfo.
//=   arg(3): Cost to use Storage
//=   arg(4): Cost to Rent a Pushcart
//==========================================================================
function	script	F_Kafra	{
	callfunc "F_ClearGarbage"; // Clear outdated, unused variables
	.@welcome = getarg(0);
	.@menu_num = getarg(1);

	// Display Kafra Welcome Message
	switch(.@welcome) {
	default:// Default message (obsolete)
		mes "[Employé de Kafra]";
		mes "Bienvenue au";
		mes "Société Kafra.";
		mes "Les services Kafra";
		mes "sont toujours de votre côté.";
		mes "Comment puis-je vous aider?";
		break;
	case 1:// Niflheim specific message
		mes "[Employé de Kafra]";
		mes "^666666W-welc-c-come";
		mes "à th-th-le K-kaaafrrrra";
		mes "C-coorpoor-r-ratioooonn...^000000";
		break;
	case 2:// Guild Castle Kafra message
		mes "[Employé de Kafra]";
		mes "Welcome. ^ff0000" + getguildname( getcharid(2) ) + "^000000 Member.";
		mes "La Corporation Kafra vous accompagnera où que vous alliez.";
		break;
	case 3:// Amatsu specific message (obsolete)
		mes "[Employé de Kafra]";
		mes "Alors, venez-vous d'un pays lointain pour étudier notre culture ou faites-vous simplement du tourisme?";
		mes "Dans les deux cas, pourquoi ne pas rester un peu?";
		mes "L'air est constamment chargé de";
		mes "l'odeur agréable des fleurs sauvages.";
		break;
	case 4:// Luoyang and Ayothaya specific message (obsolete)
		mes "[Employé de Kafra]";
		mes "Avec nos nombreux points de service Kafra,";
		mes "vous n'êtes jamais";
		mes "loin de chez vous.";
		break;
	case 5:// NPC has it's own welcome message. (Display nothing)
		break;
	}
	next;
	if (.@welcome == 2)// Guild Kafra Menu override (free Teleport, free Storage, Free Cart)
		setarray .@K_Menu0$[0], "Use Storage", "Use Guild Storage", "Rent a Pushcart", "Use Teleport Service", "Cancel";
	else {
		switch(.@menu_num) {
			// Save and Storage only
			case 1:	setarray .@K_Menu0$[0],"Save","Use Storage","Cancel"; break;
			// Storage only
			case 2:	setarray .@K_Menu0$[0],"Use Storage","Cancel"; break;
			// No Teleport (Common)
			case 3: setarray .@K_Menu0$[0],"Save","Use Storage","Rent a Pushcart","Check Other Information","Cancel"; break;
			// Case 4 is Einbroch no tele message.
			// No save, or teleport. (Common)
			case 5:	setarray .@K_Menu0$[0],"Use Storage","Rent a Pushcart","Check Other Information","Cancel"; break;
			// Storage and Check Other Information only.
			case 6:	setarray .@K_Menu0$[0],"Use Storage","Check Other Information","Cancel"; break;
			// Save, Storage, and Pushcart only (Kafra Warehouse)
			case 7:	setarray .@K_Menu0$[0],"Save","Use Storage","Rent a Pushcart","Cancel"; break;
			// Save, Storage, Other Check information. (Turbo track)
			case 8: setarray .@K_Menu0$[0],"Save","Use Storage","Check Other Information","Cancel"; break;
			// No Save (Rune Knight)
			case 9: setarray .@K_Menu0$[0],"Use Storage","Rent a Pushcart","Use Teleport Service","Check Other Information","Cancel"; break;
			// Storage, Save, and Pushcart (Dewata, reorder of case 7)
			case 10: setarray .@K_Menu0$[0],"Use Storage","Save","Rent a Pushcart","Cancel"; break;
			// Default message (obsolete)
			default: setarray .@K_Menu0$[0],"Save","Use Storage","Use Teleport Service","Rent a Pushcart","Check Other Information","Cancel"; break;
		}
	}
	.@menu$ = implode(.@K_Menu0$,":");
	while (true) {
		.@j = select(.@menu$) - 1;
		if (.@K_Menu0$[.@j] == "Save")
			return;
		else if (.@K_Menu0$[.@j] == "Use Storage") {
			if (.@welcome == 2)// Do not charge for Guild Storage
				callfunc "F_KafStor",2,0,0;
			else
				callfunc "F_KafStor",0,getarg(3),.@welcome;
			next;
		}
		else if (.@K_Menu0$[.@j] == "Use Teleport Service") {
			if (.@menu_num != 4)// Display Einbroch "No Teleport Service" notice.
				callfunc "F_KafTele",.@welcome;
			else {
				mes "[Employé de Kafra]";
				mes "En raison du ^FF0000Accord de Transport Limité^000000, la";
				mes "Corporation Kafra ne peut pas";
				mes "offrir de Services Téléportation";
				mes "dans la République du Schwarzwald.";
				mes "Nous vous demandons de bien vouloir";
				next;
				mes "[Employé de Kafra]";
				mes "utiliser le Service Aéroplane.";
				mes "use the Airship Service";
				mes "au lieu de cela. Merci pour votre";
				mes "compréhension et coopération.";
			}
			next;
		}
		else if (.@K_Menu0$[.@j] == "Rent a Pushcart") {
			if (callfunc("F_KafCart",.@welcome,getarg(4)) == 1)
				next;
		}
		else if (.@K_Menu0$[.@j] == "Check Other Information") {
			callfunc "F_KafInfo",getarg(2);
			next;
		}
		else if (.@K_Menu0$[.@j] == "Cancel") {
			callfunc "F_KafEnd",.@welcome,0;
			end;
		}
		else if (.@K_Menu0$[.@j] == "Use Guild Storage") {
			callfunc "F_KafStor",1,0;
			next;
		}
	}
}

//===================================================
// Storage Function
//===================================================
function	script	F_KafStor	{
	.@type = getarg(0);
	.@fee = getarg(1);

	// Unable to access Guild Storage (Busy)
	if (.@type == 1) {
		if (guildopenstorage()) {
			mes "[Employé de Kafra]";
			mes "Désolé mais un autre membre du guilde utilise le stockage du guilde";
			mes "en ce moment. Veuillez attendre que cette personne termine.";
			close3;
		}
		cutin "", 255;
		close;
	}
	// Unable to access Normal Storage (Insufficient Basic Skills)
	if (!callfunc("F_CanOpenStorage")) {
		mes "[Employé de Kafra]";
		// Niflheim Specific Message
		if (getarg(2) == 1) {
			mes "^666666S-s-ssoooorry,";
			mes "y-you're a-a-aaaa";
			mes "Noooviiice... N-neeeds";
			mes "B-basic sssskill l-level 6...^000000";
			return;
		}
		mes "Je suis désolé, mais toi";
		mes "j'ai besoin du statut d'apprenti";
		mes "Niveau de compétence basique 6 pour";
		mes "utiliser le service de stockage.";
		return;
	}
	// Accessing Normal Storage (Skipped if accessing Storage from Guild castle)
	if (.@type != 2) {
		// Consume "Free Ticket for Kafra Storage" if available.
		if (countitem(7059))
			delitem 7059,1;
		else {
			if (Zeny < .@fee) {
				mes "[Employé de Kafra]";
				// Niflheim Specific Message
				if (getarg(2) == 1) {
					percentheal -50,-50;
					mes "^666666Zeeeeeny...";
					mes "M-more z-zeny...!";
					mes "N-neeed 150... zeny...";
					mes "Ergh! T-taking bl-blood~!^000000";
					return;
				}
				// Standard Message
				mes "Je suis désolé, mais tu ne le fais pas";
				mes "a assez de zenny pour utiliser";
				mes "le Service de Stockage. Notre";
				mes "Storage access fee is " + .@fee + " zeny.";
				return;
			}
			Zeny -= .@fee;
			RESRVPTS += (.@fee/5);
		}
	}

	// Surpress the "End" message if in guild castle.
	if (.@type != 2) {
		// Niflheim specific Message
		if (getarg(2) == 1) {
			percentheal 0,-10;
			mes "[Employé de Kafra]";
			for ( .@i = 0; .@i < 5; .@i++ )
				mes "^666666Merci.. pour... utiliser...^000000";
		}
		// Normal message
		else {
			mes "[Employé de Kafra]";
			mes "Voici, laissez-moi ouvrir";
			mes "votre espace de stockage pour vous.";
			mes "Merci d'avoir utilisé";
			mes "le service Kafra.";
		}
	}
	callfunc("F_CheckKafCode");	//check your storage password, if set
	close2;
	openstorage;
	cutin "", 255;
	end;
}

//===================================================
// Teleport Function
//===================================================
function	script	F_KafTele	{
	mes "[Employé de Kafra]";
	mes "Veuillez choisir";
	mes "votre destination.";
	next;
	.@j = select( implode(@wrpC$,":") ) - 1;
	if (@wrpC$[.@j] == "Cancel")
		return;
	// Consume "Free Ticket for Kafra Transportation" if available.
	// Do not consume if in Guild castle
	if (countitem(7060) > 0 && getarg(0) != 2)
		delitem 7060,1;
	else {
		if (Zeny < @wrpP[.@j]) {
			mes "[Employé de Kafra]";
			mes "Désolé, mais vous n'avez pas";
			mes "assez d'argent (zeny) pour le service Téléportation.";
			mes "Le frais de téléportation est";
			mes "to " + @wrpD$[.@j] + " is " + @wrpP[.@j] + " zeny.";
			close3;
		}
		Zeny -= @wrpP[.@j];
		RESRVPTS += (@wrpP[.@j]/16);
	}
	cutin "", 255;
	if (@wrpD$[.@j] == "Al De Baran") warp "aldebaran",168,112;
	else if (@wrpD$[.@j] == "Alberta") warp "alberta", 117, 56;
	else if (@wrpD$[.@j] == "Comodo") warp "comodo", 209, 143;
	else if (@wrpD$[.@j] == "Comodo Pharos Beacon") warp "cmd_fild07", 127, 134;
	else if (@wrpD$[.@j] == "Geffen") warp "geffen", 120, 39;
	else if (@wrpD$[.@j] == "Izlude") {
		if (checkre(0) == true)
			warp "izlude", 128, 98;
		else
			warp "izlude", 91, 105;
	}
	else if (@wrpD$[.@j] == "Juno") warp "yuno", 158, 125;
	else if (@wrpD$[.@j] == "Mjolnir Dead Pit") warp "mjolnir_02", 99, 351;
	else if (@wrpD$[.@j] == "Morocc") warp "morocc", 156, 46;
	else if (@wrpD$[.@j] == "Orc Dungeon") warp "gef_fild10", 52, 326;
	else if (@wrpD$[.@j] == "Payon") warp "payon", 161, 58;
	else if (@wrpD$[.@j] == "Prontera") warp "prontera", 116, 72;
	else if (@wrpD$[.@j] == "Umbala") warp "umbala", 100, 154;
	end;
}


//===================================================
// Cart Function
//===================================================
function	script	F_KafCart	{
	// Ensure that the class wanting to rent a pushcart is a merchant
	if (BaseClass != Job_Merchant) {
		mes "[Employé de Kafra]";
		mes "^FF0000Je suis désolé, mais le";
		mes "service de location de chariot n'est disponible que pour les";
		if (checkre(0) == true)
			mes "est disponible uniquement pour les classes Commerçant.";
		else {
			mes "[Marchands], [Forgemestres], [Maîtres Forgemestres], [Alchimistes], [Bio-Alchimistes],";
			mes "^FF0000et [Mécaniciens] ainsi que [Généticiens].";
			mes "Alchimistes et Bio-Alchimistes.";
		}
		return 1;
	}
	// Make sure the invoking character does not have a cart already
	else if (checkcart() == true) {
		mes "[Employé de Kafra]";
		mes "Malheureusement, nous ne pouvons pas";
		mes "un Caddie équipé.";
		mes "Malheureusement, nous ne pouvons pas";
		mes "louer plus d'un à";
		mes "à la fois par client.";
		return 1;
	}
	// Check if the player has the Pushcart skill
	else if (getskilllv("MC_PUSHCART") == 0) {
		mes "[Employé de Kafra]";
		mes "You can only rent a cart after learning the \"Push Cart\" skill.";
		return 1;
	}
	// Consume "Free Ticket for the Cart Service" if available.
	// Do not consume if in Guild castle.
	if (countitem(7061) > 0 && getarg(0) != 2)
		delitem 7061,1;
	else {
		.@rental_fee = getarg(1);
		mes "[Employé de Kafra]";
		mes "La location du landau de marchandises";
		mes "fee is " + .@rental_fee + " zeny. Would";
		mes "vous devez payer une redevance pour le landau de marchandises";
		next;
		if (select("Rent a Pushcart.:Cancel") == 2)
			return 0;
		if (Zeny < .@rental_fee) {
			mes "[Employé de Kafra]";
			mes "Je suis désolé, mais toi";
			mes "je n'en ai pas assez";
			mes "de 800 zeny.";
			mes "rental fee of " + .@rental_fee + " zeny.";
			return 1;
		}
		Zeny -= .@rental_fee;
		RESRVPTS += 48;
	}
	setcart;
	return 1;
}

//===================================================
// Check Storage Password Function
//===================================================
function	script	F_CheckKafCode	{
	if (!#kafra_code)
		return;
	mes "Entrez le mot de passe de votre coffre :";
	input .@code_;
	if (.@code_ != #kafra_code) {
		dispbottom "Mot de passe incorrect.";
		close3;
	}
	@kafcode_try = 0;
	return;
}

// === Set / Change / Clear Storage Password Function ===
// getarg(0): NPC Name
// getarg(1): Company Name
//=======================================================
function	script	F_SetKafCode	{
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	if (#kafra_code == 0) {
		mes "" + .@comp_name$ + " proudly presents you a new service:";
		mes "Protection supplémentaire du coffre avec un mot de passe.";
		next;
		if (select( "Set new password -> 5000z", "Cancel" ) == 1)
			callsub( S_SET, .@npc_name$, .@comp_name$ );
		close3;
	}
	mes "Votre coffre est protégé par un mot de passe. Que souhaitez-vous faire maintenant ?";
	next;
	switch( select( "Change old password -> 5000z", "Remove storage password -> 1000z", "Cancel" ) ) {
	case 1:
		mes .@npc_name$;
		mes "D'abord, entrez votre ^0000FFancien mot de passe^000000.";
		.@code = callfunc("F_EntKafCode");
		if (!.@code || .@code != #kafra_code) {
			mes "Mot de passe incorrect. Vous ne pouvez pas définir un nouveau mot de passe.";
			emotion ET_SCRATCH;
			break;
		}
		next;
		callsub( S_SET, .@npc_name$, .@comp_name$ );
		break;
	case 2:
		mes .@npc_name$;
		mes "Veuillez entrer votre mot de passe avant sa suppression.";
		.@code = callfunc("F_EntKafCode");
		if (!.@code) {
			mes "Le mot de passe n'a pas été supprimé.";
			emotion ET_SCRATCH;
			break;
		}
		next;
		mes .@npc_name$;
		if (Zeny < 1000) {
			mes "Vous n'avez pas assez de zeny.";
			emotion ET_MONEY;
			break;
		}
		Zeny -= 1000;
		if (.@code == #kafra_code) {
			#kafra_code = 0;
			mes "Vous avez nettoyé avec succès le mot de passe de votre coffre.";
			mes "Thank you for using " + .@comp_name$ + ".";
			emotion ET_THANKS;
		} else {
			mes "Mot de passe incorrect. Nous ne vous rembourserons pas vos 1000z.";
			mes "Veuillez entrer correctement votre mot de passe la prochaine fois.";
			emotion ET_SORRY;
		}
		break;
	case 3:
		break;
	}
	close3;

S_SET:
	.@npc_name$ = getarg(0);
	.@comp_name$ = getarg(1);

	mes .@npc_name$;
	mes "Maintenant, entrez votre ^FF0000nouveau mot de passe^000000 pour protéger votre coffre des voleurs.";
	.@code = callfunc("F_EntKafCode");
	if (!.@code) {
		mes "Le mot de passe n'a pas été modifié.";
		emotion ET_SCRATCH;
		return;
	}
	next;
	mes .@npc_name$;
	if (Zeny < 5000) {
		mes "Vous n'avez pas assez de zeny.";
		emotion ET_MONEY;
		return;
	}
	Zeny -= 5000;
	#kafra_code = .@code;
	mes "Vous avez sécurisé votre coffre avec un mot de passe secret.";
	mes "Thank you for using " + .@comp_name$ + ".";
	emotion ET_THANKS;
	return;
}

//===================================================
// Basic Password Validation Function
//===================================================
function	script	F_EntKafCode	{
	mes "Entrez un numéro de 1000~10000000 :";
	@kafcode_try += 1;
	if (@kafcode_try > 10) {
		@kafcode_try = 0;
		logmes "Essai d'ajuster le mot de passe du coffre-fort.";
		return 0;
	}
	if (input(.@code_) == 1) {
		mes "Vous ne pouvez pas utiliser un tel grand mot de passe.";
		return 0;
	}
	if (.@code_ < 1000) {
		mes "Vous ne devriez pas utiliser un tel court mot de passe.";
		return 0;
	}
	return .@code_;
}
