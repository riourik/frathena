valkyrie,44,42,5	script	High Priest#Valkyrie	60,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[High Priest]";
			mes "Félicitations.";
			mes "Honneur aux guerriers !";
			close;
		}
		mes "[High Priest]";
		mes "Par le pouvoir";
		mes "de la sainteté, que nous";
		mes "trouvions paix, force";
		mes "et protection. Débarrassez-nous";
		mes "des forces du mal...";
		close;
	}
	else if (ADVJOB == Job_High_Priest && Class == Job_Acolyte_High && JobLevel > 39) {
		mes "[High Priest]";
		mes "Notre monde a besoin de";
		mes "personnes talentueuses et convaincues.";
		mes "Continuez vos bonnes actions en tant que";
		mes "héros de la sainteté encore plus grands...";
		mes "[Grand Prêtre]";
		mes "vouloir devenir un Grand Prêtre?";
		next;
		mes "[High Priest]";
		mes "Aimeriez-vous";
		mes "Lorsque vous serez prêt,";
		next;
		if (select("No.:Yes.") == 1) {
			mes "[High Priest]";
			mes "[Grand Prêtre]";
			mes "feel free to come back.";
			mes "Honneur aux guerriers !";
			close;
		}
		if (SkillPoint) {
			mes "[High Priest]";
			mes "Il est encore possible pour vous d'apprendre des compétences supplémentaires. Veuillez utiliser";
			mes "tous vos points de compétence restants avant de revenir me voir.";
			close;
		}
		jobchange Job_High_Priest;
		set ADVJOB,0;
		mes "[High Priest]";
		mes "Félicitations.";
		mes "En tant que Grand Prêtre,";
		mes "j'espère que vous guiderez";
		mes "les autres sur le chemin";
		mes "de la sainteté...";
		close;
	}
	mes "[High Priest]";
	mes "Accueillir";
	mes "vers Valhalla,";
	mes "la Salle de l'Honneur.";
	next;
	mes "[High Priest]";
	mes "S'il vous plaît faites-vous";
	mes "confortable";
	mes "pendant votre séjour ici.";
	mes "Honneur aux guerriers !";
	close;
}
