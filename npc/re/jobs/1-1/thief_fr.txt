moc_prydb1,39,129,2	script	Thief Guide#thief	69,{
	if (Upper == 1) {
		if (ADVJOB == Job_Assassin_Cross || ADVJOB == Job_Stalker) {
			if (Class == Job_Novice_High) {
				mes "[Guide Voleur]";
				mes "Hé ? Je te connais ? C'est flippant que tu sembles si familier. Tu n'as pas de jumeau, hein ?";
				next;
				if (!callfunc("F_CanChangeJob")) {
					mes "[Guide Voleur]";
					mes "Ah, veux-tu vraiment être un Voleur ? Désolé, mais tu as l'air d'avoir besoin de plus d'entraînement.";
					next;
					mes "[Guide Voleur]";
					mes "Prends ton temps et apprends tous les Compétences de Base, d'accord ? Eh bien, à bientôt~!";
					close;
				}
				mes "[Guide Voleur]";
				mes "Bien, j'ai l'impression que tu as vécu une vie entière en combattant, alors je te promeus immédiatement au rang de Voleur. Il va falloir que je t'offre ce qu'un vrai dur à cuire veut...";
				next;
				skill "NV_TRICKDEAD",0,SKILL_PERM;
				jobchange Job_Thief_High;
				skill "TF_SPRINKLESAND",1,SKILL_PERM;
				skill "TF_BACKSLIDING",1,SKILL_PERM;
				skill "TF_PICKSTONE",1,SKILL_PERM;
				skill "TF_THROWSTONE",1,SKILL_PERM;
				mes "[Guide Voleur]";
				mes "Maintenant que tu es un Voleur, vis comme un Voleur. Allez, vas-y ! Suivant~";
				close;
			}else{
				mes "[Guide Voleur]";
				if (Sex == SEX_MALE)
					mes "Hey, mec.";
				else
					mes "Hey, bébé~";
				close;
			}
		}else{
			mes "[Guide Voleur]";
			if (Sex == SEX_MALE)
				mes "Hey, mec.";
			else
				mes "Hey, bébé.";
			mes "...Hey ! Tu as l'air trop gentil pour vouloir être un Voleur!! Maintenant casse-toi, je suis occupé. Suivant !";
			close;
		}
	}
	if(q_job_thief == 1) {
		mes "[Homme de la Guilde des Voleurs]";
		mes "Ehh... Seems like a beloved "+(Sex == SEX_MALE?"son":"daughter")+" of a rich family, but why are you here in this dirty place?";
		next;
		mes "[Homme de la Guilde des Voleurs]";
		mes "Eh bien, je ne suis pas chargé de vous faire devenir Voleur. Je n'accepte simplement les candidatures, c'est compris ?";
		mes "Si vous voulez devenir Voleur, demandez à l'homme aux yeux perçants qui se trouve à côté de moi.";
		close;
	}
	mes "[Homme de la Guilde des Voleurs]";
	if(BaseJob != Job_Novice) {
		if(BaseJob == Job_Thief) {
			mes "Hey~ si vous avez des problèmes, venez me voir n'importe quand, hein ?";
			close;
		}else{
			mes "^FF0000Quoi donc...?";
			if(Sex == SEX_MALE) {
				mes "Huh... you are a "+jobname(Class)+", blockhead!";
				next;
				mes "[Homme de la Guilde des Voleurs]";
				mes "Hey, frère.";
			}else{
				next;
				mes "[Homme de la Guilde des Voleurs]";
				mes "Hey, dame.";
			}
			mes "Pourquoi êtes-vous ici ? Retournez chez vous~ retournez~~";
			close;
		}
	}
	mes "Ehh... Seems like a beloved "+(Sex == SEX_MALE?"son":"daughter")+" of a rich family, but why are you here in this dirty place?";
	next;
	if(select("I want to be a Thief.:Nothing.") == 2) {
		mes "[Homme de la Guilde des Voleurs]";
		mes "Vous savez qu'on ne peut pas être voleur sans une candidature...";
		mes "Qu'est-ce qui vous occupe l'esprit... ?";
		close;
	}
	mes "[Homme de la Guilde des Voleurs]";
	mes "Eh bien, êtes-vous si fier de cela ?";
	mes "Vous me le dites avec tant de fierté que vous voulez être Voleur ! Pourquoi ne pas aller dans tous les villages et faire la publicité pour votre métier de voleur ?";
	mes "'Ha ha ha ! Allez afficher une pancarte disant : 'Je serai un fier voleur qui dérobera les affaires des autres.'";
	next;
	mes "[Homme de la Guilde des Voleurs]";
	mes "Vous voulez tellement devenir voleur ?";
	next;
	switch(select("Yes.:No.:How about you?")) {
	case 1:
		mes "[Homme de la Guilde des Voleurs]";
		mes "Oh, vous... ? Euh... bon... Je vis et j'apprends à voir des gens étranges comme vous.";
		next;
		break;
	case 2:
		mes "[Homme de la Guilde des Voleurs]";
		mes "Alors pourquoi êtes-vous ici ? Pensez-vous pouvoir devenir voleur si facilement ?";
		next;
		break;
	case 3:
		mes "[Homme de la Guilde des Voleurs]";
		mes "Eh.. ? moi ? moi ?";
		mes "Bon... je me suis adapté au métier de voleur... typiquement... Je ne me soucie pas de cette bêtise.";
		next;
		break;
	}
	mes "[Homme de la Guilde des Voleurs]";
	mes "De toute façon, dans le monde extérieur, n'avez jamais dit que vous vouliez être un Voleur !!";
	next;
	mes "[Homme de la Guilde des Voleurs]";
	mes "Alors, voulez-vous postuler pour devenir Voleur ?";
	next;
	if(select("Yes, I do.:No.") == 2) {
		mes "[Homme de la Guilde des Voleurs]";
		mes "Bon... faites ce que vous voulez~ Allez votre chemin~";
		close;
	}
	mes "[Homme de la Guilde des Voleurs]";
	mes "D'accord, dites-moi votre nom. Euh ? Quoi ?";
	mes ""+strcharinfo(0)+". So "+strcharinfo(0)+", right...?";
	next;
	if (!callfunc("F_CanChangeJob")) {
		mes "[Homme de la Guilde des Voleurs]";
		mes "Je vois votre forte volonté pour devenir un Voleur......";
		mes "Mais avec seulement votre volonté, vous ne pourrez pas tenir dans un vrai combat, non ?";
		mes "Allez donc atteindre au moins le Niveau de Compétence de Base 9.";
		close;
	}
	mes "[Homme de la Guilde des Voleurs]";
	mes "Hmm.... J'ai trouvé vos dossiers criminels intéressants dans les données de l'agence détective. Eh bien, vous semblez avoir ce qu'il faut pour être un bon voleur.";
	mes ""+strcharinfo(0)+". I now declare that you have passed the job interview!";
	next;
	mes "[Homme de la Guilde des Voleurs]";
	mes "Eh bien, je ne suis pas chargé de vous faire devenir Voleur. Je n'accepte simplement les candidatures, c'est compris ?";
	mes "Si vous voulez devenir Voleur, demandez à l'homme à côté de moi.";
	set q_job_thief,1;
	close;
}
