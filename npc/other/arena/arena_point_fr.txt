prt_are_in,103,11,3	script	Arena Point Manager	732,{
	mes "[Gestionnaire des Points d'Arène]";
	mes "J'espère que vous passez un bon";
	mes "moment dans l'Arena. Si vous avez";
	mes "gagné des Points de Piste Turbo à la";
	mes "piste Turbo d'Al De Baran, je peux les";
	mes "convertir en Points d'Arena.";
	next;
	switch(select("Point Check:Convert Points:^660000Conversion Info^000000")) {
	case 1:
		mes "[Gestionnaire des Points d'Arène]";
		mes "" + strcharinfo(0) + ",";
		mes "vous avez actuellement";
		mes "" + arena_point + " Arena Points";
		mes "and " + tt_point + " Turbo Track Points.";
		close;
	case 2:
		mes "[Gestionnaire des Points d'Arène]";
		mes "" + strcharinfo(0) + ",";
		mes "vous avez actuellement";
		mes "" + arena_point + " Arena Points";
		mes "and " + tt_point + " Turbo Track Points.";
		next;
		mes "[Gestionnaire des Points d'Arène]";
		mes "Veuillez choisir parmi les";
		mes "conversions de Points de Piste en Points d'Arena.";
		mes "Notez que lorsque vous convertissez plus de 10";
		mes "Points de Piste à la fois, vous ne pouvez qu'effectuer des conversions en ^4D4DFFmultiples de 10^000000.";
		mes "[Gérant des Points d'Arena]";
		next;
		switch(select("2 TP -> 1 AP:4 TP -> 2 AP:6 TP -> 3 AP:8 TP -> 4 AP:10 TP and more:Cancel")) {
		case 1:
			callsub S_ExchangePoints,28999,2,1;
		case 2:
			callsub S_ExchangePoints,28998,4,2;
		case 3:
			callsub S_ExchangePoints,28997,6,2;
		case 4:
			callsub S_ExchangePoints,28996,8,4;
		case 5:
			mes "[Gestionnaire des Points d'Arène]";
			mes "de fois que vous souhaitez convertir";
			mes "10 Points Turbo de Piste en";
			mes "10 Turbo Track Points into";
			mes "Points d'arène. La plus";
			mes "grande valeur que vous pouvez entrer est de 20.";
			mes "Pour annuler, entrez ''^3355FF0^000000.''.";
			next;
			input .@input;
			if (.@input <= 0) {
				mes "[Gestionnaire des Points d'Arène]";
				mes "Vous avez";
				mes "annulé";
				mes "ce service.";
				close;
			}
			else if (.@input > 20) {
				mes "[Gestionnaire des Points d'Arène]";
				mes "Votre demande dépasse";
				mes "la limite maximale. Veuillez";
				mes "entrer une valeur n'excédant pas 20.";
				close;
			}
			else {
				set .@want_point1,10 * .@input;
				set .@want_point,5 * .@input;
				set .@my_arena_all,arena_point + .@want_point;
				set .@my_turbo_all,tt_point - .@want_point1;
				if (.@my_arena_all > 28999) {
					mes "[Gestionnaire des Points d'Arène]";
					mes "Vous allez dépasser la";
					mes "limite maximale de";
					mes "Arena Points si nous continuons cette conversion de vos";
					mes "Turbo Track Points. Vous ne pouvez pas";
					mes "Turbo Track Points. Vous ne pouvez pas";
					mes "avez plus de 29 000 points d'arène.";
					next;
					mes "[Gestionnaire des Points d'Arène]";
					mes "Veuillez dépenser certains de vos";
					mes "points d'arène avant";
					mes "d'utiliser à nouveau ce service.";
					mes "Merci pour votre mécénat.";
					close;
				}
				if (.@my_turbo_all < 0) {
					mes "[Gestionnaire des Points d'Arène]";
					mes "Je suis désolé, mais";
					mes "vous n'avez pas assez de";
					mes "Turbo Track Points pour";
					mes "effectuer cette conversion d'Arena";
					mes "Points.";
					close;
				}
				else {
					mes "[Gestionnaire des Points d'Arène]";
					mes "Vous avez converti";
					mes "10 Turbo Track Points";
					mes "into 5 Arena Points " + .@input + " times.";
					mes "A total of " + .@want_point1 + " Turbo Track Points were converted into";
					mes "" + .@want_point +" Arena Points.";
					set tt_point, .@my_turbo_all;
					set arena_point, .@my_arena_all;
					next;
					mes "[Gestionnaire des Points d'Arène]";
					mes "" + strcharinfo(0) + ",";
					mes "vous avez maintenant";
					mes "^4682B4" + arena_point + "^000000 Arena Points";
					mes "and ^00688B" + tt_point + "^000000 Turbo Track Points.";
					mes "Merci pour votre mécénat.";
					close;
				}
			}
		case 6:
			mes "[Gestionnaire des Points d'Arène]";
			mes "Vous avez";
			mes "annulé";
			mes "ce service.";
			close;
		}
	case 3:
		// Again. no dialog for this option.
		close;
	}
	close;

S_ExchangePoints:
	if (arena_point > getarg(0) ) {
		mes "[Gestionnaire des Points d'Arène]";
		mes "Vous allez dépasser la";
		mes "limite maximale de";
		mes "Arena Points si nous continuons cette conversion de vos";
		mes "Turbo Track Points. Vous ne pouvez pas";
		mes "Turbo Track Points. Vous ne pouvez pas";
		mes "avez plus de 29 000 points d'arène.";
		next;
		mes "[Gestionnaire des Points d'Arène]";
		mes "Veuillez dépenser certains de vos";
		mes "points d'arène avant";
		mes "d'utiliser à nouveau ce service.";
		mes "Merci pour votre mécénat.";
		close;
	}
	.@track_point = getarg(1);
	if (tt_point >= .@track_point) {
		.@arena_point = getarg(2);
		mes "[Gestionnaire des Points d'Arène]";
		mes "" + strcharinfo(0) + ",";
		mes "you've converted "+ .@track_point +" Track";
		mes "Points into "+ .@arena_point +" Arena Point.";
		tt_point = tt_point - .@track_point;
		arena_point = arena_point + .@arena_point;
		next;
		mes "[Gestionnaire des Points d'Arène]";
		mes "" + strcharinfo(0) + ",";
		mes "vous avez maintenant";
		mes "^4682B4" + arena_point + "^000000 Arena Points";
		mes "and ^00688B" + tt_point + "^000000 Turbo Track Points.";
		mes "Merci pour votre mécénat.";
		close;
	}
	else {
		mes "[Gestionnaire des Points d'Arène]";
		mes "Désolé, mais vous n'avez pas";
		mes "assez de points Turbo Track.";
		mes "Vous devez au moins avoir 2 points Turbo Track pour ce service de conversion.";
		close;
	}
}
