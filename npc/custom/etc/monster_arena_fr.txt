	if (strcharinfo(0) != $@duelist1$ && strcharinfo(0) != $@duelist2$) end;
	announce strcharinfo(0) + " won the Duel",1;
	set @fighting,0;
	set #heal,0;
	set @healing,0;
	end;
}

	announce "Illegal Kill by " + strcharinfo(0) + " Detected",1;
	percentheal -100,-100;
	end;
}
