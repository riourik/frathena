-	script	Goods Cabinet#00::pggc	-1,{
	mes "C'est un Coffre-fort de marchandises.";
	mes "Un message est écrit sur une feuille de papier.";
	next;
	mes "+ Le coffre-fort est exclusivement +";
	mes "+ pour le groupe Eden +";
	mes "Si vous souhaitez utiliser ce";
	mes "coffre-fort, veuillez vérifier";
	mes "les points suivants :";
	next;
	mes "1. Êtes-vous membre";
	mes "du groupe Eden ?";
	mes "2. Avez-vous appris";
	mes "assez de compétences de base ?";
	mes "3. La taxe du coffre-fort est";
	mes "^4d4dff500 zeny^000000 !";
	next;
	if (callfunc("F_HasEdenGroupMark")) {
		mes "Vous devez insérer des zeny pour utiliser le coffre-fort.";
		mes "Coût : 500 Zeny";
		mes "Souhaitez-vous l'utiliser ?";
		next;
		switch(select("Use the Cabinet.:Cancel.")) {
			case 1:
				if (Zeny > 499) {
					if(!callfunc("F_CanOpenStorage")){
						mes "Le coffre-fort ne fonctionne pas pour moi.";
						mes "Peut-être que je ne suis pas encore qualifié pour utiliser le Coffre-fort.";
						close;
					}
					set Zeny, Zeny - 500;
					close2;
					openstorage;
					end;
				}
				mes "Je n'ai pas assez de zeny.";
				mes "J'ai besoin d'au moins 500 zeny pour utiliser le Coffre-fort.";
				close;
			case 2:
				mes "[Nom du PNJ] Je l'utiliserai la prochaine fois.";
				close;
		}
	}
	mes "[Nom du PNJ] J'ai besoin d'un Marque de Groupe d'Eden pour utiliser ce Cabinet.";
	close;
}
moc_para01,173,120,0	duplicate(pggc)	Goods Cabinet#01	111
