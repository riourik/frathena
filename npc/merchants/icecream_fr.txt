-	script	Ice Cream Maker::IceCreamer	-1,{
	mes "[Machine à glace]";
	mes "Crème glacée fraîche faite avec de la neige de Lutie !";
	mes "Profitez-en maintenant, elle ne sera pas en vente longtemps !";
	mes "^3355FFZeny^000000 Crème glacée,";
	mes "Crème glacée !";
	next;
	if(select("Gimme Ice Cream!:Cancel Trade") == 2) {
		mes "Êtes-vous sûr de ne rien vouloir ?";
		mes "Je ne la vendrai pas longtemps,";
		mes "et une fois que je n'en aurai plus,";
		mes "il n'y en aura plus du tout !!!";
		close;
	}
	mes "[Machine à glace]";
	mes "Crème glacée fraîche faite avec de la neige de Lutie !";
	mes "Tout le monde veut notre délicieuse crème glacée,";
	mes "mais nous avons une quantité limitée,";
	mes "donc vous ne pouvez acheter que 5 à la fois !!";
	next;
	while(true) {
		input .@input;
		if(.@input < 1) {
			mes "[Machine à glace]";
			mes "Si vous n'avez pas envie d'en acheter,";
			mes "pourriez-vous permettre au client suivant";
			mes "d'effectuer un achat ?";
			mes "Merci.";
			close;
		}
		if(.@input > 5) {
			mes "[Machine à glace]";
			mes "Ouch";
			mes "Vous attendez trop.";
			mes "Cher(e) client(e),";
			mes "Si vous mangez plus de 5 crèmes glacées,";
			mes "Il se peut que vous deviez faire beaucoup de voyages";
			mes "au toilette ce soir.";
			next;
			continue;
		}
		break;
	}
	if(Zeny<.@input*100) {
		mes "[Machine à glace]";
		mes "Cher client, votre porte-monnaie semble être léger.";
		mes "Le prix est de ^3355FF100 Zeny^000000 par glace.";
		close;
	}
	if(checkweight(536,.@input) == 0) {
		mes "[Machine à glace]";
		mes "Cher client, vous avez l'air d'avoir beaucoup à porter.";
		mes "La Glace va bien,";
		mes "mais vous devez prendre en compte votre poids";
		mes "avant de faire un achat.";
		close;
	}
	set Zeny,Zeny-100*.@input;
	getitem 536,.@input;	// Ice_Cream
	close;
}

// Alberta
alberta,120,45,2	duplicate(IceCreamer)	Ice Cream Maker#1	85
