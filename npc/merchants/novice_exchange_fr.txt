function	script	F_PotExchange	{
	.@item_req = getarg(0);
	.@req_amount = getarg(1);

	if (countitem(.@item_req) < .@req_amount) {
		mes "^FF0000Hum~";
		mes "Hé, où sont ";
		mes "all those "+ getitemname(.@item_req) +"";
		mes "^FF0000ceux que tu m'as promis^ ?";
		mes "Give me "+ getitemname(.@item_req) +"!";
		close;
	}
	else {
		mes "^FF0000Hum~";
		mes "D'accord, laisse-moi vérifier";
		mes "how many "+ getitemname(.@item_req) +" you";
		mes "^FF0000Chez vous. Hmm...^000000";
		next;
		mes "^FF0000Hum~";
		mes "Vous avez";
		mes "a total of "+ countitem(.@item_req) +" "+ getitemname(.@item_req) +".";
		mes "Je peux vous donner un total";
		mes "of "+ (countitem(.@item_req)/.@req_amount) +" Red Potions for those.";
		next;
		mes "^FF0000Hum~";
		mes "Que dites-vous?";
		mes "Sommes-nous d'accord?";
		next;
		if (select("Deal.:No deal.") == 1) {
			mes "^FF0000Hum~";
			mes "Vous savez le nombre exact";
			mes "de Potions Rouges";
			mes "que vous voulez recevoir pour";
			mes "those "+ getitemname(.@item_req) +", don't you?";
			next;
			mes "^FF0000Hum~";
			mes "Maintenant, je peux échanger avec vous";
			mes "un minimum de 1 Potion Rouge";
			mes "et un maximum de 100 Potions Rouges";
			mes "à la fois. Si vous changez d'avis, entrez simplement '0' pour annuler.";
			mes "[Marchand]";
			next;
			while(1) {
				input .@input,0,101;
				if (.@input == 0) {
					mes "^FF0000Hum~";
					mes "Quoi...?";
					mes "m'adressez-vous du tout?";
					mes "even bother to talk to me?";
					mes "^FF0000Tu hésites beaucoup, non ?";
					close;
				}
				else if (.@input > 100) {
					mes "^FF0000Hum~";
					mes "Je ne peux pas te donner plus de 100 potions rouges";
					mes "d'un coup. Essayons";
					mes "ça encore.";
					mes "[Marchand]";
					next;
				}
				else {
					break;
				}
			}
			.@put_out_item = .@input * .@req_amount;
			if (countitem(.@item_req) < .@put_out_item) {
				mes "^FF0000Hum~";
				mes "[Marchand]";
				close;
			}
			mes "^FF0000Hum~";
			mes "Et voilà !";
			mes "je t'ai données, ça devrait être bon. Merci, c'était un bon marché~";
			mes "[Marchand]";
			delitem .@item_req,.@put_out_item; //Shell
			getitem 501,.@input; //Red_Potion
			close;
		}
		mes "^FF0000Hum~";
		mes "[Marchand]";
		mes "Bien.";
		mes "^FF0000N'ont pas les potions rouges plus d'utilité pour toi, aventurier?";
		mes "[Marchand]";
		mes "to an adventurer like you?";
		close;
	}
}
