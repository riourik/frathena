		next;
		mes "[Forgeron]";
		mes "For the sake of our craft, our beautiful, yet masculine faces are put at risk...";
		next;
		mes "[Forgeron]";
		mes "Mais grâce à cette merveille de technologie, il n'y a pas lieu de s'inquiéter ! Voici le ^3355FFMasque de Soudure^000000 !!";
		next;
		mes "[Forgeron]";
		mes "Il fonctionne en couvrant votre visage avec une plaque métallique épaisse. En effet, c'est un objet que tout le monde devrait vouloir ! Si ce n'est pas pour vous-même, il serait une bonne idée de donner cela à un ami qui étudie la forge.";
		next;
		mes "[Forgeron]";
		mes "Alors, qu'en dites-vous ? Si vous me donnez ^2244FF50 Acier^000000 et ^4422FFseulement 2000 Zeny^000000, je vais le faire pour vous tout de suite~!!";
		next;
		switch(select("Hm... Not bad. Alright.:How does it work again?")) {
		case 1:
			if ((countitem(999) > 49) && (Zeny > 1999)){
				mes "[Forgeron]";
				mes "Ah, je vois que vous appréciez le travail d'un Forgeron ! D'accord, je vais le faire tout de suite !!";

				delitem 999,50;  //Steel

				mes "Hahaha~! Utilisez-le bien !";

				mes "[Forgeron]";
				mes "Hmmm...";

				next;

			mes "La ^3355FFMasque Soudure^000000 protège ton visage en le recouvrant d'une plaque de métal épaisse. Cette invention est le résultat de 30 années de labeur incessant au marteau et à la tenaille.";
			close;
		}
	case 2:
		mes "[Forgeron]";
		mes "Ah...!";
		mes "Je suppose que c'est la première fois que tu rencontres un Forgeron aussi grand qu'un tel que moi, et tu es juste sans voix d'émerveillement. Prends ton temps et parle quand tu seras prêt.";
		next;
		mes "^3355FFCe type semble";
		mes "un peu gonflé";
		mes "de lui-même.^000000";
		close;
	}
}

// Number Match Game :: gef_yagu
//============================================================
geffen,122,77,4	script	Nia#yagu	700,{
	mes "[Nia]";
	mes "Bonjour, je suis Nia, la cinquième";
	mes "disciple de Mawong, le";
	mes "génie du jeu. Maintenant,";
	mes "comment aimeriez-vous jouer à";
	mes "une partie de Match Numéral avec moi?";
	next;
	switch(select("Number Match Game?:Practice Number Match:Play Number Match:Record Holder:Cancel")) {
	case 1:
		mes "[Nia]";
		mes "Ah bien, l'objectif du";
		mes "jeu Match Numéral est de";
		mes "deviner le numéro que je pense.";
		mes "Bien sûr, il y a quelques";
		mes "des règles qui limitent les chiffres";
		mes "dont je peux choisir.";
		next;
		mes "[Nia]";
		mes "Tous les chiffres que je fais";
		mes "doivent être de trois chiffres, je ne peux pas";
		mes "utiliser 0 pour aucun des chiffres,";
		mes "et je ne peux pas utiliser le même";
		mes "chiffre pour aucun des chiffres";
		mes "plus d'une fois.";
		next;
		mes "[Nia]";
		mes "Donc, par exemple, je ne peux pas";
		mes "utiliser les numéros 103, 112,";
		mes "252 ou 701. Ces chiffres";
		mes "ont soit un 0, soit ils utilisent";
		mes "le même chiffre dans les chiffres";
		mes "plus d'une fois. C'est clair maintenant?";
		next;
		mes "[Nia]";
		mes "Maintenant, vous n'avez que 5 essais,";
		mes "et après chaque essai, je vais te dire";
		mes "combien de chiffres tu as devinés";
		mes "^FF0000correctement, suivi du nombre de chiffres corrects devinés dans";
		mes "la bonne position.";
		next;
		mes "[Nia]";
		mes "Par exemple, disons que le";
		mes "nombre auquel je pense est 168.";
		mes "Si votre première supposition est 678, alors vous avez deviné deux chiffres correctement, les 6 et 8. Cependant,";
		mes "seul le 8 est à la bonne place.";
		mes "[Nia]";
		next;
		mes "[Nia]";
		mes "d'essayer d’abord le Mode Pratique.";
		mes "Ensuite, lorsque vous vous sentez plus";
		mes "à l'aise, jouez au vrai jeu de Correspondance Numérique. N'oubliez pas que la vitesse compte pour votre score!";
		mes "[Nia]";
		mes "Ah, si par hasard vous créez un nouveau record dans";
		next;
		mes "[Nia]";
		mes "donc ne manquez pas cette chance pour";
		mes "make a new record in the";
		mes "Number Match game, you'll";
		mes "have your name recorded,";
		mes "so don't miss this chance for";
		mes "fame et peut-être même la fortune~";
		close;
	case 2:
		mes "[Nia]";
		mes "Génial, faisons une répétition";
		mes "du jeu Coup de numéro. S'il vous plaît";
		mes "essayez de deviner le nombre que j'ai en tête, d'accord?";
		mes "Si vous voulez annuler, entrez un nombre supérieur à 1 000.";
		set .@practice,1;
		break;
	case 3:
		mes "[Nia]";
		mes "Génial, jouons au Coup de numéro !";
		mes "Vous aurez cinq chances pour deviner le nombre que j'ai en tête.";
		mes "Si vous voulez annuler, entrez un nombre supérieur à 1 000.";
		set .@practice,0;
		break;
	case 4:
		set .@name_record$, $050908_minus1_yagu$;
		if (.@name_record$ == "") //custom
			set .@name_record$,"(null)";
		set .@score_record, $050908_minus1_yagu;
		set .@score_min, .@score_record / 60;
		set .@score_sec, .@score_record % 60;
		mes "[Nia]";
		mes "^ff0000" + .@name_record$ + "^000000 is the";
		mes "détenteur du record dans le jeu";
		mes "Coup de numéro avec une durée de";
		mes "^ff0000" + .@score_min + " minutes, " + .@score_sec + " seconds^000000.";
		close;
	case 5:
		mes "[Nia]";
		mes "Prenez soin de vous, aventurier.";
		mes "Si un jour";
		mes "vous avez envie de jouer au Coup";
		mes "de numéro, venez me parler, d'accord? Au revoir~";
		mes "[Nia]";
		close;
	}
	next;

	// Generate 3 digits of a number.
	while(1) {
		set .@yagu100, rand(1,9);
		set .@yagu10, rand(1,9);
		set .@yagu1, rand(1,9);
		if ((.@yagu100 != .@yagu10) && (.@yagu100 != .@yagu1) && (.@yagu10 != .@yagu1)) {
			// Store the actual number (unofficial).
			set .@yagu, (100*.@yagu100) + (10*.@yagu10) + .@yagu1;
			break;
		}
	}

	// The official time calculation is based on ServerTime,
	// an int HHMMSS equivalent to atoi(gettimestr("%H%M%S",7)).
	// UNIX time is really a better way to do this.
	set .@start_time, gettimetick(2);

	while(1) {

		// Guess a number that meets all conditions.
		while(1) {
			if (input(.@input,100,999) != 0) {
				mes "[Nia]";
				mes "Vous avez annulé le";
				mes "jeu de Correspondance Numérique.";
				close;
			}
			set .@input100, .@input / 100;
			set .@input10, (.@input % 100) / 10;
			set .@input1, .@input % 10;
			if ((.@input100 > 0) && (.@input10 > 0) && (.@input1 > 0)) {
				if ((.@input100 != .@input10) && (.@input100 != .@input1) && (.@input10 != .@input1))
					break;
			}
			mes "[Nia]";
			mes "vous êtes entré l'un des";
			mes "chiffres comme 0, ou avez saisi la";
			mes "même valeur numérique plus d'une fois dans les";
			mes "chiffres.";
			mes "Veuillez saisir une autre devinette.";
			next;
		}

		set .@retry, .@retry+1;
		mes "[Nia]";
		mes "^ff0000"+strcharinfo(0)+"^000000,";
		mes "your guess is ^0000ff"+.@input+"^000000.";
		mes "Donnez-moi un moment pour";
		mes "vous annoncer vos résultats.";
		next;

		// Correct digits in correct locations.
		set .@strike,0;
		if (.@yagu100 == .@input100)
			set .@strike, .@strike+1;
		if (.@yagu10 == .@input10)
			set .@strike, .@strike+1;
		if (.@yagu1 == .@input1)
			set .@strike, .@strike+1;

		// Correct digits in incorrect locations.
		set .@ball,0;
		if ((.@yagu100 == .@input10) || (.@yagu100 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu10 == .@input100) || (.@yagu10 == .@input1))
			set .@ball, .@ball+1;
		if ((.@yagu1 == .@input100) || (.@yagu1 == .@input10))
			set .@ball, .@ball+1;

		if (.@strike == 3) {
			if (.@practice) {
				mes "[Nia]";
				mes "Oui, le nombre que";
				mes "I guessed was ^ff0000"+.@yagu+"^000000!";
				mes "Félicitations, vous venez de gagner cette partie d'entraînement ! Maintenant, vous devriez être prêts pour la vraie chose~";
				mes "Maintenant, vous devriez être";
				mes "prêt pour le vrai jeu !";
				mes "~";
				close;
			} else {

				// Calculate final time and check against the previous record.
				set .@end_time, gettimetick(2);
				if (.@end_time < .@start_time)
					set .@end_time, .@end_time + 3600;
				set .@total_time, .@end_time - .@start_time;
				set .@total_min, .@total_time / 60;
				set .@total_sec, .@total_time % 60;
				set .@name_record$, $050908_minus1_yagu$;
				if (.@name_record$ == "") //custom
					set .@name_record$,"(null)";
				set .@score_record, $050908_minus1_yagu;
				set .@score_min, .@score_record / 60;
				set .@score_sec, .@score_record % 60;

				if (.@score_record >= .@total_time || !.@score_record) { //custom (second condition)
					set $050908_minus1_yagu$, strcharinfo(0);
					set $050908_minus1_yagu, .@total_time;
					mes "[Nia]";
					mes "Génial ! Vous avez deviné mon";
					mes "nombre qui était";
					mes "^ff0000" + .@yagu + "^000000. You finished in";
					mes "^ff0000" + .@total_min + " minutes, " + .@total_sec + " seconds^000000,";
					mes "en brisant le précédent record";
					mes "of ^ff0000" + .@score_min + " minutes, " + .@score_sec + " seconds^000000.";
					next;
					mes "[Nia]";
					mes "^ff0000" + .@name_record$ + "^000000";
					mes "établi ce vieux record,";
					mes "mais maintenant vous êtes la";
					mes "nouvelle personne à battre dans le";
					mes "Match Game. Félicitations";
					mes "pour avoir établi un nouveau record !";
					close;
				}
				mes "[Nia]";
				mes "Bonne performance, vous avez deviné";
				mes "mon numéro ! La réponse, pour";
				mes "course, was ^ff0000" + .@yagu + "^000000. You managed";
				mes "deviner ce nombre correctement";
				mes "in ^ff0000" + .@total_min + " minutes, " + .@total_sec + " seconds^000000.";
				next;
				mes "[Nia]";
				mes "^ff0000" + .@name_record$ + "^000000 is";
				mes "le détenteur actuel du record de Match Numéro";
				mes "avec un temps";
				mes "of ^ff0000" + .@score_min + " minutes, " + .@score_sec + " seconds^000000.";
				mes "Bonne chance, et espérons que vous";
				mes "pourrez briser ce record un jour~";
				close;
			}
		} else {
			if (.@practice) {
				mes "[Nia]";
				mes "Your guess has ^ff0000" + .@ball + "^000000 of the";
				mes "correct numbers. ^ff0000" + .@strike + "^000000 of the";
				mes "chiffres de votre devinette ont";
				mes "le bon numéro dans la position";
				mes "correcte du chiffre.";
				mes "D'accord, essaie encore~";
				next;
			} else {
				mes "[Nia]";
				mes "Your last guess had ^ff0000" + .@ball + "^000000";
				mes "des chiffres corrects,";
				mes "and ^ff0000" + .@strike + "^000000 digits in your guess";
				mes "avait le bon chiffre dans";
				mes "la bonne position.";
				switch(.@retry) {
					case 1: mes "Vous avez quatre essais restants."; break;
					case 2: mes "Vous avez trois essais restants."; break;
					case 3: mes "Vous avez deux essais restants."; break;
					case 4: mes "Vous n'avez qu'un seul essai restant..."; break;
					case 5: mes "The answer was ^ff0000" + .@yagu + "^000000."; close;
				}
				next;
			}
		}
	}
}
