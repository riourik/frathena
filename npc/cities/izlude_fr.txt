-	script	::Sailor_izlude	-1,{
	mes "[Marin]";
	mes "Salut tout le monde !";
	mes "Attention, attention !";
	mes "Viens chevaucher le vent";
	mes "sur un navire fascinant !";
	mes "Dépêchez-vous, dépêchez-vous !";
	next;
	switch(select("Byalan Island -> 150 Zeny.:Alberta Marina  -> 500 Zeny.:Cancel.")) {
	case 1:
		if (Zeny < 150) {
			mes "[Marin]";
			mes "150 Zéniy !";
			mes "Seulement 150 Zeny à rouler !";
			close;
		}
		set Zeny, Zeny - 150;
		warp "izlu2dun",107,50;
		end;
	case 2:
		if (Zeny < 500) {
			mes "[Marin]";
			mes "500 Zéniy !";
			mes "Seulement 500 Zeny à rouler !";
			close;
		}
		set Zeny, Zeny - 500;
		warp "alberta",188,169;
		end;
	case 3:
		close;
	}
}
