prontera,132,125,4	script	Bossnia Staff#1	908,{
	mes "[Riss]";
	mes "Bonjour?";
	mes "J'ai trouvé des lieux magnifiques.";
	mes "pendant que je voyageais un peu partout dans le monde.";
	mes "Je suis aventurière.";
	mes "Haha~";
	next;
	mes "[Riss]";

	mes "Parfois, il faut être généreux.";
	mes "Il n'y a pas de temps à perdre avec ce genre de détails sans importance";
	mes "lorsqu'il faut se concentrer sur des choses plus importantes.";
	next;
	mes "[Riss]";
	mes "Hum... enfin, je voulais dire... quelque chose..";
	mes "Alors que je voyageais à travers certains endroits,";
	mes "j'ai trouvé un lieu vraiment terrifiant.";
	next;
	mes "[Riss]";
	mes "La plupart du temps lorsque vous venez dans un lieu,";
	mes "il y a une créature forte et terrifiante.";
	mes "N'est-ce pas ?";
	next;
	mes "[Riss]";
	mes "Mais... mais... là-dedans...";
	mes "Il y a beaucoup de monstres effrayants et puissants là-dedans...";
	mes "C'était vraiment terrifiant.";
	next;
	mes "[Riss]";
	mes "Si j'avais réagi un peu plus tard... quelques secondes...";
	mes "...je serais peut-être mort.";
	next;
	mes "[Riss]";
	mes "Quoi?";
	mes "Tu veux aller là-dedans?";
	mes "Oh~ Garçon~ tu ne m'as pas eu.";
	mes "Là-dedans....";
	next;
	mes "[Riss]";
	mes "Ah... tu sais déjà?";
	mes "Bien que tu connaisses l'endroit, tu veux y aller...";
	mes "D'accord, je vais te l'envoyer.";
	mes "Mais après t'y être rendu, ne garde pas";

	// If the VIP system is enabled, players pay a Reset Stone for 5 entrances (with a choice of warps).
	// If it's disabled, players pay 5,000z per entrance.
	if (VIP_SCRIPT) {
		mes "une rancœur envers moi. Cela te coûtera 1 Reset Stone pour 5 accès.";
		next;
		mes "[Riss]";
		if (bossnia_event > 0) {
			set .@type,1;
			mes "Remaining access: "+bossnia_event;
		} else if (countitem(6320)) {
			set .@type,2;
			mes "Veux-tu y aller?";
		} else {
			mes "^FF0000Vous n'avez pas de billet maintenant....";
			mes "^FF0000Alors revenez-moi plus tard avec une Pierre de Réinitialisation.";
			close;
		}
		next;
		if(select("Enter:Leave") == 2)
			close;
		set .@i, select("First:Second:Third:Fourth");
		switch(select("Warp 1:Warp 2:Warp 3:Warp 4")) {
			case 1: set .@x,31; set .@y,208; break;
			case 2: set .@x,31; set .@y,31; break;
			case 3: set .@x,208; set .@y,31; break;
			case 4: set .@x,208; set .@y,208; break;
		}
		if (.@type == 1)
			set bossnia_event, bossnia_event-1;
		else {
			delitem 6320,1; //Premium_Reset_Stone
			set bossnia_event,4;
		}
		specialeffect2 EF_MAXPOWER;
		warp "bossnia_0"+.@i,.@x,.@y;
		close;
	} else {
		mes "^FF0000une rancœur contre moi. Cela vous coûtera aussi 5 000 zeny.";
		next;
		if (Zeny > 4999) {
			mes "[Riss]";
			mes "^FF0000Voulez-vous vraiment accepter le défi?";
			mes "^FF0000D'accord, choisissez simplement l'itinéraire.";
			next;
			set .@i, select("First:Second:Third:Fourth");
			mes "[Riss]";
			mes "^FF0000Prenez soin de vous, garçon~";
			mes "^FF0000N'aitez pas une rancœur contre moi.";
			close2;
			set Zeny, Zeny - 5000;
			warp "bossnia_0"+.@i,rand(202,204),rand(202,204);
			end;
		} else {
			mes "[Riss]";
			mes "^FF0000Vous n'avez pas assez d'argent...";
			mes "^FF0000Revenez quand vous aurez au moins 5 000 zeny.";
			close;
		}
	}
}
