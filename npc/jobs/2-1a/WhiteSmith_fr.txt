valkyrie,44,50,5	script	MasterSmith#Valkyrie	731,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[MasterSmith]";
			mes "Félicitations.";
			mes "Honneur aux guerriers !";
			close;
		}
		mes "[MasterSmith]";
		mes "Ah...";
		mes "Le sommet";
		mes "du savoir-faire.";
		mes "C'est le travail d'un";
		mes "MaîtreForgeron.";
		next;
		mes "[MasterSmith]";
		mes "Une fois que vous êtes";
		mes "les meilleurs des meilleurs,";
		mes "vous commencez à forger avec";
		mes "le but de découvrir";
		mes "de nouvelles et meilleures façons de";
		mes "façonner...";
		close;
	}
	else if (ADVJOB == Job_WhiteSmith && Class == Job_Merchant_High && JobLevel > 39) {
		mes "[MasterSmith]";
		mes "Le moment est venu!";
		mes "Notre monde a besoin de braves,";
		mes "de travailleurs acharnés";
		mes "comme vous...";
		next;
		mes "[MasterSmith]";
		mes "Souhaitez-vous";
		mes "devenir un MaîtreForgeron?";
		next;
		if (select("No.:Yes.") == 1) {
			mes "[MasterSmith]";
			mes "[Grand Prêtre]";
			mes "n'hésitez pas à revenir.";
			mes "Honneur aux guerriers !";
			close;
		}
		if (SkillPoint) {
			mes "[MasterSmith]";
			mes "Il est encore possible pour vous d'apprendre des compétences supplémentaires. Veuillez utiliser";
			mes "tous vos points de compétence restants avant de revenir me voir.";
			close;
		}
		jobchange Job_Whitesmith;
		set ADVJOB,0;
		mes "[MasterSmith]";
		mes "Félicitations!";
		mes "En tant que Maître Forgeron,";
		mes "j'espère que vous forgeerez";
		mes "un chemin vers un avenir";
		mes "plus radieux pour Midgard.";
		close;
	}
	mes "[MasterSmith]";
	mes "Accueillir";
	mes "vers Valhalla,";
	mes "la Salle de l'Honneur.";
	next;
	mes "[MasterSmith]";
	mes "S'il vous plaît faites-vous";
	mes "confortable";
	mes "pendant votre séjour ici.";
	mes "Honneur aux guerriers !";
	close;
}
