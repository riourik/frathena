valkyrie,53,54,3	script	Minstrel#Valkyrie	741,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Minstrel]";
			mes "Félicitations.";
			mes "Honneur aux guerriers !";
			close;
		}
		mes "[Minstrel]";
		mes "Veux-tu";
		mes "chanter une chanson avec moi?";
		mes "Sha la la la la~";
		close;
	}
	else if (ADVJOB == Job_Clown && Class == Job_Archer_High && JobLevel > 39) {
		mes "[Minstrel]";
		mes "le monde morose";
		mes "des mortels a besoin";
		mes "d'une musique plus gaie.";
		mes "Vas-tu l'apporter à eux";
		mes "et tourner la vapeur dans la";
		mes "bataille contre le mal?";
		next;
		mes "[Minstrel]";
		mes "Feras-tu cela";
		mes "pour Midgard...";
		mes "En tant que Ménestrel?";
		next;
		if (select("No.:Yes.") == 1) {
			mes "[Minstrel]";
			mes "[Grand Prêtre]";
			mes "n'hésitez pas à revenir.";
			mes "Honneur aux guerriers !";
			close;
		}
		if (SkillPoint) {
			mes "[Minstrel]";
			mes "Il est encore possible pour vous d'apprendre des compétences supplémentaires. Veuillez utiliser";
			mes "tous vos points de compétence restants avant de revenir me voir.";
			close;
		}
		jobchange Job_Clown;
		set ADVJOB,0;
		mes "[Minstrel]";
		mes "Félicitations!";
		mes "En tant que Ménestrel, ton";
		mes "^FF0000vos chansons apporteront";
		mes "^FF0000espoir à vos alliés, et";
		mes "^FF0000désespoir à vos ennemis.";
		close;
	}
	mes "[Minstrel]";
	mes "Accueillir";
	mes "vers Valhalla,";
	mes "la Salle de l'Honneur.";
	next;
	mes "[Minstrel]";
	mes "S'il vous plaît faites-vous";
	mes "confortable";
	mes "pendant votre séjour ici.";
	mes "Honneur aux guerriers !";
	close;
}
