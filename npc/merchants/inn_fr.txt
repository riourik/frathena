lhz_in02,230,284,4	script	Hotel Employee#01	86,{
	mes "[Employé de l'hôtel]";
	mes "Bienvenue à";
	mes "le Dragon Royal,";
	mes "oç vous pouvez trouver le";
	mes "meilleurs établissements d'accueil";
	mes "et le meilleur service.";

	switch(select("Save Point:Rest - 5,000 zeny:Cancel")) {
	case 1:
		savepoint "lhz_in02",209,275,1,1;
		mes "[Employé de l'hôtel]";
		mes "Merci, votre";
		mes "point de réapparition a été enregistré ici dans";
		mes "le Dragon Royal.";
		mes "vous avez besoin de 5 000 zeny";
		close;
	case 2:
		if(Zeny < 5000) {
			mes "[Employé de l'hôtel]";
			mes "Je suis désolé, mais";
			mes "pour vous accorder un séjour.";
			mes "durant votre séjour au";
			close;
		}
		set Zeny,Zeny - 5000;
		mes "[Employé de l'hôtel]";
		mes "Merci ~";
		mes "J'espère que tu apprécieras";
		mes "Dragon Royal.";
		mes "[Gestionnaire d'Hôtel]";
		close2;
		warp "lhz_in02",219,150;
		percentheal 100,100;
		npcskill "AL_BLESSING",10,99,99;
		end;
	case 3:
		mes "[Employé de l'hôtel]";
		mes "Merci et";
		mes "passe une bonne journée.";
		close;
	}
}

//======================= Inn Function ==============================
function	script	F_InnMaid	{
	.@npc_name$ = getarg(0);

	mes .@npc_name$;
	mes "Bienvenue à";
	mes getarg(1) + ".";
	mes "Comment puis-je vous aider ?";
	next;
	switch(select("Save:Take a Rest -> 5000 zeny:Cancel")) {
	case 1:
		mes .@npc_name$;
		mes "Votre point de résurrection";
		mes "a été sauvegardé.";
		mes "Merci,";
		mes "s'il te plaît, reviens.";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close;
	case 2:
		mes .@npc_name$;
		if(Zeny < 5000){
			mes "Désolé, mais la commission est de 5 000 zeny. Veuillez vous assurer d'avoir assez d'argent pour l'enregistrement la prochaine fois, d'accord ?";
			close;
		}
		mes "Merci.";
		mes "j'espère que tu";
		mes "profitez de votre repos~";
		close2;
		set Zeny,Zeny - 5000;
		percentheal 100,100;
		return;
	case 3:
		close;
	}
}
