dali,139,118,4	script	Dimensional Device#sara	10007,{
	if (BaseLevel < 99) {
		mes "[Adventurier Leon]";
		mes "Hey ! Vous êtes trop jeune pour jouer avec ça ! Revenez quand vous aurez au moins ^0000FFNiveau 99^000000 !";
		close;
	}
	if ((saramemory < 1) || (saramemory == 2) || (saramemory == 3) || (saramemory == 5)) {
		mes "[Adventurier Leon]";
		mes "Avant de partir jouer autour de ça ; Pourquoi ne pas venir ici et me parler ?";
		close;
	}

	// 20 hour cooldown
	set .@ins_sara_time,checkquest(15002,PLAYTIME);
	if (.@ins_sara_time == -1) {

		set .@party_id,getcharid(1);
		set .@p_name$,getpartyname(.@party_id);
		set .@md_name$,"Sara's Memories";

		if (!.@party_id) {
			mes "[Adventurier Leon]";
			mes "Avant d'entrer, vous devez organiser une partie ! Vous ne savez pas comment ? Tapez /organize PARTYNAMEHERE.";
			close;
		}

		if (is_party_leader() == true)
			set .@menu$, "Boot up the dimensional device.:Use the dimensional device.:Cancel.";
		else
			set .@menu$, ":Use the dimensional device.:Cancel.";

		switch(select(.@menu$)) {
		case 1:
			if (instance_create(.@md_name$) < 0) {
				mes "Party Name: "+.@p_name$;
				mes "Party Leader: "+strcharinfo(0);
				mes "^0000ff"+.@md_name$+" ^000000- Reservation Failed!";
				close;
			}
			mes "^FF0000Le dispositif dimensionnel démarre proprement. Utilisez le dispositif pour entrer dans la fissure dans l'espace et le temps.^000000";
			close;
		case 2:
			switch(instance_enter(.@md_name$)) {
			case IE_OTHER:
				mes "Une erreur inconnue s'est produite.";
				close;
			case IE_NOINSTANCE:
				mes "The memorial dungeon "+.@md_name$+" does not exist.";
				mes "Le leader du groupe n'a pas généré le donjon encore.";
				close;
			case IE_NOMEMBER:
				mes "Only the registered members can enter the instance "+.@md_name$+".";
				close;
			case IE_OK:
				mapannounce "dali",strcharinfo(0)+", member of the party "+.@p_name$+" entered the instance "+.@md_name$+".",bc_map,"0x00ff99";
				setquest 15002;
				end;
			}
		case 3:
			mes "^FF0000Le chef de partie n'a pas démarré le dispositif dimensionnel.^000000";
			close;
		}
	} else if (.@ins_sara_time == 0 || .@ins_sara_time == 1) {
		mes "[Adventurier Leon]";
		mes "Je pense qu'environ 20 heures après votre dernière entrée dans la faille, elle sera assez stable pour que vous puissiez la traverser à nouveau.";
		close;
	}
	else {
		erasequest 15002;
		mes "^0000FF[ Souvenirs de Sara ] La période de réfréneration est écoulée. Il est à nouveau possible d'utiliser l'appareil dimensionnel.^000000";
		close;
	}
}
