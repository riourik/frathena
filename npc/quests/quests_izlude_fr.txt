// Edgar's Offer :: iz_npc
//============================================================
-	script	::Edgar_izlude	-1,{
	if (MISC_QUEST & 16) {
		mes "[Edgar]";
		mes "Alors, tu vas à Alberta encore une fois ? Laisse-moi te donner la même réduction et ne t'appliquer qu'un coût de 250 Zeny, comme la dernière fois.";
		mes "Qu'en penses-tu?";
		next;
		if (select("Alrighty~!:Why are you being so nice to me?!") == 1) {
			if (Zeny < 250) {
				mes "[Edgar]";
				mes "Euh...";
				mes "Ce n'est pas";
				mes "assez d'argent.";
				mes "Pourquoi ne vas-tu pas";
				mes "gagner un peu plus d'argent?";
				close ;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[Edgar]";
		mes "C'est juste la façon dont je suis. Et puis, ton sourire diabolique me rappelle mon fils blondinet qui a quitté le foyer des années auparavant pour devenir Marinier sur son propre bateau. Que Dieu bénisse son âme, où qu'il soit.";
		close;
	}
	mes "[Edgar]";
	mes "Ma ville, Izlude, est connectée à Alberta par le port de l'Ouest. Il y a tellement de trafic entre nous que je deviens presque Albertian. Héhéhé~";
	next;
	mes "[Edgar]";
	mes "Il y a ce type que je connais bien, Phelix, qui vit à Alberta. Ce mec est vraiment radin... Il facture pour tout !";
	next;
	mes "[Edgar]";
	mes "Mais il est vraiment une bonne personne et aime aider les autres. Il a un bon cœur et te donnera son soutien si tu acceptes ses conditions.";
	next;
	mes "[Edgar]";
	mes "Dernièrement, les gens à Alberta disent qu'il essaie vraiment d'aider les gens et que sa demande de Jellopies est juste une couverture. Eh bien, tu devrais jeter un œil à ce qu'il a à offrir.";
	next;
	if (select("Can you tell me how to get to Alberta?:End Conversation") == 1) {
		mes "[Edgar]";
		mes "Hé ? Bah, tu peux utiliser tes pieds et marcher. Mais si tu as de l'argent, je te suggère de prendre le bateau.";
		next;
		if (select("Okay, gotcha.:But I'm sick of walking and I'm broke!") == 1) {
			mes "[Edgar]";
			mes "D'accord, prends soin de toi~";
			close;
		}
		mes "[Edgar]";
		mes "D'accord...";
		mes "Vous ne voulez pas marcher ET vous n'avez pas d'argent, mais vous voulez quand même y aller ? Oh zut...";
		next;
		mes "[Edgar]";
		mes "D'accord d'accord. Moi, capitaine d'un navire, peux me permettre de vous amener là-bas à un prix réduit. Et si je prenais 250 Zeny ?";
		next;
		if (select("Alrighty~!:Bah, what a rip off!!") == 1) {
			set MISC_QUEST,MISC_QUEST | 16;
			if (Zeny < 250) {
				mes "[Edgar]";
				mes "Euh...";
				mes "Ce n'est pas";
				mes "assez d'argent.";
				mes "va chercher";
				mes "davantage.";
				close;
			}
			set Zeny, Zeny-250;
			warp "alberta",195,164;
			end;
		}
		mes "[Edgar]";
		mes "Bon sang,";
		mes "si vous pensez";
		mes "que c'est du vol...";
		close;
	}
	mes "[Edgar]";
	mes "D'accord, d'accord.";
	mes "^FF0000À plus tard~^000000";
	close;
}
