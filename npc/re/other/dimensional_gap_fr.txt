	switch(select("Open Storage 200 Zeny.:No Thanks.")) {
	case 1:
		if (Zeny > 199) {
			if(!callfunc("F_CanOpenStorage")){
				mes "[Gestionnaire Logistique]";
				mes "Vous n'êtes pas encore qualifié pour utiliser le Stockage.";
			}
			else {
				Zeny -= 200;
				openstorage;
			}
		}

			mes "Vous avez besoin d'au moins 200 zeny pour utiliser le Stockage.";
		}
		break;
	case 2:
		mes "[Gestionnaire Logistique]";
		mes "Dites-le-moi si vous changez d'avis.";
	}
	close;
}
