alde_alche,31,186,3	script	Craft Book Merchant#alde	883,{
	mes "[Vendeur de livres d'artisanat]";
	mes "Accueillir.";
	mes "Je suis ici pour vendre";
	mes "^0000FFdes livres d'artisanat spéciaux^000000 pour les Généticiens.";
	next;
	mes "[Vendeur de livres d'artisanat]";
	mes "Les Généticiens possèdent certaines compétences";
	mes "qui nécessitent des livres d'artisanat pour être activées.";
	next;
	mes "[Vendeur de livres d'artisanat]";
	mes "Plus vous avez de livres d'artisanat,";
	mes "plus vous pouvez utiliser de compétences.";
	next;
	mes "[Vendeur de livres d'artisanat]";
	mes "Comprends simplement que";
	mes "Je me spécialise dans la vente de ces livres spéciaux et";
	mes "je ^FF0000ne vends pas d'autres objets^000000.";
	next;
	mes "[Vendeur de livres d'artisanat]";
	mes "Souhaitez-vous jeter un coup d'œil à mes livres d'artisanat ?";
	next;
	if(select("Sure.:No.") == 2) {
		mes "[Vendeur de livres d'artisanat]";
		mes "Merci d'avoir visité ma boutique.";
		mes "Voyagez en toute sécurité.";
		close;
	}
	if (checkweight(1201,1) == 0) {
		mes "[Vendeur de livres d'artisanat]";
		mes "Je suis désolé, mais votre inventaire est presque plein.";
		mes "Veuillez vider votre inventaire en premier.";
		close;
	}
	if (MaxWeight - Weight < 2500) {
		mes "[Vendeur de livres d'artisanat]";
		mes "Veuillez vider votre inventaire en premier.";
		close;
	}
	mes "[Vendeur de livres d'artisanat]";
	mes "D'accord, voici la liste des livres d'artisanat disponibles.";
	next;
	switch(select("[Apple Bomb Craft Book] 100,000 zeny:[Pineapple Bomb Craft Book] 100,000 zeny:[Coconut Bomb Craft Book] 100,000 zeny:[Melon Bomb Craft Book] 100,000 zeny:[Banana Bomb Craft Book] 100,000 zeny:[Plant Gene Cultivation Method] 100,000 zeny:[Superior Potion Craft Manual] 100,000 zeny:[Mix Cooking Book] 100,000 zeny:[Health Improvement Research Book] 100,000 zeny:[Vigor Drink Recipe] 100,000 zeny:Close")){
	case 1: set .@item,6279; break;
	case 2: set .@item,6280; break;
	case 3: set .@item,6281; break;
	case 4: set .@item,6282; break;
	case 5: set .@item,6283; break;
	case 6: set .@item,6284; break;
	case 7: set .@item,6285; break;
	case 8: set .@item,11022; break;
	case 9: set .@item,11023; break;
	case 10: set .@item,11024; break;
	case 11:
		mes "[Vendeur de livres d'artisanat]";
		mes "Merci pour votre mécénat.";
		mes "S'il vous plaît, revenez.";
		close;
	}
	mes "[Vendeur de livres d'artisanat]";
	mes "^0000FF["+getitemname(.@item)+"] costs";
	mes "^FF0000100 000 zeny.^0";
	mes "Combien souhaitez-vous acheter ?";
	next;
	input .@amount;
	if (.@amount == 0) {
		mes "[Vendeur de livres d'artisanat]";
		mes "Souhaitez-vous voir d'autres livres ?";
		close;
	}
	if (.@amount < 1 || .@amount > 99 ){
		mes "[Vendeur de livres d'artisanat]";
		mes "Vous ne pouvez pas en acheter plus de 100 à la fois.";
		close;
	}
	mes "[Vendeur de livres d'artisanat]";
	mes "You've entered "+.@amount+"x ^0000FF["+getitemname(.@item)+"]^000000 to be purchased.";
	mes "Souhaitez-vous continuer ?";
	next;
	if (select("Yes:No") == 1) {
		set .@total, .@amount * 100000;
		if (Zeny < .@total) {
			mes "[Vendeur de livres d'artisanat]";
			mes "Je suis désolé, mais vous n'avez pas assez d'argent.";
			close;
		}
		if (checkweight(.@item,.@amount) == 0){
			mes "[Vendeur de livres d'artisanat]";
			mes "Il semble que vous ne puissiez tout transporter.";
			mes "Veuillez vérifier l'espace disponible dans votre inventaire.";
			close;
		}
		mes "[Vendeur de livres d'artisanat]";
		mes "Merci pour votre mécénat.";
		set Zeny,Zeny - .@total;
		getitem .@item,.@amount;
		close;
	}
	mes "[Vendeur de livres d'artisanat]";
	mes "Prenez votre temps";
	mes "avant que vous preniez votre décision.";
	close;
}
