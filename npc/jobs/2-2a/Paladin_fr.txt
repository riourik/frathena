valkyrie,53,39,3	script	Paladin#Valkyrie	752,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Paladin]";
			mes "Félicitations.";
			mes "Honneur aux guerriers !";
			close;
		}
		mes "[Paladin]";
		mes "Ne l'oublie pas";
		mes "que la Sainte Guerre";
		mes "est sur le point d'éclater !";
		mes "Nous devons nous préparer !";
		mes "Que la lumière de la justice";
		mes "éclaire toujours notre chemin !";
		close;
	}
	else if (ADVJOB == Job_Paladin && Class == Job_Swordman_High && JobLevel > 39) {
		mes "[Paladin]";
		mes "La Sainte Guerre sera";
		mes "sur nous avant que nous ne";
		mes "nous en rendions compte. Plus";
		mes "que jamais, nous avons besoin";
		mes "de gens forts pour combattre";
		mes "pour ce qui est bon et juste.";
		next;
		mes "[Paladin]";
		mes "Combattras-tu du côté de la";
		mes "justice en tant que Paladin?";
		mes "as a Paladin?";
		next;
		if (select("No.:Yes.") == 1) {
			mes "[Paladin]";
			mes "[Grand Prêtre]";
			mes "n'hésitez pas à revenir.";
			mes "Honneur aux guerriers !";
			close;
		}
		if (SkillPoint) {
			mes "[Paladin]";
			mes "Il est encore possible pour vous d'apprendre des compétences supplémentaires. Veuillez utiliser";
			mes "tous vos points de compétence restants avant de revenir me voir.";
			close;
		}
		jobchange Job_Paladin;
		set ADVJOB,0;
		mes "[Paladin]";
		mes "Félicitations.";
		mes "En tant que Paladin, j'espère";
		mes "que vous protégerez ceux";
		mes "qui sont plus faibles que vous, et apporterez";
		mes "la victoire à notre camp lors de la guerre imminente";
		mes "entre le bien et le mal.";
		close;
	}
	mes "[Paladin]";
	mes "Accueillir";
	mes "vers Valhalla,";
	mes "la Salle de l'Honneur.";
	next;
	mes "[Paladin]";
	mes "S'il vous plaît faites-vous";
	mes "confortable";
	mes "pendant votre séjour ici.";
	mes "Honneur aux guerriers !";
	close;
}
