// Stephen - Valentine Event Chocolate seller
//============================================================
alberta,26,243,4	script	Stephen	58,{
	mes "[Stephen]";
	mes "Devine ce que j'ai?";
	mes "Un petit plaisir qu'on ne trouve pas facilement dans Midgard....";
	next;
	mes "[Stephen]";
	mes "Du chocolat!";
	mes "C'est ça, tu aimes le chocolat... moi aussi.";
	mes "Et tu es chanceux car je les vends pour seulement 5,000 zeny l'unité!";
	next;
	if(select("I want some chocolate!:No thanks.")==2) {
		mes "[Stephen]";
		mes "Tu ne veux pas de chocolat?";
		mes "Je te le dis ! Tu vas le regretter !";
		mes "Il faudrait que tu t'en achètes maintenant... car tu n'auras jamais la chance de trouver du chocolat comme ça à nouveau!";
		mes "Réfléchis-y et reviens me voir quand tu veux.";
		close; }
	mes "[Stephen]";
	mes "Hah!";
	mes "Je savais que c'était ça!";
	mes "Mais je ne peux pas te vendre plus de 5 à la fois... mais, si tu en as vraiment besoin...";
	mes "tu peux revenir me voir.";
	mes "Alors combien en veux-tu?";
	next;
	input .@i;
	if (.@i <= 0) close;
	mes "[Stephen]";
	if (.@i > 5)  {
		mes "Je suis désolé, mais je ne peux pas te donner autant.";
		close; }
	if (Zeny < .@i*5000) {
		mes "Je suis désolé, mais il semble que vous ne pouvez pas vous offrir ces articles.";
		close; }
	set Zeny, Zeny - .@i*5000;
	getitem 558,.@i;
	mes "Et voilà !";
	mes "Vous pouvez l'offrir à quelqu'un comme cadeau, ou en profiter vous-même !";
	mes "Hmm... du sucre de chocolat...";
	mes "N'hésitez pas à me rendre visite quand bon vous semble...!";
	close;
}
