//===== Compatible With: =====================================
//= rAthena Project
//===== Description: =========================================
//= Removes cards from equipped items.
//===== Additional Comments: =================================
//= 1.0 First version. [TyrNemesis^]
//= 1.2 Optimized and fixed getequipname menu. [Kisuka]
//= 1.2a Added 'disable_items' command. [Euphy]
//============================================================

prt_in,28,73,4	script	Wise Old Woman#eAcustom	78,{

	set .zenycost,200000;    // base cost of the card remover services (in Zeny)
	set .percardcost,25000;  // cost per card of the card remover services (in Zeny)
	set .faildestroy,1;      // should the card remover have a chance of failure that destroys items? (1=yes, 0=no)

	disable_items;
	mes "[Vieille femme sage]";
	mes "Bonne journée, mon jeune. J'ai le pouvoir de supprimer les cartes que vous avez ajoutées à votre équipement. Cette idée vous plaît ?";
	next;
	switch(select("Yes, it does.:What do you charge?:No thanks.")) {
	case 1:
		mes "[Vieille femme sage]";
		mes "Très bien. Quel article dois-je examiner pour vous ?";
		next;

		setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
		for( set .@i,1; .@i <= 10; set .@i,.@i+1 ) {
			if( getequipisequiped(.@indices[.@i]) )
				set .@menu$, .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			set .@menu$, .@menu$ + ":";
		}
		set .@part, .@indices[ select(.@menu$) ];
		if(!getequipisequiped(.@part)) {
			mes "[Vieille femme sage]";
			mes "Jeune... Vous ne portez rien là-bas dont je puisse retirer des cartes.";
			close;
		}
		if(getequipcardcnt(.@part) == 0) {
			mes "[Vieille femme sage]";
			mes "Jeune... Il n'y a aucune carte composée sur cet objet. Je ne peux rien y faire, j'en ai peur.";
			close;
		}
		set .@cardcount,getequipcardcnt(.@part);
		
		if (!checkweight(1202,(.@cardcount+1))) { 
			mes "^3355FFJuste une minute !";
			mes "Je ne peux offrir aucun de mes";
			mes "services pour vous parce que";
			mes "tu en portes trop";
			mes "truc. Mettez vos articles supplémentaires dans";
			mes "Kafra Storage et revenez ~";
			close;
		}
		mes "[Vieille femme sage]";
		mes "This item has " + .@cardcount + " cards compounded on it. To perform my magic, I will need " + (.zenycost+(.@cardcount * .percardcost)) + " zeny, a ^0000FFStar Crumb^000000, and a ^0000FFYellow Gemstone^000000.";
		next;
		if(select("Very well. Do it.:Never mind.") == 2) {
			mes "[Vieille femme sage]";
			mes "Très bien. Revenez immédiatement si vous sollicitez mes services.";
			close;
		}
		if((zeny < (.zenycost+(.@cardcount * .percardcost))) || (countitem(1000) < 1) || (countitem(715) < 1)) {
			mes "[Vieille femme sage]";
			mes "Tu n'as pas tous les objets dont j'ai besoin pour faire fonctionner ma magie, mon enfant. Revenez quand vous le ferez.";
			close;
		}
		mes "[Vieille femme sage]";
		mes "Avant de commencer, je dois vous prévenir : je pourrais échouer. Si je le fais, je peux détruire les cartes, l'objet ou les deux. Je ne fais pas de remboursement. Cela étant dit, qu’est-ce qui est le plus important pour vous : les cartes ou l’objet ?";
		next;
		switch(select("I changed my mind about this.:The item.:The cards.")) {
		case 1:
			mes "[Vieille femme sage]";
			mes "Très bien. Revenez immédiatement si vous sollicitez mes services.";
			close;
		case 2:
			set .@failtype,1;
			break;
		case 3:
			set .@failtype,2;
		}
		mes "[Vieille femme sage]";
		mes "Très bien. Je vais commencer.";
		set Zeny, Zeny - (.zenycost+(.@cardcount * .percardcost));
		delitem 1000,1; //Star_Crumb
		delitem 715,1; //Yellow_Gemstone
		
		// Replace the constants in the next 3 lines with failure chance values defined in refine_db.txt
		// First value = Total failure chance (item and cards destroyed)
		// Second value = Partial failure chance (one or the other is destroyed, player decides which one is safe)
		// Third value = Harmless failure chance (all that's lost is your investment)

		set .@failchance,rand(100);
		if (.faildestroy==1) {
			if(.@failchance < 2) {
				next;
				failedremovecards .@part,0;
				mes "[Vieille femme sage]";
				mes "Le processus a été un échec total. Je crains que l'objet et les cartes aient été détruits.";
				close;
			}

			if(.@failchance < 8) {
				if (.@failtype == 1) {
					next;
					failedremovecards .@part,1;
					mes "[Vieille femme sage]";
					mes "Même si j'ai réussi à retirer les cartes de l'objet, elles ont été détruites au cours du processus. L'article est cependant ok.";
					close;
				}

				if (.@failtype == 2) {
					next;
					failedremovecards .@part,2;
					mes "[Vieille femme sage]";
					mes "Très malheureux. J'ai réussi à retirer les cartes, mais l'objet lui-même a été détruit au cours du processus.";
					close;
				}
			}
		}

		if(.@failchance < 10) {
			next;
			failedremovecards .@part,3;
			mes "[Vieille femme sage]";
			mes "Je n'ai pas réussi à retirer les cartes. Heureusement, l’objet et les cartes sont toujours en bon état.";
			close;
		}
		next;
		successremovecards .@part;
		mes "[Vieille femme sage]";
		mes "Le processus a été un succès. Voici vos cartes et votre article. Adieu.";
		close;
	case 2:
		mes "[Vieille femme sage]";
		mes "I charge a flat fee of "+callfunc("F_InsertComma",.zenycost)+" zeny, plus "+callfunc("F_InsertComma",.percardcost)+" zeny for each card I remove from the item. In addition, I need a star crumb and a yellow gemstone to work my magic.";
		close;
	case 3:
		mes "[Vieille femme sage]";
		mes "Très bien. Revenez immédiatement si vous sollicitez mes services.";
		close;
	}
}
