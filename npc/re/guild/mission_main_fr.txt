function calcjobexp;

	if (!checkweight(1201,3)){

		mes "Veuillez alléger votre charge et réessayez.";
		close;
	}

	.@zone$ = strnpcinfo(2);
	if(.@zone$ == "gqaldeg"){
		.@zoneid = 3;
		.@edition = 1;
	} else if(.@zone$ == "gqaru"){
		.@zoneid = 4;
		.@edition = 2;
	} else if(.@zone$ == "gqgefg"){
		.@zoneid = 2;
		.@edition = 1;
	} else if(.@zone$ == "gqpayg"){
		.@zoneid = 1;
		.@edition = 1;
	} else if(.@zone$ == "gqprtg"){
		.@zoneid = 0;
		.@edition = 1;
	} else if(.@zone$ == "gqsch"){
		.@zoneid = 5;
		.@edition = 2;
	} else if(.@zone$ == "gqtealde"){
		.@zoneid = 6;
		.@edition = 3;
	} else if(.@zone$ == "gqteprt"){
		.@zoneid = 7;
		.@edition = 3;
	} else {
		// Custom error message.
		debugmes "Guild mission script called from an unknown agit region [" + .@zone$ + "].";
		end;
	}
	.@npcname$ = .npcname$[.@zoneid];

	mes .@npcname$;
	mes "We provide missions and daily quests that you can complete in the region of " + .region$[.@zoneid] + ".";
	mes "Il existe des quêtes quotidiennes simples ainsi que des missions que vous devez accomplir uniquement pendant les heures de siège. Pensez-y donc~";
	next;
	switch(select("Conquering " + .region$[.@zoneid], "Guild Dungeon Daily Quests", "About quests", "I don't need it.")){
		case 1: // Conquering.
			.@gid = getcharid(2);
			if(.@gid && is_guild_leader() == true){
				.@time_check1 = checkquest(.conquer_delay_questid[.@zoneid], PLAYTIME);
				if(.@time_check1 == 0){
					mes .@npcname$;
					mes "Il y a moins d'une heure depuis la dernière fois que nous avons vérifié !";
					mes "Revenez lorsque le temps d'attente sera écoulé.";
					next;
					mes .@npcname$;
					mes "Si vous avez de la chance, vous obtiendrez de bons résultats.";
					close;
				} else if(.@time_check1 == 2){
					erasequest .conquer_delay_questid[.@zoneid];
					mes .@npcname$;
					mes "J'ai confirmé que le temps d'attente pour les missions de conquête est terminé.";
					mes "Je supprimerai les enregistrements inutiles pour vous.";
					close;
				} else {
					.@siege_check = (.@edition == 1 ? agitcheck() : .@edition == 2 ? agitcheck2() : agitcheck3());
					.@time = atoi(gettimestr("%H%M",4));
					.@after_time = ((.weekday[.@zoneid] == gettime(4)) && ((.@time > .start_time[.@zoneid]) && (.@time < .end_time[.@zoneid])));

					if(.@siege_check && .@after_time){
						mes .@npcname$;
						mes "Selon ^4d4dffle nombre de forteresses que votre guilde a conquises à cet instant précis,^000000 je vous remettrai une récompense.";
						mes "Voulez-vous vérifier maintenant ? Vous savez bien que ^4d4dffce chiffre peut changer même en ce moment-même,^000000 non ?";
						next;
						if(select("I'll come back later.","Let's check right now!") == 1){
							mes .@npcname$;
							mes "Exact. La vie est tout question de timing.";
							mes "Revenez lorsque cela brillera le plus.";
							close;
						} else {
							.@total = 0;
							for(.@i = 0; .@i < 5; ++.@i){
								.@total += (.@gid == getcastledata(.castle$[.@i+.@zoneid*5],CD_GUILD_ID));
							}

							if(!.@total){
								mes .@npcname$;
								mes "Malheureusement, il n'y a aucune forteresse que votre guilde ait conquise.";
								mes "Réessayez !";
								close;
							} else {
								mes .@npcname$;
								if(.@total == 5)
									mes "Incroyable.";
								else
									mes "Actuellement, le nombre total de forteresses que votre guilde a conquises est de :";
								for(.@i = 0; .@i < 5; ++.@i){
									mes getcastlename(.castle$[.@i+.@zoneid*5]) + (.@gid == getcastledata(.castle$[.@i+.@zoneid*5],CD_GUILD_ID) ? ": ^4d4dffOccupied^000000" : "");

						mes "Je vous donnerai des points d'expérience en retour.";
					next;

						mes .@npcname$;
						mes "L'accès à la donjon de guilde est restreint. Mais ce n'est pas comme si vous ne pouviez pas trouver un moyen d'y aller non plus.";
						mes "Je prierai pour votre succès dans le combat contre ces monstres.";
						setquest .daily_hunting_questid[.@index];
						close;
					} else {
						mes .@npcname$;
						mes "Eh bien, faites comme vous le souhaitez.";
						close;
					}
				}
			}
			break;

			mes .@npcname$;
			mes "Ce n'est du tout difficile.";
			mes "Je donne une récompense ^4d4dffselon le nombre de forteresses que vous avez conquises.^000000";

			mes "^4d4dffRecevoir une récompense est suivi d'une heure de temps d'attente. Il vaut donc mieux calculer le bon moment également.^000000";
			next;
			mes .@npcname$;
			mes "Si vous êtes chef d'une guilde qui participe actuellement aux batailles d'assaut, vous devriez essayer.";
			next;

			mes "Les quêtes quotidiennes de donjon de guilde consistent en des quêtes de combat que vous pouvez accomplir dans les donjons de guilde.";
			mes "C'est simple.";
			close;
		case 4: // I don't need it.
			mes .@npcname$;
			mes "Vous êtes vraiment quelque chose d'autre~!";
			close;
	}
	end;

	function calcexp {
		.@zoneid = getarg(0);
		.@index = getarg(1);
		.@maxlevel = .max_level[.@index];
		.@isboss = (.@zoneid < 4 && !((.@index+1)%3));
		
		if(!.@isboss && BaseLevel > .@maxlevel){
			return 1000 * (100 + ((.@maxlevel - 100) * 3));
		} else {
			return 1000 * (100 + ((BaseLevel - 100) * 3)) * (1 + .@isboss);
		}
	}
	
	function calcjobexp {
		.@edition = getarg(0);
		
		if(.@edition == 3){
			if(jobcanentermap("te_prtcas01")){
				if (JobLevel > 50)
					return 1000 * (50 + ((JobLevel - 50) * 3));
				else
					return 5000;
			} else {
				debugmes "Unexpected case for Job: " + Class + "(" + jobname(Class) + ") for gld_mission#core::calcjobexp";
				return 0;
			}
		} else if(.@edition == 2 ) {
			if(JobLevel > 50)
				return 1000 * (50 + ((JobLevel - 50) * 3));
			else
				return 3000;
		} else {
			return 5000 * JobLevel;
		}
		return 0;
	}
