payon_in01,193,30,4	script	Rice Mill Grandma#rat	78,{
	mes "[Rice Mill Grandma]";
	if (!lunar_rat) {
		mes "ces sales petits";
		mes "créatures ! Se faufilant";
		mes "ici et là, volant ce qui";
		mes "m'appartient ! Si seulement ils";
		mes "n'étaient pas aussi sacrément rapides...";
		next;
		select("What's wrong?");
		mes "[Rice Mill Grandma]";
		mes "Oh, je préparais simplement quelques";
		mes "galettes et pâtisseries pour célébrer";
		mes "la nouvelle année, mais ces animaux ont";
		mes "été voler les Poches de Riz que j'ai préparées.";
		mes "[Grand-mère du Moulin de Riz]";
		mes "Oui, mon fils a besoin de ces";
		next;
		select("Rice Pouches?");
		mes "[Rice Mill Grandma]";
		mes "faire de pâtisseries si je n'ai même pas de riz. Penses-tu que tu";
		mes "pourrais m'aider ?";
		mes "rice in a mortar, but I can't";
		mes "make any pastries if I don't";
		mes "even have the rice. Do you";
		mes "think you can help me?";
		next;
		if (select("Of course.:I don't even know you.") == 1) {
			set lunar_rat,1;
			mes "[" + strcharinfo(0)+"]";
			mes "^FF0000Bien sûr.^0";
			mes "^FF0000Que puis-je faire pour vous ?^0";
			next;
			mes "[Rice Mill Grandma]";
			mes "Eh bien, pensez-vous que vous";
			mes "pouvez attraper les animaux qui";
			mes "ont volé mes Poches de riz ?";
			mes "Ce sont ces rats bleus et";
			mes "blancs et ces";
			mes "petits taupes méchantes.";
			next;
			select("Not a problem.");
			mes "[" + strcharinfo(0)+"]";
			mes "Pas de problème.";
			mes "Je vais attraper ces animaux,";
			mes "et ramener toute Poche de riz que";
			mes "je peux trouver.";
			next;
			mes "[Rice Mill Grandma]";
			mes "Oh, merci !";
			mes "Bonne chance pour attraper ces";
			mes "nuisibles pour moi~";
			close;
		}
		mes "[Rice Mill Grandma]";
		mes "Je... Eh bien...";
		mes "C'est vrai, mais j'étais";
		mes "^FF0000je compte vous en donner";
		mes "^FF0000en retour quelque chose de sympa";
		mes "^FF0000pour votre aide. Je sais que vous n'êtes";
		mes "^FF0000pas une mauvaise personne, alors...";
		close;
	}
	else if ((lunar_rat >= 1 && lunar_rat <= 2) || lunar_rat == 4) {
		if (countitem(7770) >= 1 && lunar_rat == 2) {
			mes "^FF0000Oh, est-ce le";
			mes "^FF0000Sweet Rice^FF0000 (Riz sucré) que mon fils a fait?";
			mes "^FF0000Voudriez-vous me le laisser?";
			mes "^FF0000S'il vous plaît attendez ici un moment,";
			mes "^FF0000et je vais vous faire une pâtisserie~";
			delitem 7770,1; //Sweet_rice
			set lunar_rat,3;
			close;
		}
		else if (lunar_rat == 4) {
			if (countitem(7770) >= 1) {
				mes "^FF0000Oh, est-ce le";
				mes "^FF0000Sweet Rice^FF0000 (Riz sucré) que mon fils a fait?";
				mes "^FF0000Voudriez-vous me le laisser?";
				mes "^FF0000S'il vous plaît attendez ici un moment,";
				mes "^FF0000et je vais vous faire une pâtisserie~";
				delitem 7770,1; //Sweet_rice
				set lunar_rat,5;
				close;
			}
			mes "^FF0000Oh, où sont tous les";
			mes "^FF0000Rice Pouches^FF0000 (Sac de riz)? Nous en avons besoin";
			mes "^FF0000pour faire plus de gâteaux au riz...";
			close;
		}
		else if (countitem(7869) < 1 && lunar_rat <= 2) {
			mes "^FF0000Oh, ces petits souris blanches et";
			mes "^FF0000bleues pourraient avoir";
			mes "^FF0000empoigné mes Rice Pouches.";
			mes "Ces taupins doivent avoir pris";
			mes "ça aussi. Quels vilains petits";
			mes "bêtes, n'est-ce pas?";
			close;
		}
		mes "Oh, est-ce l'un de mes";
		mes "Sac de Riz ? Je suis désolé(e),";
		mes "mais la douleur à mes hanches..";
		mes "Ne seriez-vous pas assez gentil";
		mes "pour me faire un service et";
		mes "le livrer à mon fils pour moi?";
		set lunar_rat,2;
		close;
	}
	else if (lunar_rat == 3 || lunar_rat == 5) {
		set .@reward,rand(1,100);
		if (.@reward <= 5) {
			if (lunar_rat == 3) {
				getitem 9038,1; // New_Year_Doll_Egg
				set lunar_rat,4;
			}
			else if (lunar_rat == 5) {
				getitem 668,1; // Red_Envelope
			}
		}
		else if (.@reward >= 6 && .@reward <= 10) getitem 12198,2; // Lucky_Rice_Cake
		else if (.@reward >= 11 && .@reward <= 30) getitem 12195,3; // Plain_Rice_Cake
		else if (.@reward >= 31 && .@reward <= 40) getitem 12196,2; // Hearty_Rice_Cake
		else if (.@reward >= 41 && .@reward <= 55) getitem 12123,2; // Honey_Pastry
		else if (.@reward >= 56 && .@reward <= 70) getitem 12122,2; // Sesame_Pastry
		else if (.@reward >= 71 && .@reward <= 80) getitem 12124,2; // Rainbow_Cake
		else if (.@reward >= 81 && .@reward <= 90) getitem 12198,2; // Korean_Rice_Cake
		else if (.@reward >= 91 && .@reward <= 100) getitem 12197,3; // Salty_Rice_Cake
		// After getting the egg there is a chance to get an envelope with each turn in
		if (rand(1,20) <= 5 && lunar_rat == 4) getitem 668,1; // Red_Envelope
		// Set the quest accordingly for repeat turn-ins.
		if (lunar_rat == 3) set lunar_rat,2;
		else if (lunar_rat == 5) set lunar_rat,4;
		mes "Ce n'est pas suffisant pour vous";
		mes "remercier de ce que vous avez fait";
		mes "pour moi, mais j'aimerais que vous";
		mes "ayez cette pâtisserie que je viens";
		mes "de faire. Profitez-en~";
		next;
		mes "[Rice Mill Grandma]";
		mes "Merci pour le";
		mes "Gros Riz Doux ! Je vais être sûr de vous";
		mes "faire quelque chose de délicieux si vous";
		mes "apportez plus, d'accord?";
		mes "bring me more, okay?";
		close;
	}
}
