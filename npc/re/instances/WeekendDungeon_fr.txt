pay_arche,44,124,5	script	Marry Jae	4_F_JOB_BLACKSMITH,{
	mes "[Mariage Jae]";
	if (BaseLevel < 60) {
		mes "Votre niveau de base est trop bas, revenez lorsque vous atteindrez le niveau 60+.";
		close;
	}
	mes "^ff0000Si vous tentez d'apprivoiser des monstres, cela ne comptera pas comme une tuerie. Veuillez faire attention.^000000";
	next;
	.@day = gettime(DT_DAYOFWEEK);
	if (.@day != SATURDAY && .@day != SUNDAY) {
		mes "[Mariage Jae]";
		mes "Oh, un observateur curieux ? Je suis Marry Jae, la personne en charge des week-ends parmi les jours de la semaine.";
		next;
		mes "[Mariage Jae]";
		mes "Quel est le donjon du weekend ? En réalité, c'est un espace spécialement conçu pour que les nouveaux aventuriers s'entraînent.";
		next;
		mes "^0000FFEn regardant autour de soi, il est évident que Marry Jae parle à vous.^000000";
		next;
		mes "[Mariage Jae]";
		mes "Pour ceux qui n'ont pas de groupe le week-end, cet endroit est un terrain de chasse et d'acquisition d'expérience pour les aventuriers solitaires !";
		next;
		mes "[Mariage Jae]";
		mes "De plus, la gain d'expérience ici est très généreux.";
		next;
		select( "When is the opening time?" );
		mes "[Mariage Jae]";
		mes "Le donjon sera ouvert pendant :";
		mes "^0000FFSamedi 00:00 ~ Dimanche 23:59^000000";
		next;
		mes "[Mariage Jae]";
		mes "J'espère vous voir lorsque le week-end arrivera~";
		close;
	}
	.@md_name$ = "Weekend Dungeon";

	if (getcharid(1) < 1) {
		mes "[Mariage Jae]";
		mes "Vous devez former un groupe d'un membre ou plus.";
		close;
	}
	switch( checkquest(12378,PLAYTIME) ) {
	case -1:
		break;
	case 0:
	case 1:
		switch( checkquest(12377,PLAYTIME) ) {
		case -1:
		case 2:
			mes "[Mariage Jae]";
			mes "Sorry, but the entry period to the " + .@md_name$ + " has expired.";
			mes "Prenez un repos en attendant que le donjon soit à nouveau disponible.";
			close;
		case 0:
		case 1:
			break;
		}
		break;
	case 2:
		erasequest 12377;
		erasequest 12378;
		mes "[Mariage Jae]";
		mes "Le temps de réinitialisation est écoulé.";
		mes "Vous pouvez maintenant réentrer dans le Donjon Commémoratif.";
		close;
	}

	if (is_party_leader() == true)
		.@menu$ = "Prepare Memorial Dungeon";

	mes "[Mariage Jae]";
	mes "Je pense que vous êtes prêt.";
	mes "Avez-vous envie de l'accepter maintenant ?";
	next;
	switch( select( .@menu$, "Enter Weekend Dungeon", "Cancel" ) ) {
	case 1:
		instance_create(.@md_name$);
		mes "[Mariage Jae]";
		mes "Le Donjon Commémoratif a été créé.";
		mes "Vous pouvez maintenant entrer.";
		close;
	case 2:
		switch( instance_enter(.@md_name$) ) {
		case IE_NOMEMBER:
			end;
		case IE_NOINSTANCE:
			mes "Aucun donjon commémoratif n'est enregistré.";
			close;
		case IE_OTHER:
			mes "Une erreur inconnue s'est produite.";
			close;
		case IE_OK:
			mes "[Mariage Jae]";
			mes "Allez-y !";
			mapannounce "pay_arche", "" + strcharinfo(0) + " of the party, " + getpartyname( getcharid(1) ) + ", is entering the " + .@md_name$ + ".", bc_map, 0xFF99;
			if (isbegin_quest(12377) == 0)
				setquest 12377;
			if (isbegin_quest(12378) == 0)
				setquest 12378;
			end;
		}
		end;
	case 3:
		mes "[Mariage Jae]";
		mes "Vous n'êtes pas prêt ?";
		mes "Je ne peux vraiment pas y changer, supposons.";
		close;
	}
}

		.@item = 23142;
		.@box = 23144;
		break;

		.@item = 23143;
		.@box = 23145;
		break;
	}
	mes "[Vendeur de fournitures pour cadeaux]";
	if (countitem(.@item) < 5) {
		mes "Veuillez revenir lorsque vous aurez assez de matériaux.";
		close;
	}
	if (Zeny < .@zeny) {
		mes "Veuillez revenir lorsque vous aurez assez de Zeny.";
		mes F_InsertComma(.@zeny) + "z to be specific.";
		close;
	}
	if (checkweight(23144,1) == 0) {
		mes "Vous avez trop de types d'objets. Videz votre inventaire.";
		close;
	}
	if (MaxWeight - Weight < 10000) {
		mes "Impossible de continuer car vous avez trop d'objets lourds. Réduisez votre poids total en vidant votre inventaire.";
		close;
	}
	mes "Confirm to spend " + F_InsertComma(.@zeny) + "z to package 5 <ITEM>" + getitemname(.@item) + "<INFO>" + .@item + "</INFO></ITEM> into <ITEM>" + getitemname(.@box) + "<INFO>" + .@box + "</INFO></ITEM>?";
	next;
	if (select( "Yes", "No" ) == 2) {
		mes "[Vendeur de fournitures pour cadeaux]";
		mes "Revenez quand vous aurez besoin de services d'emballage !";
		close;
	}
	Zeny -= .@zeny;
	delitem .@item,5;
	getitem .@box,1;
	end;
}
