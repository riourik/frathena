grademk,34,184,4	script	Sratos#sratos	4_JP_GARM_H,{
	mes "[Sratos]";
	mes "Bonjour, cher client qui est entré, espérons un miracle aujourd'hui.";
	mes "Comment puis-je t'aider?";
	next;
	switch( select( "Enhance the equipment's grade.", "Exchange Etel items", "What is Grade Enhancement?" ) ){
		case 1:
			mes "[Sratos]";
			mes "Notre client. Vous souhaitez améliorer le niveau de l'équipement.";
			mes "Il n'est pas facile de déplacer la puissance magique des gemmes.";
			next;
			mes "[Sratos]";
			mes "La puissance magique peut exploser sur place.";
			mes "Alors l'arme qui héritera de cette puissance magique explose également !";
			next;
			switch( select( "I'll still do it!", "I'll think about it." ) ){
				case 1:
					mes "[Sratos]";
					mes "Je souhaite bonne chance à nos courageux clients !";
					close2;
					enchantgradeui();
					end;
				case 2:
					mes "[Sratos]";
					mes "Chaque fois que vous aurez le courage de relever le défi, revenez.";
					close;
			}
		case 2:
			mes "[Sratos]";
			mes "Ce sont des gemmes utilisées pour améliorer les niveaux.";
			mes "Poussière d'Etel, les gemmes qui serviront de base, et si vous me versez une petite somme, je l'échangerai contre des gemmes d'Etel.";
			close2;
			callshop "EnchantGradeExchange";
			end;

		case 3:
			mes "[Sratos]";
			mes "Parfois, on trouve des bijoux instables contenant beaucoup de puissance magique pure.";
			mes "On appelle cela des pierres éthérées, n'est-ce pas ?";
			next;
			mes "[Sratos]";
			mes "Vous pouvez transférer la puissance magique de la pierre éthérée vers d'autres armes, mais bien sûr, il y a un certain risque.";
			mes "C'est fragile, donc si vous n'êtes pas prudent, paf ! Cela va exploser.";
			next;
			mes "[Sratos]";
			mes "De toute façon, si vous utilisez cette gemme éthérique, vous pouvez améliorer l'arme que vous utilisez.";
			mes "Les performances s'améliorent. Nous appelons cela une amélioration de grade.";
			next;
			mes "[Sratos]";
			mes "Vous aussi, ne seriez-vous pas heureux si les armes que vous aimez pouvaient progresser d'un cran ?";
			mes "Nous sommes ceux qui y assistons. C'est un peu risqué, cependant. ahahaha.";
			close;
	}

OnInit:
	setunittitle(getnpcid(0), "<Grade Enhancer>");
	end;
}
