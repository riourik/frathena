mosk_dun02,53,220,4	script	#babayagain	WARPNPC,1,1,{
	end;
OnTouch:
	if (mos_nowinter == 11) {
		donpcevent "Soldier1#mos::OnEnable";
		donpcevent "Soldier2#mos::OnEnable";
		donpcevent "Soldier3#mos::OnEnable";
		mes "[Soldat]";
		mes "Arrêtez-vous là !";
		next;
		mes "[Soldat]";
		mes "Nous avons des informations";
		mes "selon lesquelles vous avez secrètement";
		mes "communiqué avec";
		mes "Baba Yaga, l'Horrible.";
		next;
		mes "[Soldat]";
		mes "Si vous avez quelque chose à dire,";
		mes "parlez au Csar.";
		mes "Suivez-nous.";
		close2;
		donpcevent "Soldier1#mos::OnInit";
		donpcevent "Soldier2#mos::OnInit";
		donpcevent "Soldier3#mos::OnInit";
		set mos_nowinter,12;
		warp "mosk_in",125,92;
		end;
	}
	else if (mos_nowinter == 12 || mos_nowinter == 13) {
		donpcevent "Soldier1#mos::OnEnable";
		donpcevent "Soldier2#mos::OnEnable";
		donpcevent "Soldier3#mos::OnEnable";
		mes "[Soldat]";
		mes "Nous pensions";
		mes "que vous réapparaîtriez.";
		mes "Quel est votre but auprès du Csar ?";
		next;
		if(select("Run away.:Explain the situation.") == 1) {
			mes "["+strcharinfo(0)+"]";
			mes "Je devrais sortir d'ici...";
			close2;
			warp "mosk_dun02",100,100;
		}
		else {
			mes "-J'explique ce qui s'est passé jusqu'à présent.-";
			next;
			mes "[Soldat]";
			mes "Si vous avez quelque chose à dire, parlez";
			mes "au Csar.";
			mes "Si vous quittez cet endroit,";
			mes "vous serez considéré comme notre ennemi.";
			close2;
		}
		donpcevent "Soldier1#mos::OnInit";
		donpcevent "Soldier2#mos::OnInit";
		donpcevent "Soldier3#mos::OnInit";
		end;
	}
	else {
		warp "mosk_in",215,41;
		end;
	}
}
