S_SellSets:
	.@item_id = getarg(0);
	.@item_cost = getiteminfo(.@item_id, ITEMINFO_BUY);
	.@item_weight = getiteminfo(.@item_id, ITEMINFO_WEIGHT);
	mes "[Madeleine Chu]";
	mes "How many "+ (.@item_id == 12125 ? "Outdoor":"Indoor");
	mes "Les kits de cuisine";
	mes "tu aimes acheter ? Si tu";
	mes "je veux annuler, s'il te plaît";
	mes "entrez le chiffre 0.";
	next;
	while(1) {
		input .@sell;
		if (.@sell == 0) {
			mes "[Madeleine Chu]";
			mes "Vous avez changé votre";
			mes "esprit? Eh bien, si tu as besoin";
			mes "to buy "+ getitemname(.@item_id);
			mes "Kits plus tard, revenez";
			mes "pour moi à tout moment, d'accord ?";
			close;
		}
		else if (.@sell > 100) {
			mes "[Madeleine Chu]";
			mes "Oh, je suis désolé, mais";
			mes "je ne vends pas plus que";
			mes "100 "+ getitemname(.@item_id);
			mes "à la fois, juste pour être en sécurité.";
			next;
		}
		else break;
	}
	set .@total_cost,.@sell * .@item_cost;
	set .@total_weight,.@sell * .@item_weight;
	if (Zeny < .@total_cost) {
		mes "[Madeleine Chu]";
		mes "Oh, je suis désolé, mais toi";
		mes "je ne peux pas me permettre autant";
		mes getitemname(.@item_id);
		mes "Veuillez vérifier votre zeny";
		mes "avant d'acheter mes kits~";
		close;
	}
	if (!checkweight(.@item_id,.@sell)) {
		mes "[Madeleine Chu]";
		mes "Je suis désolé, mais tu ne le fais pas";
		mes "ayez suffisamment de place dans votre";
		mes "Inventaire pour autant";
		mes getitemname(.@item_id) +"...";
		close;
	}
	set Zeny, Zeny-.@total_cost;
	getitem .@item_id,.@sell;
	mes "[Madeleine Chu]";
	mes "Te voilà ~";
	mes "Bonne chance avec";
	mes "votre formation culinaire !";
	return;
}
