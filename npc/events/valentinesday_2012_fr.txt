mosk_in,21,246,5	script	Pinkamenia	71,{
	set .@n$,"^0000FF[Pinkamenia]^000000";
	if (#V_QUE12==3) {
		set .@LFlowerID,7864;
		setarray .@SweetsID[0],538,558,539,573,559,560,12062,596,597,12414,12319;
		setarray .@SweetsCost[0],1,2,5,10,10,10,15,15,15,20,20;
		mes .@n$;
		mes "Hello "+strcharinfo(0)+", do";
		mes "Vous voulez acheter des objets?";
		mes "^FF0000Vous^ devez me donner de l'Amour.";
		mes "Des fleurs pour eux, bien sûr!";
		next;
		if(select("Yes, sure!","No, never mind.") == 2) close;
		next;
		mes .@n$;
		mes "Sélectionnez un objet.";
		mes "Le nombre de Fleurs d'Amour dont vous aurez besoin est entre parenthèses.";
		for(set .@i,0; .@i < getarraysize(.@SweetsID); set .@i,.@i+1)
			set .@menu$, .@menu$+"^00AA00["+.@SweetsCost[.@i]+"]^000000 "+getitemname(.@SweetsID[.@i])+":";
		next;
		set .@choice, select(.@menu$)-1;
		mes .@n$;
		if (countitem(.@LFlowerID) < .@SweetsCost[.@choice]) {
			mes "^FF0000Vous^ en aurez besoin davantage si vous voulez celui-là !";
			close;
		}
		mes "Are you sure you want to trade ^00aa00"+.@SweetsCost[.@choice]+"x Love Flower ^000000 for ^0055FF"+getitemname(.@SweetsID[.@choice])+"^000000?";
		if(select("No, I've changed my mind.:Yes, trade!") == 1) close;
		set .@new, .@LFlowerID - .@SweetsCost[.@choice];
		if (.@new < 0) set .@new, 0;
		delitem .@LFlowerID, .@SweetsCost[.@choice];
		getitem .@SweetsID[.@choice],1;
		mes "Amusez-vous bien avec votre objet!";
		close;
	} else if (#V_QUE12==2) {
		mes .@n$;
		if (countitem(7864) < 15) {
			mes "^FF0000Vous^ devez me rapporter 15 Fleurs d'Amour!";
			close;
		}
		mes "Merci beaucoup pour";
		mes "avoir ramené nos Fleurs!";
		delitem 7864,15;
		next;
		mes .@n$;
		mes "Prenez cela comme un petit";
		mes "'merci'.";
		getexp 500000,400000;
		getitem 617,1;
		getitem 12319,2;
		next;
		mes .@n$;
		mes "Si vous obtenez plus de Fleurs d'";
		mes "Amour, vous pouvez les échanger ici contre des douceurs.";
		mes "them for some sweets here.";
		mes "À bientôt!";
		set #V_QUE12,3;
		close;
	} else if (#V_QUE12==1) {
		mes .@n$;
		mes "[NPC Name] Vous prie d'apporter un Chapeau Gâteau +8";
		mes "à l'[Baker Extraordinaire] se tenant à côté de moi!";
		mes "C'est la Fête des Amoureux et nous";
		close;
	} else if (BaseLevel >= 45) {
		mes .@n$;
		mes "Hello "+strcharinfo(0)+", it's";
		mes "Valentine's Day and we";
		mes "^FF0000J'adore livrer des douceurs";
		mes "^FF0000chocolats.";
		next;
		mes .@n$;
		mes "Notre problème est que nous";
		mes "avons besoin d'une personne";
		mes "forte comme vous pour nous aider,";
		mes "mais en premier lieu, il va falloir";
		mes "apporter un Chapeau Gâteau +8 à l'Extraordinaire Pâtissier, qui";
		mes "se trouve juste à côté de moi!";
		mes "Échangez vos Fleurs d’Amour avec Pinkamenia!";
		set #V_QUE12,1;
		close;
	}
	mes .@n$;
	mes "Hello "+strcharinfo(0)+"!";
	close;

OnInit:
	// Unknown item drop rate (currently set at 5%)
	setitemscript 5105,"{ bonus bDex,1; bonus bMaxSP,80; bonus3 bAddMonsterDropItem,7864,RC_DemiHuman,500; }",0;
	end;
}

mosk_in,21,244,3	script	Baker Extraordinaire	410,{
	set .@n$,"^0000FF[Baker Extraordinaire]^000000";
	if(#V_QUE12==3) {
		mes .@n$;
		mes "Apportez-moi 15 Fleurs d’Amour, Pinkamenia vous l'a dit!";
		close;
	} else if(#V_QUE12==2) {
		mes .@n$;
		mes "Ah, donc Pinkamenia t'a demandé";
		close;
	} else if(#V_QUE12==1) { 
		mes .@n$;
		if (getequipid(EQI_HEAD_TOP) == 5024 && getequiprefinerycnt(EQI_HEAD_TOP) >= 8) {
			mes "de me rapporter le Chapeau Gâteau +8. Maintenant, je vais";
			mes "t'échanger ton Chapeau Gâteau +8 contre un autre.";
			mes "Maintenant, si tu portes ce nouveau Chapeau Gâteau, il y a une chance que les";
			mes "Monstres Demi-Humains";
			mes "Hat for another Cake Hat.";
			next;
			mes .@n$;
			delitem 5024,1;
			getitem 5105,1;
			equip 5105;
			mes "Now, if you wear the";
			mes "new Cake Hat, there";
			mes "is a chance that";
			mes "Demi-Human Monsters will";
			mes "lancez une Fleur d'Amour !";
			next;
			mes .@n$;
			mes "Les monstres nous ont volé nos Fleurs,";
			mes "dont nous avons besoin pour créer";
			mes "nos chocolats. Maintenant ramenez-moi";
			mes "15 Fleur d'Amour et apportez-les à Pinkamenia !";
			mes "Désolé, mais où est votre";
			set #V_QUE12,2;
			close;
		} else {
			mes "+8 Cake Hat ? Apportez-le-moi et n'oubliez pas";
			mes "de l'équiper !";
			mes "it to me, and remember";
			mes "to have it equipped!";
			close;
		}
	} else {
		mes .@n$;
		mes "Hello, "+strcharinfo(0)+"!";
		close;
	}
}
