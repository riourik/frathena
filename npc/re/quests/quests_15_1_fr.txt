function checkstatusquest {
	switch( getarg(0) ) {
	case 4297:
		if (lhz_curse >= 31)// Cursed Spirit Quest
			return true;
		break;
	case 4298:
		if (lhz_boss >= 43)// President Quest
			return true;
		break;
	case 4299:
		if (yuno_hist >= 10)// Juperos Ruins History Quest
			return true;
		break;
	case 4300:
		if (metto_q >= 9)// Metto Quest
			return true;
		break;
	case 4301:
		if (hg_bio >= 9)// Biological Weapon Quest
			return true;
		break;
	case 4302:
		if (hg_odin >= 60)// Odin Temple Excavation Quest
			return true;
		break;
	}
	return false;
}
}

				select("Task initialization");

			}
			end;
		case 3:
			mes "[Petit Assistant]";
			mes "Ajustez la section d'activité soudaine, pas besoin de confirmer la mise en œuvre, veuillez donc l'utiliser après évaluation.";

				mes "Fin de la tâche soudaine.";
