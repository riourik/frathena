					break;

					next;
					break;

					callsub L_Mark, F_Navi("Weapon & Armor Shop","morocc,253,56","^7F3300");

					next;
					break;
				case 8:
					set .@loop,0;
					break;
				}
			}
			break;
		case 2:
			set .@loop,1;
			while (.@loop) {
				switch(select("[ Kafra Employee ]:[ Poison Herb Salesman ]:[ Dye Maker ]:[ Seiyablem ]:Previous Menu")) {
				case 1:
					callsub L_Mark,"^0000FFKafra Employees^000000";

					break;

					mes "[Guide de Morocc]";

					mes "actuellement en séjour";
					mes "au premier étage de l'auberge.";
					mes "Laissez-moi marquer son emplacement";
					mes "[Ricael]";
					mes "Souhaitez-vous vérifier d'autres emplacements ?";

					break;
				case 4:
					mes "[Guide de Morocc]";
					mes F_Navi("Seiyablem","morocc,51,41","^800080");
					mes "est un technicien";
					mes "qui peut créer des ^0000FFslots^000000 sur les armes et les armures,";

					mes "Souhaitez-vous vérifier d'autres emplacements ?";
					viewpoint 1,51,41,11,0xDA70D6;
					next;
					break;
				case 5:
					set .@loop,0;

			viewpoint 2,1,1,10,0xFFFF00;
			viewpoint 2,1,1,11,0xFFFF00;
			next;
			break;
		case 4:
			mes "[Guide de Morocc]";
			mes "Soyez prudent lors de vos voyages.";
			close2;
			cutin "moc_soldier",255;
			end;
		}
	}
	end;

	mes "[Guide de Morocc]";
	mes "Laissez-moi marquer l'emplacement de";
	mes getarg(0);

}
