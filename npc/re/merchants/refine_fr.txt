//============================================================
// +11 and above Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//=
//= On official servers, if an item is unsuccessfully refined
//= it will break at a 20% rate and downgrade at an 80% rate.
//============================================================
function	script	refinenew	{
	.@npc_name$ = getarg(0);
	disable_items;
	mes "["+ .@npc_name$ +"]";
	mes "Je suis le meilleur armurier qui soit !";
	mes "Je ne raffine pas les objets normaux et ennuyeux.";
	mes "Je ne raffine que les objets de niveau 10 ou supérieur.";
	next;
	mes "["+ .@npc_name$ +"]";
	mes "De toute façon, vous pouvez utiliser mes services si votre objet est de niveau 10 ou supérieur.";
	mes "Que voulez-vous que je rafine ?";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<=10; ++.@i) {
		if (getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "Je ne pense pas pouvoir affiner les objets que vous avez...";
		close;
	}
	.@part = .@indices[ select(.@menu$) ];

	if (!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "Tu ne portes pas";
		mes "quoi que ce soit là-bas qui";
		mes "que je puisse affiner.";
		emotion ET_FRET;
		close;
	}
	if (!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "je ne pense pas pouvoir";
		mes "affiner cet objet du tout...";
		close;
	}
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	if (.@refinerycnt < 10) {
		mes "["+ .@npc_name$ +"]";
		mes "J'ai dit que je ne travaille pas avec des objets inférieurs au niveau 10.";
		close;
	}
	if (.@refinerycnt >= 20) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "Je ne peux plus affiner";
		mes "cela. C'est aussi";
		mes "refiné que cela peut l'être !";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@safe = 10;
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );
	
	if( .@itemtype == IT_WEAPON ){
		.@article$ = "a";
		.@type$ = "weapon";
	}else if( .@itemtype == IT_ARMOR ){
		.@article$ = "an";
		.@type$ = "armor";
	}else{
		// TODO:
		close;
	}

	mes "["+ .@npc_name$ +"]";
	mes "Hmm " + .@article$ + " " + .@type$ + ", is that ok?";
	mes "If you want to refine this " + .@type$ + ",";
	mes "I will need 1 ^003366" + getitemname(.@material) + "^000000 and " + callfunc("F_InsertComma",.@price) + " zeny.";
	mes "Êtes-vous sûr de vouloir continuer ?";
	next;
	if (select("Yes:No") == 2) {
		mes "["+ .@npc_name$ +"]";
		mes "Hm... si cela vous dérange... peu importe...";
		close;
	}
	if (getarg(1) != 1) {
		if (getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "This "+.@type$+" already has been refined several times.";
			mes "Il pourrait être détruit si j'essaie à nouveau.";
			mes "Il ne cassera pas sûrement, mais il y a une petite chance.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "You could be ^FF0000lowering the upgrade level^000000 of the "+.@type$+",";
			mes "ou s'il casse, vous perdrez ^FF0000toutes les cartes^000000 ou propriétés spéciales ajoutées à celui-ci.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "Voulez-vous toujours que je le raffine ?";
			mes "Je pense que je vous ai donné assez d'avertissements.";
			next;
			if(select("Yes.:No.") == 2) {
				mes "["+ .@npc_name$ +"]";
				mes "Eh bien, pas de défi n'est pas la seule voie...";
				mes "Pas de risque... cela pourrait être sage.";
				close;
			}
		}
		if (countitem(.@material) < 1 || Zeny < .@price) {
			mes "["+ .@npc_name$ +"]";
			mes "Hm. You don't seem to have enough money or "+getitemname(.@material)+".";
			mes "Veuillez revenir lorsque vous les aurez.";
			close;
		}
		Zeny = Zeny - .@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "Attends une seconde...";
			mes "Croyez-vous que je suis stupide ?!";
			mes "Vous avez échangé l'objet quand je ne regardais pas ! Fichez le camp d'ici !";
			close;
		}

		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "Clang! Clang! Clang! Clang!";
			successrefitem .@part;
			next;
			emotion ET_BEST;
			mes "["+ .@npc_name$ +"]";
			mes "Bien ! Succès !!!";
			mes "Je suis le meilleur armurier.";
			close;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "Clang! Clang! Clang! Clang!";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "Ahhh !!!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "Oh mon dieu !";
				mes "Le niveau de transformation a baissé...";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "Clang! Clang! Clang!";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "Hmmm !";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "Oh my ! J'ai raté la transformation...";
				mes "Je ne voulais pas !";
			}
			mes "J'aurais pu faire une erreur, même si je suis le meilleur armurier du monde.";
			mes "Ce n'était simplement pas destiné à être ainsi.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "La prochaine fois, je ferai un meilleur travail ! Ne vous inquiétez pas !";
			close;
		}
	}
// New +11 and above Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "^FF0000Je peux raffiner cet objet jusquà la limite sécuritaire ou le nombre de fois que vous souhaitez. C'est votre choix.";
		next;
		.@menu2 = select("To the safe limit, please.","I'll decide how many times.","I've changed my mind...");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "^FF0000Combien de fois voudriez-vous que je raffine votre objet ?";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 20) {
			mes "["+ .@npc_name$ +"]";
			mes "^FF0000Je ne peux pas raffiner cet objet autant de fois.";
			close;
		}
		if (.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
			mes "This will try to refine the equipment " + .@refinecheck + " times past the safe limit. Your equipment may be destroyed... is that ok?";
			next;
			if(select("Yes...","No...") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "^FF0000Vous l'avez dit... Comme vous voulez.";
				close;
			}
		}
		break;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "^FF0000Vous l'avez dit... Comme vous voulez.";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "That will cost you " + .@refinecnt + " " + getitemname(.@material) + " and " + .@fullprice + " Zeny. Is that ok?";
	next;
	if(select("Yes:No...") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "^FF0000Vous l'avez dit... Comme vous voulez.";
		close;
	}
	if (countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
		mes "Hm. You don't seem to have enough money or "+getitemname(.@material)+".";
		mes "Veuillez revenir lorsque vous les aurez.";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "^FF0000Regardez ici... Vous n'avez aucun objet sur...";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt))
			close;

		if (getequipid(.@part) != .@refineitemid || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "^FF0000Clang... Non, mais avez-vous imaginé que je pourrais être aussi stupide?!";
			mes "Vous l'avez changé...";
			mes "Sortez d'ici avant que je ne vous assomme avec mon Marteau!!";
			close;
		}
		if (getequippercentrefinery(.@part) > rand(100)) {
			mes "Clang! Clang! Clang! Clang!";
			successrefitem .@part;
			.@refinecnt = .@refinecnt - 1;
			next;
		} else {
			if (rand(100) < 80) {
				mes "["+ .@npc_name$ +"]";
				mes "Clang! Clang! Clang! Clang!";
				downrefitem .@part,3; // Failed refine attempts decrease the item's refine level by 3
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "Ahhh !!!";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "Oh mon dieu !";
				mes "Le niveau de transformation a baissé...";
			} else {
				mes "["+ .@npc_name$ +"]";
				mes "Clang! Clang! Clang!";
				failedrefitem .@part;
				next;
				emotion (!rand(5))?ET_MONEY:ET_HUK;
				mes "["+ .@npc_name$ +"]";
				mes "Hmmm !";
				next;
				mes "["+ .@npc_name$ +"]";
				mes "Oh my ! J'ai raté la transformation...";
				mes "Je ne voulais pas !";
			}
			mes "J'aurais pu faire une erreur, même si je suis le meilleur armurier du monde.";
			mes "Ce n'était simplement pas destiné à être ainsi.";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "La prochaine fois, je ferai un meilleur travail ! Ne vous inquiétez pas !";
			close;
		}
		.@refinerycnt = getequiprefinerycnt(.@part);
	}
	emotion ET_BEST;
	mes "["+ .@npc_name$ +"]";
	mes "Bien ! Succès !!!";
	mes "Je suis le meilleur forgeron.";
	close;
}

	delitem .@material,1;
	Zeny = Zeny-.@price;

	// anti-hack

		mes "[Clink]";
		emotion ET_FRET;
		mes "Attends une seconde...";
		mes "Croyez-vous que je suis stupide ?!";
