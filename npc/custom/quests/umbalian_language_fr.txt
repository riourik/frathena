yuno,157,366,2	script	Wise Man Fabius	743,{
	mes "[Fabius]";
	if(event_umbala >= 3) set umb_lang,2; //You already learned Umbalian language from Umbalian Chief. So this Quest is disabled
	if(umb_lang<2) goto L_QUEST;
	mes "Vous parlez umbalien couramment ! Vous n'oublierez jamais la langue des Umbaliens !";
	close;
L_QUEST:
	if((countitem(1006) < 1) || (countitem(7118) < 1) || (countitem(7117) < 1)) goto L_DONE;//Items: Old Magic Book, Torn Scroll, Torn Spell Book,
	delitem 1006,1;	//Items: Old Magic Book,
	delitem 7118,1;	//Items: Torn Scroll,
	delitem 7117,1;//Items: Torn Spell Book,
	emotion ET_THANKS;
	mes "Ah, vous avez bien fait de récupérer mes références. Ces artefacts contiennent des informations sur les ruines du monde entier, de l'enfer enflammé de la Magma Dungeon, à la forêt étrange et inquiétante de Yggdrasil.";
	next;
	mes "[Fabius]";
	mes "Mais assez parler, vous êtes impatient d'avoir votre récompense n'est-ce pas ? Eh bien ce ne semble pas grand chose, mais ma récompense pour vous est le savoir...";
	next;
	mes "[Fabius]";
	mes "En fait, c'est le savoir sur les Umbaliens. Ils sont connectés à ce monde de Midgard d'une manière spéciale, nous ne pouvons pas encore le confirmer. Il y a une théorie selon laquelle ils pourraient être comme les elfes et protéger l'arbre ancien de Yggdrasil.";
	next;
	mes "[Fabius]";
	mes "Et moi, un grand sage de Juno, vais vous enseigner leur langue !";
	next;
	mes "^0000FF J'ai appris la langue des Umbaliens ! ^000000";
	set umb_lang,2;
	//clear subquests
	set umb_marie,0;
	set umb_mason,0;
	set umb_mojo,0;
	//make you understand Umbalian people
	//it's an alternative way to learn Umbalian Language
	if(event_umbala < 3) set event_umbala,3;
	close;
L_DONE:
	set umb_lang,1; //got Quest
	mes "Argh ! Mes trois élèves ont emporté mes références avec eux ! Hey toi ! Pouvez-vous les récupérer pour moi ? Je vous récompenserai généreusement.";
	next;
	mes "[Fabius]";
	mes "D'après ce que je sais, ils sont dans des villes différentes et se fabriquent quelque chose d'eux-mêmes.";
	if(umb_marie==0 && umb_mason==0 && umb_mojo==0) goto L_NOT_MET;
	if(umb_marie==1) mes "Oh... Vous avez trouvé Marie à Amatsu.";
	if(umb_mojo==1) mes "^FF0000Vous avez vu Mojo à Aldebaran ?";
	if(umb_mason==1) mes "^FF0000Alors... [Mason] est maintenant à Prontera... Eh bien...";
	if(umb_marie==0) mes "^FF0000Je me demande où est notre petite Marie de la mer...";
	if(umb_mojo==0) mes "^FF0000Mojo devrait être quelque part près de l'eau...";
	if(umb_mason==0) mes "^FF0000Et [Mason] alors, il est allé où...";
	mes "^FF0000Oh oui, dites-leur que je vous ai envoyé";
	close;
L_NOT_MET:
	mes "^FF0000Oh oui, dites-leur que je vous ai envoyé, leurs noms sont Marie, Mason et Mojo.";
	close;
}
