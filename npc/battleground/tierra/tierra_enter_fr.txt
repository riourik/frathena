// First Tierra Gorge Officers - Guillaume
//============================================================
bat_room,124,178,5	script	Tierra Gorge Officer#01a	418,{
	if (getbattleflag("feature.bgqueue"))
		end;
	if (checkweight(1201,1) == 0) {
		mes "- Attends une minute !! -";
		mes "- Actuellement, vous portez -";
		mes "- trop d'objets avec vous. -";
		mes "- Veuillez réessayer -";
		mes "- après avoir perdu du poids. -";
		close;
	}
	mes "[Officier de l'armée Guillaume]";
	mes "Montrons la puissance de l'Armée Guillaume à ces Croix puantes !";
	next;
	switch(select("I want to join your army!:End Conversation")) {
	case 1:
		if ((BaseJob == Job_Novice) || (BaseJob == Job_Super_Novice)) {
			mes "[Officier de l'armée Guillaume]";
			mes "Je suis très heureux que vous vouliez nous rejoindre et combattre l'Armée Croix, mais je suis désolé : je ne peux pas envoyer une recrue comme vous mourir sur le cruel champ de bataille.";
		}
		else {
			if (BaseLevel < 80) {
				mes "[Officier de l'armée Guillaume]";
				mes "Je suis très heureux que vous vouliez nous rejoindre et combattre l'Armée Croix, mais je suis désolé : je ne peux pas envoyer une recrue comme vous mourir sur le cruel champ de bataille.";
			}
			else {
				set .@chk_urtime,checkquest(2069,PLAYTIME);
				if ((.@chk_urtime == 0)|| (.@chk_urtime == 1)) {
					mes "[Officier de l'armée Guillaume]";
					mes "Vous semblez revenir tout juste du champ de bataille.";
					mes "Il est trop tôt pour que tu rentres. Allez vous reposer et laissez-nous les ennemis !";
				}
				else {
					if (.@chk_urtime == 2)
						erasequest 2069;
					set .@mapcount,getmapusers("bat_a01");
					if (.@mapcount > 0) {
						mes "[Officier de l'armée Guillaume]";
						mes "J'ai reçu un rapport m'informant qu'un corps d'élite a déjà été envoyé sur le champ de bataille pour combattre les Croix.";
						mes "Pourquoi n'attends-tu pas un peu ? Je vous suggère d'affûter vos armes et de préparer vos provisions d'ici là.";
					}
					else {
						if ($@TierraBG1) {
							mes "[Officier de l'armée Guillaume]";
							mes "Un corps d’élite est déjà prêt à être envoyé sur le champ de bataille.";
							mes "Pourquoi n'attends-tu pas un peu ? Je vous suggère d'affûter vos armes et de préparer vos provisions d'ici là.";
						}
						else {
							mes "[Officier de l'armée Guillaume]";
							mes "Vous semblez définitivement prêt pour le combat !";
							mes "Allez montrer aux Croix ce que signifie vraiment la peur !";
							mes "Aujourd'hui, notre cri de victoire résonnera sur tout le champ de bataille !";
							close2;
							warp "bat_room",57,223;
							end;
						}
					}
				}
			}
		}
		break;
	case 2:
		mes "[Officier de l'armée Guillaume]";
		mes "Aujourd'hui, nous serons victorieux !";
		break;
	}
	close;
}

bat_room,127,178,5	script	Guillaume Knight#1	417,{
	mes "[Guillaume Chevalier]";
	mes "Tierra Gorge se compose de deux versants abrupts placés verticalement et dispose de dépôts de rations pour les armées Guillaume et Croix aux directions 11 et 7 heures.";
	next;
	mes "[Guillaume Chevalier]";
	mes "La bataille commence sur le navire de votre armée, et le but est d'avancer et de détruire le dépôt de rations de votre ennemi plus rapidement qu'il ne peut détruire le vôtre.";
	next;
	mes "[Guillaume Chevalier]";
	mes "Le guérisseur du champ de bataille ressuscitera les soldats toutes les 25 secondes afin qu'ils puissent rejoindre la bataille.";
	mes "L'armée qui capture le drapeau neutre au centre du champ de bataille sera récompensée par des points de régénération supplémentaires, ce qui signifie que ses soldats ressusciteront davantage que l'autre camp, ce qui lui donnera un avantage.";
	next;
	mes "[Guillaume Chevalier]";
	mes "Ces deux dépôts de rations sont reliés entre eux par des chemins courts et étroits, mais les portes sont bloquées par des barricades.";
	mes "Réfléchissez et avancez aussi vite que possible. La victoire de votre armée dépend de votre contribution.";
	next;
	mes "[Guillaume Chevalier]";
	if (getbattleflag("feature.bgqueue"))
		mes "Si vous êtes prêt, commencez par cliquer sur le bouton ^0000FFbattle de l'interface^000000.";
	else {
		mes "N'oubliez pas : lorsque vous rejoignez une bataille, vous recevez un jeton qui indique la durée définie pendant laquelle vous ne pouvez pas participer au même type de bataille.";
		mes "Vous pouvez vérifier la durée en appuyant sur les touches Alt+U.";
	}
	next;
	mes "[Guillaume Chevalier]";
	if (getbattleflag("feature.bgqueue"))
		mes "Veuillez garder à l'esprit qu'une fois que vous annulez, ^0000FF vous ne pourrez pas présenter une nouvelle demande pendant 1 minute^000000 !";
	else
		mes "Êtes-vous prêt à vous battre ? Alors postulez auprès du recruteur à côté de moi !";
	close;
}
