//= Write you name on your rare equipment or weapon ^_-
//===== Additional Comments: =================================

//= 1.1a Added 'disable_items' command. [Euphy]

prt_in,24,61,7	script	Perchik	47,{
	
	setarray .@Item[0],644,3;	// Item requirements: <ID>,<Count>{,...} (0 to disable)
	setarray .@Cost[0],0,5000;	// Zeny requirements: <base price>,<price per refine>

	disable_items;
	mes "[Pertchik]";
	mes "Je peux ^0055FFsigner votre nom^000000 sur presque tous les objets rares que vous détenez.";
	next;
	if(select("Tell me more...:Sign my items, please!") == 1) {
		mes "[Pertchik]";
		mes "Je peux mettre votre nom sur n’importe quel équipement ou arme sans emplacement.";
		emotion ET_AHA;
		next;
		mes "[Pertchik]";
		if (getarraysize(.@Item) || getarraysize(.@Cost)) {
			mes "Pour mon travail j'accepte :";
			if (getarraysize(.@Item)) for(set .@i,0; .@i<getarraysize(.@Item); set .@i,.@i+2)
				mes " ~ "+.@Item[.@i+1]+"x "+getitemname(.@Item[.@i]);
			if (.@Cost[0]) mes " ~ "+callfunc("F_InsertComma",.@Cost[0])+" Zeny";
			if (.@Cost[1]) mes " ~ "+callfunc("F_InsertComma",.@Cost[1])+" Zeny per refine";
		}
		else mes "Je travaille gratuitement, mais...";
		next;
		emotion ET_CRY;
		mes "[Pertchik]";
		mes "Hélas, j'ai 12 enfants affamés";
		mes "et une femme très en colère.";
		next;
		mes "[Pertchik]";
		mes "Ou c'était 12 enfants en colère";
		mes "et une femme très affamée...";
		emotion ET_SCRATCH;
		close;
	}
	mes "[Pertchik]";
	mes "Montre-moi tes objets à signer...";
	next;
	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for (set .@i,1; .@i<=10; set .@i,.@i+1) {
		if (getequipisequiped(.@indices[.@i]))
			set .@menu$, .@menu$ + F_getpositionname(.@indices[.@i]) +" [^0055FF"+getequipname(.@indices[.@i])+"^000000]";
		set .@menu$, .@menu$+":";
	}
	set .@part, .@indices[ select(.@menu$) ];
	set .@id, getequipid(.@part);
	set .@ref, getequiprefinerycnt(.@part);
	mes "[Pertchik]";
	if (!getequipisequiped(.@part)) {
		mes "Rien n'y est équipé !";
		emotion ET_KEK;
		close;
	}
	for(set .@i,0; .@i<4; set .@i,.@i+1)
		set .@slot[.@i], getequipcardid(.@part,.@i);
	if (.@slot[0]==255 || .@slot[0]==254 || .@slot[0]<0) {
		mes "Hélas, cet article est déjà signé.";
		mes "Je ne toucherais jamais à l'œuvre d'un maître.";
		emotion ET_SCRATCH;
		close;
	}
	if (.@slot[0]>4000 && .@slot[0]<5000) {
		mes "Une carte ? Ici?!";
		mes "Comme je l'ai déjà dit, je ne signe pas d'objets avec des cartes.";
		emotion ET_SCRATCH;
		close;
	}
	if (getiteminfo(.@id, ITEMINFO_SLOT)) {
		mes "Désolé, je ne signe pas les objets à fente.";
		emotion ET_SORRY;
		close;
	}
	getinventorylist;
	for(set .@i,0; .@i<@inventorylist_count; set .@i,.@i+1)
		if (@inventorylist_expire[.@i] != 0) {
			mes "Désolé, je ne signe pas les articles de location !";
			emotion ET_SCRATCH;
			close;
		}
	set .@price, .@Cost[0]+(.@Cost[1]*.@ref);
	if (getarraysize(.@Item) || .@price) {
		mes "J'aurai besoin de :";
		if (getarraysize(.@Item)) for(set .@i,0; .@i<getarraysize(.@Item); set .@i,.@i+2) {
			mes " ~ "+.@Item[.@i+1]+"x "+getitemname(.@Item[.@i]);
			if (countitem(.@Item[.@i]) < .@Item[.@i+1]) set .@nr,1;
		}
		if (.@price) mes " ~ "+.@price+" Zeny";
	}
	mes "Shall I sign your ^0055FF"+getitemname(.@id)+"^000000?";
	next;
	if(select("Ok!:Leave")==2) {
		mes "[Pertchik]";
		mes "À bientôt...";
		emotion ET_SLEEPY;
		close;
	}
	mes "[Pertchik]";
	if (Zeny < .@price || .@nr) {
		mes "Je ne travaille pas pour « merci ».";
		emotion ET_SORRY;
		close;
	}
	if (!countitem2(.@id,1,.@ref,0,.@slot[0],.@slot[1],.@slot[2],.@slot[3])) {
		mes "Where is "+getitemname(@id)+"...?";
		npctalk "Perchik : Tu es un tricheur snoozy !";
		logmes "Hack: Tried to sign an item not having it: "+getitemname(@id);
		emotion ET_KEK;
		close;
	}
	if (.@price) set Zeny, Zeny-.@price;
	if (getarraysize(.@Item))  for(set .@i,0; .@i<getarraysize(.@Item); set .@i,.@i+2)
		delitem .@Item[.@i], .@Item[.@i+1];
	delitem2 .@id,1,1,.@ref,0,.@slot[0],.@slot[1],.@slot[2],.@slot[3];
	mes "Fait!";
	emotion ET_OK;
	getitem2 .@id,1,1,.@ref,0,254,0,getcharid(0)&0xffff,(getcharid(0)>>16)&0xffff;
	equip .@id;
	close;
}
