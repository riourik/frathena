job3_rune01,88,62,5	script	Dragon Breeder	105,{
	mes "[Dragon Breeder]";
	if ((eaclass()&EAJ_THIRDMASK)==EAJ_RUNE_KNIGHT) {
		mes "Bienvenue. Souhaitez-vous louer un Dragon ?";
		next;
		switch(select("Yes:No")) {
		case 1:
			//if (!getskilllv("RK_DRAGONTRAINING")) {
			if (!getskilllv("KN_RIDING")) {
				mes "[Dragon Breeder]";
				mes "Apprenez d'abord à monter sur un Dragon.";
				close;
			}
			else if (checkdragon()) {
				mes "[Dragon Breeder]";
				mes "Vous avez déjà un Dragon.";
				close;
			}
			else if (ismounting()) {
				mes "[Dragon Breeder]";
				mes "Veuillez enlever votre monture cash.";
				close;
			}
			setdragon;
			close;
		case 2:
			mes "[Dragon Breeder]";
			mes "Je vois. Alors passez une excellente journée.";
			close;
		}
	}
	mes "Que faites-vous ici?";
	mes "^FF0000Seuls les Chevaliers de la Rune peuvent louer un Dragon.^FF0000";
	close;
}

// Mado Gear Renter :: madogear
//============================================================
-	script	::mgm	-1,{
	mes "[Mado Gear Armorer]";
	if ((eaclass()&EAJ_THIRDMASK)==EAJ_MECHANIC) {
		mes "Bienvenue, Mécanicien.";
		mes "Souhaitez-vous louer un chariot ou";
		mes "prendre une Mado Gear ?";
		next;
		switch(select("Rent a Pushcart:Ride a Mado Gear:Buy Emergency Mado Gear:Upgrade Cooling Device:Cancel")) {
		case 1:
			if (checkcart()) {
				mes "[Mado Gear Armorer]";
				mes "Désolé, mais vous avez déjà";
				mes "un chariot.";
				close;
			}
			setcart;
			mes "[Mado Gear Armorer]";
			mes "Et voilà !";
			close;
		case 2:
			if (checkmadogear()) {
				mes "[Mado Gear Armorer]";
				mes "Désolé, mais vous êtes déjà";
				mes "à bord d'une Mado Gear.";
				close;
			}
			else if (!getskilllv("NC_MADOLICENCE")) {
				mes "[Mado Gear Armorer]";
				mes "Apprenez d'abord la compétence pour obtenir le permis de conduite Mado Gear.";
				close;
			}
			else if(ismounting()) {
				mes "[Mado Gear Armorer]";
				mes "Veuillez enlever votre monture cash.";
				close;
			}
			setmadogear;
			mes "[Mado Gear Armorer]";
			mes "Amusez-vous bien et revenez nous voir !";
			close;
		case 3:
			mes "[Mado Gear Armorer]";
			mes "Le Mado Gear d'urgence est vraiment utile pour les situations d'urgence et il se vend à 1 000 000 de Zénies.";
			next;
			if (select("Purchase:Cancel") == 2) {
				mes "[Mado Gear Armorer]";
				mes "Je vois. N'hésitez pas à me demander.";
				mes "si vous changez d'avis.";
				close;
			}
			if (countitem(23277) > 0) {
				mes "[Mado Gear Armorer]";
				mes "Désolé, mais vous avez déjà un Mado Gear d'urgence.";
				close;
			}
			if (Zeny < 1000000) {
				mes "[Mado Gear Armorer]";
				mes "Désolé, mais vous n'avez pas assez de Zénies pour acheter le Mado Gear d'urgence.";
				close;
			}
			Zeny -= 1000000;
			getitem 23277,1; //Mado_Box
			mes "[Mado Gear Armorer]";
			mes "Et voilà !";
			close;
		case 4:
			mes "[Mado Gear Armorer]";
			mes "Quel dispositif souhaitez-vous améliorer ?";
			next;
			if (select("Cooling Device:High Quality Cooler") == 1) {
				mes "[Mado Gear Armorer]";
				mes "L'amélioration du Dispositif de Refroidissement vers le Refroidisseur Haute Qualité nécessite 1 Dispositif de Refroidissement et 2 000 000 de Zénies.";
				next;
				.@itemid = 2804;
				.@cost = 2000000;
			} else {
				mes "[Mado Gear Armorer]";
				mes "L'amélioration du Refroidisseur Haute Qualité vers le Refroidisseur Spécial nécessite 1 Refroidisseur Haute Qualité et 4 000 000 de Zénies.";
				next;
				.@itemid = 2809;
				.@cost = 4000000;
			}
			if (select("Upgrade:Cancel") == 2) {
				mes "[Mado Gear Armorer]";
				mes "Je vois. N'hésitez pas à me demander.";
				mes "si vous changez d'avis.";
				close;
			}
			if (!countitem(.@itemid)) {
				mes "[Mado Gear Armorer]";
				mes "I'm sorry, but you don't have the " + getitemname(.@itemid) + ".";
				close;
			}
			if (Zeny < .@cost) {
				mes "[Mado Gear Armorer]";
				mes "Désolé, mais vous n'avez pas assez de Zénies pour améliorer le dispositif.";
				close;
			}
			Zeny -= .@cost;
			delitem .@itemid,1;
			getitem (.@itemid == 2804 ? 2809 : 2810),1; //High_Quality_Cooler,Special_Cooler
			mes "[Mado Gear Armorer]";
			mes "Here you are! Your very own " + getitemname(.@itemid) + ".";
			close;
		case 5:
			close;
		}
	}
	mes "Comment puis-je vous aider ?";
	mes "Les engrenages de Mado ne sont disponibles que pour les Mécaniciens.";
	close;
}
