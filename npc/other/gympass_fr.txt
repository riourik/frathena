payon,173,141,4	script	Ripped Cabus#GymPass	899,{
	mes "[Déchiré]";
	mes "Salut. Les gens ne le sont pas";
	mes "aussi actifs physiquement qu'eux";
	mes "l'était. Même si tu te bats";
	mes "pour gagner ma vie, ton corps pourrait";
	mes "être faible et flasque chez certains";
	mes "zones. Tu vois ce que je veux dire ?";
	next;
	mes "[Déchiré]";
	mes "Hé, entraîne-toi avec moi, et je pourrai";
	mes "garantie que vous pourrez";
	mes "pour soulever et transporter davantage de vos";
	mes "truc. Donne-moi juste ton";
	mes "^FF0000Gym Pass^000000 à chaque fois,";
	mes "et nous serons prêts à partir.";
	next;
	mes "[Déchiré]";
	mes "Mais ne vous énervez pas trop :";
	mes "peu importe la quantité d'entraînement";
	mes "Je t'accompagne, tu peux";
	mes "en faire trop. Vous avez déjà entendu parler de";
	mes "quelqu'un qui est devenu trop musclé ?";
	mes "C'est parce qu'ils sont morts. Voir?";
	next;
	mes "[Déchiré]";
	mes "Je dirais que ce serait sûr";
	mes "pour que tu t'entraînes sérieusement";
	mes "avec moi et augmente ton";
	mes "capacité de charge de l'article ^FF000010 fois ^000000.";
	mes "Alors... Êtes-vous prêt à transpirer ?";
	next;
	switch(select("Yes:No:Um, my workouts wore off.")) {
	case 1:
		if (gympassmemory < 10) {
			set .@add_carry,gympassmemory + 1;
			set .@remain_carry,10 - .@add_carry;
			if (countitem(7776) > 0) {
				mes "[Déchiré]";
				mes "Oh, génial, je te vois";
				mes "apporté votre Gym Pass.";
				mes "Très bien, fais juste ce que je fais,";
				mes "et essayez de ressentir la brûlure.";
				mes "Prêt? Faisons ça.";
				next;
				specialeffect2 EF_EARTHSPIKE;
				next;
				specialeffect2 EF_DEVIL;
				next;
				specialeffect2 EF_COIN;
				next;
				specialeffect2 EF_SIGHTRASHER;
				next;
				mes "[Déchiré]";
				mes "Là, tu devrais pouvoir";
				mes "pour emporter plus de choses avec vous.";
				mes "Voyons, nous pouvons augmenter";
				mes "capacité de charge de votre article";
				mes "^FF00000" + .@remain_carry + "^000000 more times if we continue";
				mes "s'entraîner ensemble comme ça.";
				delitem 7776,1; //Max_Weight_Up_Scroll
				set gympassmemory,.@add_carry;
				skill "ALL_INCCARRY",.@add_carry,SKILL_PERM_GRANT;
				close;
			}
			else {
				mes "[Déchiré]";
				mes "Mec, qu'est-ce que je t'ai dit ?";
				mes "Tu dois m'apporter ton";
				mes "^FF0000Gym Pass^000000 si vous voulez";
				mes "entraînez-vous et développez votre";
				mes "objet porteur de muscles.";
				close;
			}
		}
		else {
			mes "[Déchiré]";
			mes "Mec, je ne pense pas que nous puissions";
			mes "construisez votre objet en transportant";
			mes "muscles plus que ça.";
			mes "C'est trop dangereux pour toi";
			mes "corps si nous essayions même! Allez,";
			mes "Je vous ai parlé des limites.";
			close;
		}
	case 2:
		mes "[Déchiré]";
		mes "Oh, c'est dommage.";
		mes "Eh bien, reviens si tu";
		mes "changez d'avis. Dire";
		mes "tes amis à propos de moi :";
		mes "s'ils sont flasques, je vais les aider";
		mes "mettez-les en forme.";
		close;
	case 3:
		if (gympassmemory > 0) {
			mes "[Déchiré]";
			mes "Ce qui s'est passé?";
			mes "Vous laissez votre objet porter";
			mes "les muscles s'atrophient simplement ? Chanceux";
			mes "pour toi, il y a une telle chose";
			mes "comme mémoire musculaire. ça ne prendra pas";
			mes "à condition de les reconstruire...";
			next;
			specialeffect2 EF_EARTHSPIKE;
			next;
			specialeffect2 EF_DEVIL;
			next;
			specialeffect2 EF_COIN;
			next;
			specialeffect2 EF_SIGHTRASHER;
			next;
			mes "[Déchiré]";
			mes "Et ça ?";
			mes "Votre article transportant";
			mes "les muscles ont repoussé,";
			mes "comme ça! Essayez de ne pas";
			mes "Encore une mauviette, d'accord ?";
			skill "ALL_INCCARRY",gympassmemory,SKILL_PERM_GRANT;
			close;
		}
		else {
			mes "[Déchiré]";
			mes "Euh...";
			mes "Nous n'avons pas travaillé";
			mes "ensemble avant.";
			mes "J'en suis sûr.";
			close;
		}
	}
}
