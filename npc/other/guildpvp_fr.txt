pvp_y_room,77,85,4	script	Guild Battle Guide	105,{

	setarray .@map1X[1],7,9,16,8,20;
	setarray .@map2X[1],91,90,83,91,79;
	setarray .@mapY[1],49,49,50,49,50;

	mes "[Guide de bataille de guilde]";
	mes "Comment allez-vous?";
	mes "Je suis le guide des combats de guilde";
	mes "pour les nouvelles cartes PvP.";
	mes "Faites-moi savoir sur quelle carte";
	mes "tu veux que je te déplace.";
	next;
	set .@i, select("Map 1:Map 2:Map 3:Map 4:Map 5");
	mes "[Guide de bataille de guilde]";
	mes "You've chosen Map "+.@i+".";
	mes "Maintenant, quelle équipe est";
	mes "tu es allumé ? Vous pouvez choisir";
	mes "soit l'équipe 1, soit l'équipe 2.";
	next;
	set .@j, select("Team 1:Team 2");
	mes "[Guide de bataille de guilde]";
	mes "Très bien, je vais te déplacer";
	mes "to Map "+.@i+" as a member";
	mes "of Team "+.@j+". Are you ready";
	mes "y être transporté maintenant ?";
	next;
	if(select("Yes:No") == 2) {
		mes "[Guide de bataille de guilde]";
		mes "Très bien, j'espère";
		mes "on se reverra sur";
		mes "les champs PvP !";
		close;
	}
	mes "[Guide de bataille de guilde]";
	mes "Super! Passez un bon moment~";
	close2;
	warp "guild_vs"+.@i,getd(".@map"+.@j+"X["+.@i+"]"),.@mapY[.@i];
	end;
}
