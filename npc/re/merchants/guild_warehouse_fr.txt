-	script	GuildWarehouse	-1,{
	mes "[Gérant de l'entrepôt]";
	mes "Comment allez-vous ? Nous sommes spécialisés dans les entrepôts de guilde. Voici notre ^0000cdhistoire^000000 :";
	mes "Pourquoi les membres de la guilde ne peuvent-ils pas partager un stockage ? C'est avec cette question simple que nous avons commencé.";
	next;
	mes "[Gérant de l'entrepôt]";
	mes "Vous pouvez ouvrir l'entrepôt pour 1000 Zeny ou consulter l'historique d'utilisation de la guilde.";
	mes "Comment puis-je t'aider?";
	next;
	if( select( "Open guild warehouse:View warehouse usage history" ) == 1 ){
		.@guildid = getcharid( 2 );

		if( .@guildid == 0 ){
			mes "[Gérant de l'entrepôt]";
			mes "Le stockage de la guilde n'est disponible que pour les membres de la guilde.";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "[Gérant de l'entrepôt]";
			mes "Il semble que votre guilde ne puisse pas encore utiliser l'entrepôt de guilde ou que vous n'avez pas les permissions d'accès requises.";
			mes "Veuillez revenir après avoir vérifié la compétence de guilde et les permissions d'accès à l'entrepôt.";
			close;
		}

		if( Zeny < 1000 ){
			mes "[Gérant de l'entrepôt]";
			mes "Je crains que vous n'ayez pas assez d'argent pour régler les frais.";
			mes "Les frais s'élèvent à 1000 Zeny.";
			close;
		}

		mes "[Gérant de l'entrepôt]";
		mes "Je vais alors ouvrir le stockage de la guilde pour vous. Passez un moment inoubliable !";
		close2;

		if( Zeny < 1000 ){
			// Cheat prevention
			end;
		}

		if( guildopenstorage() == GSTORAGE_OPEN ){
			Zeny -= 1000;
			end;
		}else{
			mes "[Gérant de l'entrepôt]";
			mes "Il semble qu'une autre personne utilise actuellement l'entrepôt.";
			mes "Veuillez réessayer dans quelques instants.";
			close;
		}
	}else{
		.@guildid = getcharid( 2 );

		if( .@guildid == 0 ){
			mes "[Gérant de l'entrepôt]";
			mes "Le stockage de la guilde n'est disponible que pour les membres de la guilde.";
			close;
		}

		if( getgdskilllv( .@guildid, "GD_GUILD_STORAGE" ) == 0 || !guild_has_permission( GUILD_PERM_STORAGE ) ){
			mes "[Gérant de l'entrepôt]";
			mes "Il semble que votre guilde ne puisse pas encore utiliser l'entrepôt de guilde ou que vous n'avez pas les permissions d'accès requises.";
			mes "Veuillez revenir après avoir vérifié la compétence de guilde et les permissions d'accès à l'entrepôt.";
			close;
		}

		mes "[Gérant de l'entrepôt]";
		mes "Je vais vous montrer l'historique des utilisations de l'entrepôt. L'historique est conservé pendant un maximum de 3 mois.";
		mes "Passez une excellente journée !";
		close2;
		guildopenstorage_log();
		end;
	}
}
