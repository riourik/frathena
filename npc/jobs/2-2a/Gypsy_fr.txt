valkyrie,53,56,3	script	Gypsy#Valkyrie	101,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Gypsy]";
			mes "Félicitations.";
			mes "Honneur aux guerriers !";
			close;
		}
		mes "[Gypsy]";
		mes "Déplace-toi vers la gauche,";
		mes "déplace-toi vers la droite~!";
		mes "Et danse...!";
		mes "La danse peut être";
		mes "plus qu'un hobby.";
		mes "Pour moi, c'est un mode de vie~";
		close;
	}
	else if (ADVJOB == Job_Gypsy && Class == Job_Archer_High && JobLevel > 39) {
		mes "[Gypsy]";
		mes "Le monde de Midgard";
		mes "a besoin de femmes talentueuses";
		mes "pour changer subtilement les équilibres";
		mes "dans la lutte entre le bien";
		mes "et le mal.";
		next;
		mes "[Gypsy]";
		mes "Es-tu prêt(e)";
		mes "Veux-tu endosser ce rôle,";
		mes "et devenir une Tzigane?";
		next;
		if (select("No.:Yes.") == 1) {
			mes "[Gypsy]";
			mes "[Grand Prêtre]";
			mes "n'hésitez pas à revenir.";
			mes "Honneur aux guerriers !";
			close;
		}
		if (SkillPoint) {
			mes "[Gypsy]";
			mes "Il est encore possible pour vous d'apprendre des compétences supplémentaires. Veuillez utiliser";
			mes "tous vos points de compétence restants avant de revenir me voir.";
			close;
		}
		jobchange Job_Gypsy;
		set ADVJOB,0;
		mes "[Gypsy]";
		mes "Félicitations!";
		mes "Comme une Bohémienne, je sais";
		mes "que vos représentations";
		mes "ébranlent les cœurs de tous";
		mes "ceux qui assisteront à cela...";
		close;
	}
	mes "[Gypsy]";
	mes "Accueillir";
	mes "vers Valhalla,";
	mes "la Salle de l'Honneur.";
	next;
	mes "[Gypsy]";
	mes "S'il vous plaît faites-vous";
	mes "confortable";
	mes "pendant votre séjour ici.";
	mes "Honneur aux guerriers !";
	close;
}
