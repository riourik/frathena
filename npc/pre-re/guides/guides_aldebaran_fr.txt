aldebaran,139,63,4	script	Soldier#alde	105,{
	cutin "prt_soldier",2;
	mes "[Garde d'Al de Baran]";
	mes "Je suis juste un gardien ordinaire";
	mes "que vous pourriez trouver dans n'importe quelle autre ville. Je ne crois même pas avoir de nom...";
	next;
	mes "[Garde d'Al de Baran]";
	mes "Je suis en charge des Guides de Service du Garrison d'Al De Baran. Laissez-moi vous guider";
	mes "à travers notre ville !";
	next;
	if (select("Get Location Guide.:End conversation.") == 1) {
		viewpoint 1,61,229,0,0xFF6633;
		viewpoint 1,72,197,1,0x0000FF;
		viewpoint 1,223,222,2,0x00FFFF;
		viewpoint 1,233,105,3,0x515151;
		viewpoint 1,197,70,4,0x3355FF;
		viewpoint 1,60,60,5,0xFF5555;
		mes "^FF6633+^000000 -> Bureau principal de Kafra";
		mes "^0000FF+^000000 -> Magasin d'armes";
		mes "^00FFFF+^000000 -> Guilde des Sorciers (Fermé)";
		mes "^515151+^000000 -> Auberge";
		mes "^3355FF+^000000 -> Magasin d'Objets";
		mes "^FF5555+^000000 -> Guilde des Alchimistes";
		close2;
		cutin "prt_soldier",255;
		end;
	}
	mes "[Garde d'Al de Baran]";
	mes "Nous sommes jurés de protéger Al De Baran ! Que les forces du mal soient toujours écrabouillées par le poing juste du bien !";
	mes "righteous fist of good!";
	close2;
	cutin "prt_soldier",255;
	end;
}

		mes "^00FFFF+^000000 -> Guilde des Sorciers (Fermé)";
		break;

		break;
