veins,210,345,5	script	Veins Guide#1::ve_guide	934,{
	mes "[Guide des Veines]";
	mes "La ville de Veins accueille les aventuriers à la recherche d'abri contre les tempêtes de sable.";
	mes "Si c'est la première fois que vous utilisez nos services de guide, pourquoi ne pas vérifier le...";
	set .@loop1,1;
	while(.@loop1) {
		next;
		switch(select("Village Guide:Remove Marks from Mini-Map:Notice:Cancel")) {
		case 1:
			mes "[Guide des Veines]";
			mes "Je peux vous indiquer l'emplacement de n'importe quel bâtiment tant qu'il est à Veins.";
			mes "Alors où voulez-vous aller?";
			if (.@compass_check == 0) {
				mes "Voulez-vous que je marque les emplacements";
				mes "sur votre Mini-Mappe ?";
				mes "^FF0000 sur votre Mini-Map ?";
				next;
				if (select("Yes:No") == 1) set .@compass_check,1;
			}
			set .@loop2,1;
			while(.@loop2) {
				if (.@wait_button_chk == 0) set .@wait_button_chk,1;
				else next;

				switch(select("Temple:Inn:Weapon Shop:Tool Shop:Airship:Tavern:Geological Research Institute:Cancel")) {
				case 1:
					mes "[Guide des Veines]";
					mes "Notre temple se trouve au nord,";
					mes "et est toujours bondé de fidèles sincères de la déesse Freya.";
					if (.@compass_check == 1) viewpoint 1,196,258,1,0xFF0000;
					break;
				case 2:
					mes "[Guide des Veines]";
					mes "Vous pouvez vous reposer de la fatigue du voyage à l'auberge.";
					mes "Le bâtiment de gauche à côté de moi est l'auberge de Veins.";
					if (.@compass_check == 1) viewpoint 1,128,266,2,0xFF00FF;
					break;
				case 3:
					mes "[Guide des Veines]";
					mes "Oui, vous devriez protéger vous-même des dangers.";
					mes "Achetez des armes de qualité au prix abordable.";
					mes "Le magasin d'armes de Veins est situé à l'Ouest.";
					if (.@compass_check == 1) viewpoint 1,150,175,3,0x99FFFF;
					break;
				case 4:
					mes "[Guide des Veines]";
					mes "Avez-vous emporté assez de nécessités pour votre aventure?";
					mes "Si ce n'est pas le cas, je vous suggère de vérifier ce que les centres Veins peuvent offrir.";
					if (.@compass_check == 1) viewpoint 1,230,161,4,0x0000FF;
					break;
				case 5:
					mes "[Guide des Veines]";
					mes "Veuillez noter que Veins ne fonctionne que pour les cargos aériens.";
					if (.@compass_check == 1) viewpoint 1,273,285,5,0x00FF00;
					break;
				case 6:
					mes "[Guide des Veines]";
					mes "Si vous souhaitez faire ami avec";
					mes "les habitants de la ville, je vous suggère d'";
					mes "aller boire un verre au tavernier situé à l'ouest.";
					mes "the west.";
					if (.@compass_check == 1) viewpoint 1,150,217,6,0x00FF00;
					break;
				case 7:
					mes "[Guide des Veines]";
					mes "Êtes-vous intéressé(e) par l'étude de géologie?";
					mes "Alors il serait préférable que vous alliez vérifier le";
					mes "Institut de Recherche Géologique situé au deuxième étage du magasin d'armes.";
					mes "the 2nd floor of the weapon shop.";
					if (.@compass_check == 1) viewpoint 1,150,175,7,0x00FF00;
					break;
				case 8:
					mes "[Guide des Veines]";
					mes "Si vous aimez vous débarrasser de tous les marqueurs de lieu sur votre Minimaps,";
					mes "demandez-moi encore et choisissez l'option 'Supprimer les repères sur la carte dépliable'.";
					set .@loop2,0;
					break;
				}
			}
			break;

		case 2:
			viewpoint 2,196,258,1,0xFF0000;
			viewpoint 2,128,266,2,0xFF00FF;
			viewpoint 2,150,175,3,0x99FFFF;
			viewpoint 2,230,161,4,0x0000FF;
			viewpoint 2,273,285,5,0x00FF00;
			viewpoint 2,150,217,6,0x00FF00;
			viewpoint 2,150,175,7,0x00FF00;
			mes "[Guide des Veines]";
			mes "D'accord, ils sont partis maintenant. Si vous avez plus de lieux à demander, faites-le-moi savoir.";
			mes "Profitez de votre séjour dans les Veines.";
			break;
		case 3:
			mes "[Guide des Veines]";
			mes "Quand vous utilisez le menu ''Village Guide'',";
			mes "assurez-vous que les emplacements des bâtiments seront marqués sur votre minimap située en haut à droite de l'écran.";
			mes "Si vous ne pouvez pas voir votre minimap, utilisez la touche raccourcie ''ctrl+tab'' ou appuyez sur le bouton ''Carte'' dans vos fenêtres d'informations de base, d'accord?";
			mes "Et vous pouvez également zoomer out sur votre minimap en utilisant le bouton ''-'', au cas où vous ne pourriez pas voir tout le village.";
			break;
		case 4:
			mes "[Guide des Veines]";
			mes "Profitez de votre séjour dans les Veines.";
			close2;
			set .@loop1,0;
			break;
		}
	}
	close;
}

veins,189,101,5	duplicate(ve_guide)	Veins Guide#2	934
