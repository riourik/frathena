moro_cav,32,73,5	script	Hero Token Exchanger#1	4_F_SHADOWCHASER,{
	disable_items;
	mes "[Échangeur de Jetons Héros]";
	mes "Si vous avez combattu Morocc";
	mes "et recueilli";
	mes "des Tokens Héros,";
	mes "je peux les échanger contre une bague spéciale pour vous.";
	mes "Voulez-vous jeter un coup d'œil à ma bague ?";
	while(true) {
		next;
		if (checkweight(2163,2) == 0) {
			mes "[Échangeur de Jetons Héros]";
			mes "Votre sac contient beaucoup d'objets,";
			mes "Merci de revenir après l'avoir rangé !";
			close;
		}
		switch( select( "Ring description", "Exchange with the ring.", "End conversation." ) ) {
		case 1:
			mes "[Échangeur de Jetons Héros]";
			mes "Cette Bague Héroïque";
			mes "pourrait ne pas";
			mes "avoir l'air";
			mes "impressionnante,";
			next;
			mes "[Échangeur de Jetons Héros]";
			mes "mais emmenez-la chez";
			mes "l'Enchanteur Expert Byrnes,";
			mes "et il pourra Enchanter";
			mes "les quatre alvéoles";
			mes "et faire ressortir";
			mes "ses capacités latentes.";
			next;
			mes "[Échangeur de Jetons Héros]";
			mes "Anneau Héros";
			mes "peut vous aider";
			mes "à soumettre";
			mes "Morocc.";
			mes "Des questions supplémentaires ?";
			continue;
		case 2:
			mes "- Anneau Héros -";
			mes "^006400Une bague conçue pour les guerriers^000000";
			mes "^006400dans la quête de vaincre Morocc..^000000";
			mes "^006400Lorsqu'elle est enchantée,^000000";
			mes "^006400elle prête à son propriétaire^000000";
			mes "^006400le pouvoir de défier Morocc.^000000";
			mes "^006400Classe : Accessoire/DEF : 0^000000";
			mes "^006400Poids : 20/Niveau requis : 160^000000";
			mes "^006400Emplacement : 0/Équipement pour tous les métiers^000000";
			next;
			mes "[Échangeur de Jetons Héros]";
			mes "L'Anneau Héros nécessite";
			mes "30 Jetons Héros";
			mes "pour l'échange.";
			mes "Vous voulez échanger ?";
			next;
			if (select( "Previous menu.", "Exchange 30 Hero Tokens." ) == 1) {
				mes "[Échangeur de Jetons Héros]";
				mes "Pensez-y bien";
				mes "encore.";
				continue;
			}
			if (countitem(6684) < 30) {
				mes "[Échangeur de Jetons Héros]";
				mes "^006400Jetons Héros^000000";
				mes "est remis à";
				mes "ceux qui ont bravement combattu Morocc.";
				mes "Tu n'as pas";
				mes "30 ^006400Jetons Héros^000000,";
				mes "et par conséquent, je ne peux pas vous aider.";
				continue;
			}
			mes "[Échangeur de Jetons Héros]";
			mes "Je vais échanger";
			mes "30 ^006400Jetons Héros^000000";
			mes "contre 1 ^006400Anneau Héros^000000.";
			mes "^FF0000Autre chose?";
			delitem 6684,30;	// TokenOfHero
			getitem 2981,1;		// RingOfHero
			continue;
		case 3:
			mes "[Échangeur de Jetons Héros]";
			mes "Un jour,";
			mes "vous vous retrouverez";
			mes "dans le besoin d'un Anneau Héros.";
			close;
		}
	}
}

moro_cav,34,65,5	script	Expert Enchanter Byrnes	1_M_03,{
	disable_items;
	mes "[Byrnes]";
	mes "Avez-vous entendu parler de l'Anneau Héros ? Seuls les vrais héros méritent de posséder cet magnifique anneau, et j'attends celui-ci depuis longtemps.";
	mes "^FF0000Que puis-je faire pour vous?";
	next;
	switch( select( "Enchant Hero Ring.", "Reset Enchantments on Hero Ring.", "Cancel." ) ) {
	case 1:
		mes "[Byrnes]";
		mes "Je ne peux enchanté que les anneaux équipés. Quel anneau équipé souhaitez-vous faire enchâter ?";
		next;
		setarray .@eqi_slot[0], EQI_ACC_L, EQI_ACC_R;
		.@s = select( "Left", "Right" ) - 1;
		switch( getequipid(.@eqi_slot[.@s]) ) {
		case -1:
			mes "[Byrnes]";
			mes "Veuillez équiper l'Anneau Héroïque que vous souhaitez enchanté.";
			close;
		case 2981:
			break;
		default:
			mes "[Byrnes]";
			mes "Veuillez équiper votre Anneau Héroïque. Je ne peux rien d'autre enchâter.";
			close;
		}
		// custom at this point
		setarray .@card[0],
			getequipcardid(.@eqi_slot[.@s],0),
			getequipcardid(.@eqi_slot[.@s],1),
			getequipcardid(.@eqi_slot[.@s],2),
			getequipcardid(.@eqi_slot[.@s],3);

		if (.@card[0] > 0) {
			mes "[Byrnes]";
			mes "Un Anneau Héroïque entièrement enchanté ne peut plus être enchanté davantage.";
			close;
		}
		if (.@card[3] == 0) {
			setarray .@enchant_list$[0], "DEF", "MDEF";
			.@slot = 4;
		}
		else if (.@card[2] == 0) {
			setarray .@enchant_list$[0], "Max HP", "Max SP";
			.@slot = 3;
		}
		else if (.@card[1] == 0) {
			setarray .@enchant_list$[0], "Atk", "MAtk";
			.@slot = 2;
		}
		else {
			setarray .@enchant_list$[0], "STR", "INT", "AGI", "VIT", "DEX", "LUK";
			.@slot = 1;
		}
		mes "[Byrnes]";
		mes "Which ability do you want as an enchantment on slot " + .@slot + "?";
		next;
		.@enchant_list_size = getarraysize(.@enchant_list$);
		for ( .@i = 0; .@i < .@enchant_list_size; ++.@i )
			.@menu$ += .@enchant_list$[.@i] + ":";
		.@type = select(.@menu$) - 1;

		switch(.@slot) {
		case 4:
			switch(.@type) {// enchant list index
			case 0:	// DEF
				setarray .@bonus[0], 4791, 4792, 4793, 4794, 4893;
				break;
			default:	// MDEF
				setarray .@bonus[0], 4890, 4786, 4891, 4787, 4892;
				break;
			}
			break;
		case 3:
			switch(.@type) {
			case 0:	// Max HP
				setarray .@bonus[0], 4861, 4862, 4867, 4868, 4900;
				break;
			default:	// Max SP
				setarray .@bonus[0], 4800, 4870, 4871, 4801, 4802;
				break;
			}
			break;
		case 2:
			switch(.@type) {
			case 0:	// Atk
				setarray .@bonus[0], 4882, 4766, 4767, 4894, 4895;
				break;
			default:	// Matk
				setarray .@bonus[0], 4883, 4896, 4897, 4898, 4899;
				break;
			}
			break;
		case 1:
			switch(.@type) {
			case 0:	// STR
				setarray .@bonus[0], 4700, 4701, 4702, 4703, 4704;
				break;
			case 1:	// INT
				setarray .@bonus[0], 4710, 4711, 4712, 4713, 4714; 
				break;
			case 2:	// AGI
				setarray .@bonus[0], 4730, 4731, 4732, 4733, 4734;
				break;
			case 3:	// VIT
				setarray .@bonus[0], 4740, 4741, 4742, 4743, 4744;
				break;
			case 4:	// DEX
				setarray .@bonus[0], 4720, 4721, 4722, 4723, 4724;
				break;
			default:	// LUK
				setarray .@bonus[0], 4750, 4751, 4752, 4753, 4754;
				break;
			}
			break;
		}

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@eqi_slot[.@s], 2981) || callfunc("F_IsEquipCardHack", .@eqi_slot[.@s], .@card[0], .@card[1], .@card[2], .@card[3]))
			close;

		.@bonus_size = getarraysize(.@bonus);
		.@card[.@slot-1] = .@bonus[ rand(.@bonus_size) ];	// unknown rates
		specialeffect2 EF_ANGEL2;
		mes "[Byrnes]";
		mes "Magie terminée !";
		delequip .@eqi_slot[.@s];
		getitem2 2981,1,1,0,0,.@card[0],.@card[1],.@card[2],.@card[3];
		close;
	case 2:
		mes "[Byrnes]";
		mes "Pour réinitialiser les Enchantements sur votre Anneau Héroïque, j'ai besoin de 3 Tokens Héroïques.";
		next;
		if (countitem(6684) < 3) {
			mes "[Byrnes]";
			mes "Vous n'avez pas assez de Tokens Héroïques. Je ne peux pas réinitialiser votre Anneau Héroïque.";
			close;
		}
		mes "[Byrnes]";
		mes "Je ne peux réinitialiser que les anneaux équipés. Lequel des anneaux équipés souhaitez-vous réinitialiser ?";
		next;
		setarray .@eqi_slot[0], EQI_ACC_L, EQI_ACC_R;
		.@s = select( "Left", "Right" ) - 1;
		switch( getequipid(.@eqi_slot[.@s]) ) {
		case -1:
			mes "[Byrnes]";
			mes "Vous devez d'abord équiper l'objet pour réinitialiser son statut d'Enchantement.";
			close;
		case 2981:
			break;
		default:
			mes "[Byrnes]";
			mes "Veuillez équiper votre Anneau Héroïque. Je ne peux rien d'autre réinitialiser.";
			close;
		}
		// custom at this point
		if (countitem(6684) < 3) {
			mes "[Byrnes]";
			mes "Vous n'avez pas assez de Tokens Héroïques. Je ne peux pas réinitialiser votre Anneau Héroïque.";
			close;
		}
		setarray .@card[0],
			getequipcardid(.@eqi_slot[.@s],0),
			getequipcardid(.@eqi_slot[.@s],1),
			getequipcardid(.@eqi_slot[.@s],2),
			getequipcardid(.@eqi_slot[.@s],3);

		if (.@card[3] == 0) {
			mes "[Byrnes]";
			mes "Votre Anneau de Héros est actuellement dans un état sans enchantement.";
			close;
		}

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@eqi_slot[.@s], 2981) || callfunc("F_IsEquipCardHack", .@eqi_slot[.@s], .@card[0], .@card[1], .@card[2], .@card[3]))
			close;

		specialeffect2 EF_ANGEL2;
		mes "[Byrnes]";
		mes "L'initialisation est terminée.";
		delitem 6684,3;
		delequip .@eqi_slot[.@s];
		getitem 2981,1;
		close;
	case 3:
		close;
	}
}
