//callfunc "F_CoolEventCorp","<menu>","<location>","<save map>",<x>,<y>,"<warp map>";
function	script	F_CoolEventCorp	{
	cutin "zonda_01",2;
	mes "La Cool Event Corp. s'efforce toujours";
	mes "Bienvenue chez Cool Event Corp.";
	mes "Notre personnel travaille toujours";
	mes "pour surpasser vos attentes";
	mes "en matière de service de qualité. Alors,";
	mes "comment puis-je vous aider aujourd'hui ?";
	next;
	switch(select(getarg(0))) {
	case 1:
		mes "La Cool Event Corp. s'efforce toujours";
		mes "Votre point de respawn";
		mes "a été sauvegardé ici";
		mes getarg(1)+".";
		mes "Merci d'avoir utilisé le";
		mes "Service Cool Event Corp.~";
		savepoint getarg(2),getarg(3),getarg(4),1,1;
		close2;
		break;
	case 2:
		mes "La Cool Event Corp. s'efforce toujours";
		if(!callfunc("F_CanOpenStorage")){
			mes "Je suis désolé, mais toi";
			mes "j'ai besoin du statut d'apprenti";
			mes "Niveau de compétence basique 6 pour";
			mes "utiliser le service de stockage.";
			close2;
			break;
		}
		if(Zeny < 40){
			mes "Je suis désolé, mais tu ne le fais pas";
			mes "avez-vous assez de Zeny pour utiliser";
			mes "le Service de Stockage. Notre";
			mes "taxe d'accès au stockage est de 40 Zeny.";
			close2;
			break;
		}
		set Zeny, Zeny - 40;
		mes "Permettez-moi d'ouvrir votre espace";
		mes "de stockage personnel pour vous tout de suite.";
		mes "Merci de soutenir Cool Event Corp. en utilisant nos";
		mes "services. Passez une bonne journée~";
		mes "services. Have a good day~";
		callfunc("F_CheckKafCode");	//check your storage password, if set
		close2;
		openstorage;
		break;
	case 3:
		mes "Veuillez choisir";
		mes "votre destination.";
		next;
		if (VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE))
			set .@cost,4400;
		else
			set .@cost,2200;
		.@destination$ = getarg(5);
		if (select(.@destination$ +" -> "+ .@cost +" z:Cancel") == 1) {
			if ((countitem(7060) != 0) || (Zeny >= .@cost)) {
				if (countitem(7060) != 0) {
					delitem 7060,1; //Warp_Free_Ticket
				}
				else {
					set Zeny,Zeny-.@cost;
				}
				if (.@destination$ == "Veins") warp "veins",205,101;
				else if (.@destination$ == "Rachel") warp "rachel",115,125;
				cutin "",255;
				end;
			}
			else {
				mes "La Cool Event Corp. s'efforce toujours";
				mes "Désolé, mais vous n'avez pas";
				mes "assez d'argent (zeny) pour le service Téléportation.";
				mes "Le frais de téléportation est";
				mes "to "+ .@destination$ +" is "+.@cost+" zeny.";
			}
		}
		close2;
		break;
	case 4:
		mes "La Cool Event Corp. s'efforce toujours";
		if(BaseClass != Job_Merchant){
			mes "^FF0000Je suis désolé, mais le";
			mes "service de location de chariot n'est disponible que pour les";
			if (checkre(0))
				mes "est disponible uniquement pour les classes Commerçant.";
			else {
				mes "[Marchands], [Forgemestres], [Maîtres Forgemestres], [Alchimistes], [Bio-Alchimistes],";
				mes "Forgerons, Artilleurs blancs,";
				mes "Alchimistes et Créateurs.";
			}
			close2;
			break;
		}
		if(getskilllv("MC_PUSHCART") == 0){
			mes "Vous ne pouvez louer un chariot qu'après";
			mes "avoir appris la compétence Caddie.";
			close2;
			break;
		}
		if(checkcart() == 1){
			mes "Malheureusement, nous ne pouvons pas";
			mes "un Caddie équipé.";
			mes "Malheureusement, nous ne pouvons pas";
			mes "louer plus d'un à";
			mes "à la fois par client.";
			close2;
			break;
		}
		if(countitem(7061)){
			delitem 7061,1; //Cart_Free_Ticket
			setcart;
			close2;
			break;
		}
		mes "La location du landau de marchandises";
		mes "la location d'un chariot est de 800 Zeny. Accepteriez-vous de payer cette";
		mes "vous devez payer une redevance pour le landau de marchandises";
		next;
		if(select("Rent a Pushcart:Cancel") == 1){
			if(Zeny < 800){
				mes "La Cool Event Corp. s'efforce toujours";
				mes "Je suis désolé, mais toi";
				mes "je n'en ai pas assez";
				mes "800 Zeny pour la location du chariot.";
				mes "[Cool Event Corp. Staff]";
			} else {
				set Zeny, Zeny - 800;
				setcart;
			}
		}
		close2;
		break;
	case 5:
		callfunc("F_SetKafCode","[Cool Event Corp. Staff]","Cool Event Corp.");
		//the function closes msg and removes cutins
	case 6:
	default:
		mes "La Cool Event Corp. s'efforce toujours";
		mes "de fournir les meilleurs services à nos clients.";
		mes "Aidez-nous à devenir le meilleur en nous faisant part de vos avis";
		mes "et de vos retours honnêtes. Merci.";
		mes "Help us become the best by";
		mes "providing us with your opinions";
		mes "and honest feedback. Thank you.";
		close2;
	}
	cutin "", 255;
	end;
}

	callfunc "F_CoolEventCorp",
		"Save:Use Storage::Rent a Pushcart:Storage Password Service:Cancel",
		"in the town of Lighthalzen","lighthalzen",95,240;
	end;
}
