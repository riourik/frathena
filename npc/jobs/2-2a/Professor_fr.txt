			mes "[Scholar]";
			mes "Félicitations.";
			mes "Honneur aux guerriers !";

		}
		mes "[Scholar]";
		mes "Cela prend une vie entière...";
		mes "Littéralement toute une vie";
		mes "pour rassembler le savoir";
		mes "nécessaire pour devenir";
		mes "un Savant...";
		next;
		mes "[Scholar]";
		mes "C'est écrasant.";
		mes "Plus vous apprenez, plus";
		mes "vous découvrez ce que";
		mes "vous ne savez pas encore.";
		mes "Il n'y a pas de fin au";
		mes "processus d'apprentissage...";
		close;
	}
	else if (ADVJOB == Job_Professor && Class == Job_Mage_High && JobLevel > 39) {
		mes "[Scholar]";
		mes "Midgard n'a pas assez de";
		mes "Savants pour aider à";
		mes "accueillir une nouvelle ère";

		mes "monde a besoin de vous...";

		mes "^FF0000Voudrez-vous^000000 prendre cette";
		mes "^FF0000responsabilité^000000 étonnante?";
		mes "^FF0000Voudrez-vous^000000 servir Midgard";
		mes "^FF0000en tant que^000000 Scholar?";

			mes "[Scholar]";
			mes "[Grand Prêtre]";
			mes "n'hésitez pas à revenir.";
			mes "Honneur aux guerriers !";
			close;
		}
		if (SkillPoint) {
			mes "[Scholar]";
			mes "Il est encore possible pour vous d'apprendre des compétences supplémentaires. Veuillez utiliser";
			mes "tous vos points de compétence restants avant de revenir me voir.";
			close;
		}
		jobchange Job_Professor;
		set ADVJOB,0;
		mes "[Scholar]";
		mes "Félicitations!";
		mes "^FF0000En tant que Professeur, j'espère^000000 que vous prendrez";

	mes "[Scholar]";
	mes "S'il vous plaît faites-vous";
	mes "confortable";
	mes "pendant votre séjour ici.";
	mes "Honneur aux guerriers !";
	close;
}
