// PecoPeco Breeder (for Knights) :: breeder
//============================================================
prontera,55,350,5	script	Peco Peco Breeder#knt	105,{
	if(Upper==0) set .@price,2500; //Normal Peco - default price
	if(Upper==1) set .@price,2500; //Armored Peco
	if(Upper==2) set .@price,2500; //Baby Peco

	mes "[Peco Peco Breeder]";
	if (BaseJob == Job_Knight && Class < Job_Rune_Knight) {
		mes "Accueillir.";
		mes "Chevalier honoré,";
		mes "aimeriez-vous louer";
		mes "un Peco Peco? La location";
		mes "fee is "+.@price+" zeny.";
		next;
		switch(select("Rent Peco Peco:Cancel")) {
		case 1:
			if (Zeny < .@price) {
				mes "[Peco Peco Breeder]";
				mes "Vous ne";
				mes "Vérifiez combien vous avez de zeny";
				mes "Êtes-vous...";
				mes "êtes pas en faillite?";
				close;
			}
			else if(getskilllv("KN_RIDING") == 0) {
				mes "[Peco Peco Breeder]";
				mes "Je suis désolé, mais vous n'êtes";
				mes "pas éligible pour ce";
				mes "service. Veuillez apprendre";
				mes "la compétence Monture Peco Peco d'abord.";
				close;
			}
			else if(checkriding()) {
				mes "[Peco Peco Breeder]";
				mes "Vous êtes déjà";
				mes "en selle sur un";
				mes "Peco Peco.";
				close;
			}
			else if(ismounting()) {
				mes "[Peco Peco Breeder]";
				mes "Veuillez enlever votre monture cash.";
				close;
			}
			set Zeny, Zeny-.@price;
			setriding;
			close;
		case 2:
			mes "[Peco Peco Breeder]";
			mes "Je vois.";
			mes "Bien alors,";
			mes "passe une bonne journée.";
			close;
		}
	} else {
		mes "Je suis désolé, mais ces";
		mes "Les Peco Pecos sont uniquement";
		mes "disponibles pour les Chevaliers";
		mes "et les Seigneurs de Guerre.";
		close;
	}
}

// Grand PecoPeco Breeder (for Crusaders)
//============================================================
prontera,232,318,3	script	Peco Peco Breeder#cru	105,{
	if(Upper==0) set .@price,3500; //Normal Peco - default price
	if(Upper==1) set .@price,3500; //Armored Peco
	if(Upper==2) set .@price,3500; //Baby Peco

	mes "[Peco Peco Breeder]";
	if (BaseJob == Job_Crusader && Class < Job_Rune_Knight) {
		if(Upper != 1 ) mes "Bienvenue, Croisé(e).";
		else mes "Bienvenue, Paladin(e).";
		mes "Nous avons un Peco Peco spécial";
		mes "préparé pour vous. Pour le louer";
		mes "veuillez vous rendre à";
		mes "will cost "+.@price+" zeny.";
		next;
		switch(select("Rent a PecoPeco:Quit")) {
		case 1:
			if (Zeny < .@price) {
				mes "[Peco Peco Breeder]";
				mes "Vous ne";
				mes "Vérifiez combien vous avez de zeny";
				mes "Si vous souhaitez obtenir un";
				mes "Peco Peco, merci de revenir plus tard.";
				mes "bring "+.@price+" zeny...";
				close;
			}
			else if(getskilllv("KN_RIDING") == 0) {
				mes "[Peco Peco Breeder]";
				mes "Vous devez d'abord apprendre";
				mes "à monter sur un Peco Peco avant que";
				mes "je puisse vous en louer un.";
				close;
			}
			else if(checkriding()) {
				mes "[Peco Peco Breeder]";
				mes "Vous êtes déjà";
				mes "assis(e) sur un Peco Peco.";
				close;
			}
			else if(ismounting()) {
				mes "[Peco Peco Breeder]";
				mes "Veuillez enlever votre monture cash.";
				close;
			}
			set Zeny, Zeny-.@price;
			setriding;
			close;
		case 2:
			mes "[Peco Peco Breeder]";
			mes "Au revoir pour maintenant.";
			close;
		}
	} else {
		mes "^FF0000Que puis-je faire pour vous?";
		mes "Sachez que";
		mes "ce service de location de Peco Peco";
		mes "est strictement réservé aux";
		mes "Croisés et Paladins.";
		close;
	}
}

// Falcon Master :: breeder

hu_in01,381,304,5	script	Falcon Breeder#hnt	105,{
	if(Upper==0) set .@price,2500; //Normal Falcon - default price
	if(Upper==1) set .@price,2500; //Scarf Falcon
	if(Upper==2) set .@price,2500; //Baby Falcon

	mes "^FF0000Jeune fou !";
	if (BaseJob == Job_Hunter) {
		if (checkwug()) {
			mes "Euh...";
			mes "Vous ne pouvez pas louer un Falcon";
			mes "jusqu'à ce que vous démontiez votre warg d'abord !";
			close;
		}
		mes "Voulez-vous louer un Falcon ?";
		mes "Vous pouvez louer votre propre";
		mes "fiable oiseau de proie pour une";
		mes "fee of just "+.@price+" zeny~";
		next;
		switch(select("Rent Falcon:Cancel")) {
		case 1:
			if (checkwug()) {
				mes "^FF0000Jeune fou !";
				mes "Euh...";
				mes "Vous ne pouvez pas louer un Falcon";
				mes "jusqu'à ce que vous démontiez votre warg d'abord !";
				close;
			}
			if (Zeny < .@price) {
				mes "^FF0000Jeune fou !";
				mes "C'est quoi ça ?";
				mes "Tu n'as pas";
				mes "assez de zeny ?!";
				mes "^FF0000Commence plutôt à chercher de l'argent";
				mes "^FF0000plutôt que des monstres~";
				mes "^FF0000!";
				close;
			}
			else if(getskilllv("HT_FALCON") == 0) {
				mes "^FF0000Jeune fou !";
				mes "^FF0000Oh non~";
				mes "^FF0000Va apprendre comment gérer un";
				mes "^FF0000Faucon avant ! Je ne peux pas te";
				mes "^FF0000louer si tu n'es pas capable de le";
				mes "^FF0000gérer, comprends-tu ?";
				mes "[Falcon Breeder]";
				close;
			}
			else if(checkfalcon()) {
				mes "^FF0000Jeune fou !";
				mes "Euh...";
				mes "Malheureusement, nous ne pouvons pas";
				mes "[Falcon Breeder]";
				mes "^FF0000Hé ! Où vas-tu ? Ces faucons sont de première qualité, je te le garantis ! Allez, viens~";
				close;
			}
			set Zeny, Zeny-.@price;
			setfalcon;
			close;
		case 2:
			mes "^FF0000Jeune fou !";
			mes "^FF0000Les faucons ne peuvent que";
			mes "you goin'? These";
			mes "Falcons are top notch,";
			mes "I guarantee it! C'mon, yo~";
			close;
		}
	} else {
		mes "Young fool!";
		mes "Falcons can only";
		mes "^FF0000utilisé par les Chasseurs";
		mes "^FF0000et les Tireurs, compris?";
		mes "^FF0000...Héhé, jaloux?";
		close;
	}
}
