		mes "[Kong]";

		next;

		next;

		mes .@s2$;
		close;

			warp "izlude",195,212; // Old coordinates: (194,180)

		close;

		mes .@no_money$;
		close;
	}
	set Zeny, Zeny - .@price;
	mes .@yes_money$;
	if (.@i == 1)
		warp "izlude",195,212;
	else
		warp "alberta",192,150;
	close;
