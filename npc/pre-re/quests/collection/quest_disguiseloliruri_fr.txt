			mes "[Deadman]";
			mes "Avez-vous terminé l'extermination des 50 Disguise ?";
			next;
			switch(select("Yes:No:I want to quit")) {
			case 1:
				mes "[Deadman]";
				mes "Hm, je ne pense pas que vous ayez encore exterminé 50 d'entre eux.";
				close;
			case 2:
				mes "[Deadman]";
				mes "Rappelez-vous, j'ai besoin de votre aide pour chasser les Disguise alors allez-en exterminer 50.";
				close;
			case 3:
				mes "[Deadman]";
				mes "Êtes-vous sûr de vouloir arrêter la chasse?";
				mes "Tout progrès que vous avez fait sera effacé";
				next;
				switch(select("Yes:No")) {
				case 1:
					mes "[Deadman]";
					mes "D'accord, revenez ici si vous changez d'avis.";
					erasequest 60173;
					close;
				case 2:
					mes "[Deadman]";
					mes "Veuillez exterminer 50 Disguise afin que ce lieu devienne un endroit où les gens peuvent venir et accumuler de l'Expérience.";
					close;
				}
			}
		}
		mes "[Deadman]";
		mes "Impressionnant, vous l'avez fait rapidement.";
		getexp 140600,95800;
		erasequest 60173;
		close;
	}
	if (checkquest(60176) != -1) {
		if (checkquest(60176,HUNTING) != 2 ) {
			mes "[Deadman]";
			mes "Avez-vous terminé l'extermination des 50 Loli Ruri ?";
			next;
			switch(select("Yes:No:I want to quit")) {
			case 1:
				mes "[Deadman]";
				mes "Hm, je ne pense pas que vous ayez encore exterminé 50 d'entre eux.";
				close;
			case 2:
				mes "[Deadman]";
				mes "Rappelez-vous, j'ai besoin de votre aide pour chasser les Loli Ruri alors allez-en exterminer 50.";
				close;
			case 3:
				mes "[Deadman]";
				mes "Êtes-vous sûr de vouloir arrêter la chasse?";
				mes "Tout progrès que vous avez fait sera effacé";
				next;
				switch(select("Yes:No")) {
				case 1:
					mes "[Deadman]";
					mes "D'accord, revenez ici si vous changez d'avis.";
					erasequest 60176;
					close;
				case 2:
					mes "[Deadman]";
					mes "Veuillez tuer 50 Loli Ruri afin de rendre cet endroit un lieu où les joueurs peuvent se rassembler et gagner de l'expérience.";
					close;
				}
			}
		}
		mes "[Deadman]";
		mes "Impressionnant, vous l'avez fait rapidement.";
		getexp 332000,239500;
		erasequest 60176;
		close;
	}

		mes "[Deadman]";
		mes "Oh super!";
		mes "Vous êtes ici pour m'aider, n'est-ce pas?";
		mes "Sinon, je ne sais pas quoi faire avec les monstres dans cet endroit!";
		next;
		switch(select("Sure:No")) {
		case 1:
			mes "[Deadman]";
			mes "Super! Je savais dès que je vous ai vu que je pouvais compter sur vous!";
			mes "J'ai besoin de réduire la Quantité de monstres qui sont actuellement en train de dévaster cette carte!";
			mes "Il y a trop de Disguise et de Loli Ruri autour, ce qui rend impossible pour les joueurs de niveau inférieur d'obtenir une bonne expérience ici.";

			mes "[Deadman]";
			mes "Alors, lequel de ces monstres voudriez-vous chasser pour moi?";
			next;
			switch(select("Disguise:Loli Ruri:Cancel")) {
			case 1:
				if (BaseLevel > 90) {
					mes "[Deadman]";
					mes "Merci pour votre aide extraordinaire avec les Disguise!";
					mes "Actuellement la Situation est stable et je n'ai plus besoin de votre aide pour les tuer.";

				mes "Tuez 50 de ces Disguise et faites-moi savoir quand vous avez terminé. Je vous récompenserai pour vos efforts.";

				mes "[Deadman]";

			}
		case 2:
			mes "[Deadman]";
			mes "^FF0000Si tu changes d'avis, reviens s'il te plaît.";
			close;
		}
	}

	mes "Désolé, je ne peux pas parler en ce moment, j'ai vraiment mal. Ces fichues Disguise et Loli Ruri vont me tuer.";
