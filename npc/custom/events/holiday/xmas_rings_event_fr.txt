prontera,156,174,4	script	Snower	713,{

	} else if (Q_XMSRINGS == 3) {
		mes "Snoowy m'a envoyé un oiseau avec un message. Elle m'a parlé de ton aide...";
		mes "Je vais te révéler le secret des anneaux magiques.";
		emotion ET_BEST;
		next;
		mes "[Snower]";
		mes "Ils ont une puissance magique seulement du 8 décembre au 8 janvier.";
		next;

		next;
		mes "[Snower]";
		mes "Si tu mets les deux anneaux ensemble, leur puissance sera doublée.";
		mes "Bonne journée!";
		close;
	} else {
		mes "Salut ! Moi et mes deux sœurs préparons des cadeaux pour toutes les trois fêtes !";
	}
	emotion ET_CRY;
	next;
	menu "Where are your sisters?",-, "What holidays?",M_PR,"What gifts?",M_WRAP,"May I help you?",M_QUEST;

	mes "[Snower]";
	mes "Ma petite sœur Sneewy est partie dans un pays lointain pour préparer des cadeaux pour leurs enfants pauvres... Je ne sais pas où. Tu devrais demander à ma grande sœur et à ses oiseaux.";
	next;
	mes "[Snower]";
	mes "Ma grande sœur ? Son nom est Snoowy. Elle est partie sur une île verte... Elle parle de cadeaux pour les dauphins ou quelque chose comme ça...";
	close;

M_PR:
	mes "[Snower]";
	mes "Les trois fêtes ? Eh bien, elles arrivent les unes après les autres !";
	mes "^000080Noël^000000, ^FF0000Karachun^000000 et ^008000Réveillon du Nouvel An^000000!";
	next;
	mes "[Snower]";
	mes "^FF0000Karachun^000000 ? Heh... C'est un ancien festival païen. Ils l'organisaient avant l'apparition des fêtes de Noël et du Nouvel An, tu sais.";
	next;
	goto M_QUEST;

M_WRAP:
	mes "[Snower]";
	mes "Mes sœurs disent que tous les cadeaux doivent avoir la même apparence. Sinon, certains enfants pourraient se battre pour ceux qui sont plus beaux!";
	next;
	mes "[Snower]";
	mes "Voulez-vous que j'emballasse vos boîtes de cadeaux luxueuses dans du papier gris?";
	next;
	menu "Wrap them, please!",-, "No, thanks.",M_QUEST;

	set @prizes, countitem(664)+countitem(665)+countitem(666)+countitem(667);
	if (@prizes == 0) {
		mes "[Snower]";
		mes "Je ne vois aucun cadeau dans des boîtes luxueuses sur vous...";
		if(countitem(644))mes "Vous avez seulement des boîtes grises.";
		close;
	}
	delitem 664,countitem(664);
	delitem 665,countitem(665);
	delitem 666,countitem(666);
	delitem 667,countitem(667);
	getitem 644,@prizes;

	mes "[Snower]";
	mes "*mêler* Voilà.";
	next;

M_QUEST:
	mes "[Snower]";
	mes "J'espère que vous trouverez mes sœurs rapidement... Elles ont besoin d'aide.";
	if (Q_XMSRINGS == 0) set Q_XMSRINGS,1;
	close;

OnInit:
//Santa's Hat
	setitemscript 2236,"{ bonus bMdef,1; bonus bLuk,1; if(isequipped(2636,2637)){if(@xmr == gettime(DT_MINUTE))end; set @xmr,gettime(DT_MINUTE); misceffect EF_RAINBOW; end;} if(isequipped(2636)){if(@xmr == gettime(DT_MINUTE))end; set @xmr,gettime(DT_MINUTE); misceffect EF_SPHERE;} if(isequipped(2637)){if(@xmr == gettime(DT_MINUTE))end; set @xmr,gettime(DT_MINUTE); misceffect EF_ANGEL2;}}";
//Gold Xmas Ring
	setitemscript 2636,"{ bonus bLoseSPWhenUnequip,30; if(isequipped(2236)==0)end; if(getskilllv(\"AL_HEAL\")){skill \"TF_HIDING\",4+isequipped(2637);}else{skill \"AL_HEAL\",1+4*isequipped(2637);} }";
//Silver Xmas Ring
	setitemscript 2637,"{ bonus bDamageWhenUnequip,40; if(isequipped(2236)==0)end; if(getskilllv(\"AL_TELEPORT\")){skill \"MC_IDENTIFY\",1;}else{skill \"AL_TELEPORT\",1+isequipped(2636);} }";
	end;
}
