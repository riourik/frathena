//============================================================
//= Main Refiner Function
//============================================================
//= To allow auto safe refining/multiple refining set the
//= second argument to '1' in the function call.
//= If you enable this function, be sure to edit the value of
//= .@safe to the max safe refine in refine.yml as well.
//============================================================
function	script	refinemain	{
	disable_items;
	.@npc_name$ = getarg(0);
	.@features = getarg(1);
	mes "["+ .@npc_name$ +"]";
	mes "Je suis l'Armsmith.";
	mes "Je peux affiner tous types d'armes, armures et équipements, alors dis-moi ce que tu veux que je fasse affiner.";
	mes "Dis-moi ce que tu veux que je fasse affiner.";
	next;

	setarray .@indices[1], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_L, EQI_HAND_R, EQI_GARMENT, EQI_SHOES, EQI_ACC_L, EQI_ACC_R, EQI_HEAD_MID, EQI_HEAD_LOW;
	for(.@i = 1; .@i<getarraysize(.@indices); ++.@i) {
		if(getequipisequiped(.@indices[.@i])) {
			.@menu$ = .@menu$ + F_getpositionname(.@indices[.@i]) + "-[" + getequipname(.@indices[.@i]) + "]";
			.@equipped = 1;
		}
		.@menu$ = .@menu$ + ":";
	}
	if (.@equipped == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "Je ne pense pas pouvoir affiner les objets que vous avez...";
		close;
	}
	.@part = .@indices[select(.@menu$)];

	if(!getequipisequiped(.@part)) { //custom check
		mes "["+ .@npc_name$ +"]";
		mes "Tu ne portes pas";
		mes "quoi que ce soit là-bas qui";
		mes "que je puisse affiner.";
		emotion ET_FRET;
		close;
	}
	//Check if the item is refinable...
	if(!getequipisenableref(.@part)) {
		mes "["+ .@npc_name$ +"]";
		mes "je ne pense pas pouvoir";
		mes "affiner cet objet du tout...";
		close;
	}
	//Check to see if the items is already +10
	if(getequiprefinerycnt(.@part) >= 10) {
		mes "["+ .@npc_name$ +"]";
		mes "Je ne peux plus affiner";
		mes "cela. C'est aussi";
		mes "refiné que cela peut l'être !";
		close;
	}
	.@refineitemid = getequipid(.@part); // save id of the item
	.@refinerycnt = getequiprefinerycnt(.@part); //save refinery count
	setarray .@card[0], getequipcardid(.@part,0), getequipcardid(.@part,1), getequipcardid(.@part,2), getequipcardid(.@part,3);
	.@price = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_ZENY_COST);
	.@material = getequiprefinecost(.@part, REFINE_COST_NORMAL, REFINE_MATERIAL_ID);
	.@itemtype = getiteminfo( .@refineitemid, ITEMINFO_TYPE );

	if( .@itemtype == IT_ARMOR ){
		.@equip_lv = getequiparmorlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 10;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else if( .@itemtype == IT_WEAPON ){
		.@equip_lv = getequipweaponlv( .@part );

		switch( .@equip_lv ){
			case 1:
				.@safe = 7;
				break;
			case 2:
				.@safe = 6;
				break;
			case 3:
				.@safe = 5;
				break;
			case 4:
				.@safe = 4;
				break;
			default:
				// TODO:
				close;
		}

		// If the VIP system is enabled, the prices for non-VIP players are considerably higher.
		if( VIP_SCRIPT && !vip_status(VIP_STATUS_ACTIVE) ){
			switch( .@equip_lv ){
				case 1:
					.@price = .@price * 40;
					break;
				case 2:
					.@price = .@price * 50;
					break;
				case 3:
					.@price = .@price * 2;
					break;
				case 4:
					.@price = .@price * 2;
					break;
				default:
					// TODO:
					close;
			}
		}
	}else{
		.@safe = 4;
	}

	// TODO: What was this? Maybe shadow?
	// case 5: .@price = .@price * 10; break;

	if(.@features != 1) {
		mes "["+ .@npc_name$ +"]";
		mes "Pour affiner ceci, j'ai besoin de";
		mes "one ^003366"+getitemname(.@material)+"^000000 and";
		mes "a service fee of " + .@price + " Zeny.";
		mes "Voulez-vous vraiment continuer ?";
		next;
		if(select("Yes:No") == 2){
			mes "["+ .@npc_name$ +"]";
			mes "Ouais...";
			mes "Il n'y a pas besoin d'";
			mes "se presser. Prenez votre temps.";
			close;
		}
		if(getequippercentrefinery(.@part) < 100) {
			mes "["+ .@npc_name$ +"]";
			mes "Oh non ! Si je continue à";
			mes "affiner cela, il y a un risque qu'il soit";
			switch(.@material) {
			case 985:
				mes "détruit ! Cela signifie que ^FF0000cet équipement^000000, et ^FF0000toutes les cartes^000000 ou propriétés spéciales ajoutées à cette armure, ^FF0000seront perdues^000000.";
				break;
			default:
				mes "sera détruit, et vous perdriez ^FF0000l'arme^000000, toutes les ^FF0000cartes dans l'arme^000000";
				mes ", ou toute propriété spéciale ajoutée.";
				break;
			}
			next;
			mes "["+getarg(0)+"]";
			mes "Je ne peux pas être plus clair.";
			mes "Une fois qu'une arme est détruite,";
			mes "elle ne peut pas revenir.";
			mes "Vous avez vraiment une chance de";
			mes "^FF0000perdre cette arme^000000 pour toujours.";
			mes "Voulez-vous vraiment affiner ?";
			next;
			if(select("Yes:No") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "Je suis tout à fait d'accord...";
				mes "Même si je suis un grand affineur, parfois j'ai des erreurs.";
				close;
			}
		}
		if((countitem(.@material) < 1) || (Zeny < .@price)) {
			mes "["+ .@npc_name$ +"]";
			mes "Vous ne semblez pas avoir de";
			mes "enough Zeny or "+getitemname(.@material)+"...";
			mes "Va chercher plus. Je serai";
			mes "ici toute la journée si vous avez besoin de moi.";
			close;
		}
		Zeny = Zeny-.@price;
		delitem .@material,1;

		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
		    callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt)) {
			mes "["+ .@npc_name$ +"]";
			emotion ET_FRET;
			mes "Attends une seconde...";
			mes "Croyez-vous que je suis stupide ?!";
			mes "Vous avez échangé l'objet quand je ne regardais pas ! Fichez le camp d'ici !";
			close;
		}

		if(getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			mes "["+ .@npc_name$ +"]";
			emotion (!rand(5))?ET_MONEY:ET_HUK;
			.@lose = rand(1,3);
			if (.@lose == 1) {
				mes "OH ! MON DIEU !";
				mes "Merde ! Encore une fois !";
				mes "Je suis désolé, mais tu sais que l'entraînement rend parfait.";
				mes "Um, non ? Heh heh...";
			} else if(.@lose == 2) {
				mes "Nooooooo!";
				mes "C'est cassé!";
				mes "M-Merci de m'avoir prévenu !";
			} else {
				mes "Merde!";
				mes "Il ne pouvait pas supporter";
				mes "beaucoup plus d'affinage !";
				mes "Je suis désolé pour ça...";
			}
			close;
		}
		mes "["+getarg(0)+"]";
		successrefitem .@part;
		emotion ET_SMILE;
		.@win = rand(1,3);
		if (.@win == 1) {
			mes "Parfait !";
			mes "^FF0000Hé hé!";
			mes "Encore une fois,";
			mes "^FF0000travail parfait";
			mes "^FF0000de la maêtre~";
		} else if(.@win == 2) {
			mes "^FF0000Succès...!";
			mes "^FF0000encore une fois, mon incroyable";
			mes "^FF0000talent impressionne vraiment";
			mes "^FF0000et brille aujourd'hui.";
		} else {
			mes "^FF0000Hé hé!";
			mes "^FF0000Je suis terminé.";
			mes "^FF0000Sans aucun doute, mon travail est";
			mes "^FF0000au niveau de vos attentes.";
			mes "^FF0000Pure perfection~";
		}
		close;
	}

// New Refining Functions ========================
	if (.@refinerycnt < .@safe) {
		mes "["+ .@npc_name$ +"]";
		mes "^FF0000Je peux raffiner cet objet jusquà la limite sécuritaire ou le nombre de fois que vous souhaitez. C'est votre choix.";
		next;
		.@menu2 = select("To the safe limit, please.","I'll decide how many times.","I've changed my mind...");
	} else
		.@menu2 = 2;
	switch(.@menu2){
	case 1: 
		.@refinecnt = .@safe - .@refinerycnt;
		break;
	case 2:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "^FF0000Combien de fois voudriez-vous que je raffine votre objet ?";
		next;
		input .@refinecnt;
		.@refinecheck = .@refinecnt + .@refinerycnt;
		if (.@refinecnt < 1 || .@refinecheck > 10) {
			mes "["+ .@npc_name$ +"]";
			mes "^FF0000Je ne peux pas raffiner cet objet autant de fois.";
			close;
		}
		if(.@refinecheck > .@safe) {
			.@refinecheck = .@refinecheck - .@safe;
			mes "["+ .@npc_name$ +"]";
			mes "This will try to refine the equipment " + .@refinecheck + " times past the safe limit. Your equipment may be destroyed... is that ok?";
			next;
			if(select("Yes...","No...") == 2){
				mes "["+ .@npc_name$ +"]";
				mes "^FF0000Vous l'avez dit... Comme vous voulez.";
				close;
			}
		}
		break;
	case 3:
		next;
		mes "["+ .@npc_name$ +"]";
		mes "^FF0000Vous l'avez dit... Comme vous voulez.";
		close;
	}
	.@fullprice = .@price * .@refinecnt;
	mes "["+ .@npc_name$ +"]";
	mes "That will cost you " + .@refinecnt + " " + getitemname(.@material) + " and " + .@fullprice + " Zeny. Is that ok?";
	next;
	if(select("Yes","No...") == 2){
		mes "["+ .@npc_name$ +"]";
		mes "^FF0000Vous l'avez dit... Comme vous voulez.";
		close;
	}
	if(countitem(.@material) < .@refinecnt || Zeny < .@fullprice) {
		mes "["+ .@npc_name$ +"]";
		mes "^FF0000C'est tout ce que vous avez ? Malheureusement, je ne peux pas travailler pour vous à un prix plus bas. Essayez de mettre-vous à ma place.";
		close;
	}
	Zeny = Zeny - .@fullprice;
	delitem .@material,.@refinecnt;
	while(.@refinecnt){
		if (getequipisequiped(.@part) == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "^FF0000Regardez ici... Vous n'avez aucun objet sur...";
			close;
		}
		// anti-hack
		if (callfunc("F_IsEquipIDHack", .@part, .@refineitemid) || callfunc("F_IsEquipCardHack", .@part, .@card[0], .@card[1], .@card[2], .@card[3]) ||
				callfunc("F_IsEquipRefineHack", .@part, .@refinerycnt) || (.@menu2 == 1 && getequippercentrefinery(.@part) < 100)) {
			mes "["+ .@npc_name$ +"]";
			mes "^FF0000Clang... Non, mais avez-vous imaginé que je pourrais être aussi stupide?!";
			mes "Vous l'avez changé...";
			mes "Sortez d'ici avant que je ne vous assomme avec mon Marteau!!";
			close;
		} 
		mes "Clang, clang!!!";
		if(.@menu2 == 2 && getequippercentrefinery(.@part) <= rand(100)) {
			failedrefitem .@part;
			emotion ET_HUK;
			mes "["+ .@npc_name$ +"]";
			mes "WAHHHH!!! Je suis tellement désolé... J'avais prévenu que cela pourrait arriver...";
			.@refinecnt = .@refinecnt - 1;
			if(.@refinecnt == 0) close;
			mes "Voici les pièces de Zeny et les matériaux non utilisés...";
			getitem .@material,.@refinecnt;
			.@fullprice = .@refinecnt * .@price;
			Zeny = Zeny + .@fullprice;
			close;
		}
		successrefitem .@part;
		emotion ET_BEST;
		.@refinecnt = .@refinecnt - 1;
		.@refinerycnt = getequiprefinerycnt(.@part);
		next;
	}
	mes "["+ .@npc_name$ +"]";
	mes "Tout est terminé... Revenez bientôt.";
	close;
}

	mes "Si vous ne voulez rien, veuillez entrer le nombre '0'.";
	next;
	while(1) {
		input .@input;
		if (.@input == 0) {
			mes "["+ .@npc_name$ +"]";
			mes "La transaction a";
			mes "été annulée.";
			close;
		}
		else if (.@input < 0 || .@input > 500) {
			mes "["+ .@npc_name$ +"]";
			mes "D'accord, vous pouvez";
			mes "acheter jusqu'à 500.";
			mes "^FF0000Rien de plus.";
			mes "^FF0000C'est bien compris? D'accord.";
			next;
		}
		else {
			break;
		}
	}
	.@sell = .@input * .@price;
	if (Zeny < .@sell) {
		mes "["+ .@npc_name$ +"]";
		mes "Euh...";
		mes "Tu n'as pas";
		mes "assez de Zeny pour acheter";
		mes ""+ .@input +" of them.";
		close;
	}
	if (checkweight(.@material,.@input) == 0) {
		mes "["+ .@npc_name$ +"]";
		mes "Hmm...";
		mes "Je ne peux rien vous donner si vous n'avez pas assez d'espace dans votre inventaire. Pourquoi ne mettez-vous pas vos objets supplémentaires en dépositoire Kafra et réessayez?";
		close;
	}
	getitem .@material,.@input;
	Zeny = Zeny-.@sell;
	mes "["+ .@npc_name$ +"]";
	mes "^FF0000Tenez!";
	mes "Merci pour";
	mes "^FF0000votre parrainage.";
	close;
}

// Equipment Repair Function
//============================================================
function	script	repairmain	{
	.@repairprice = 5000;
	.@npc_name$ = getarg(0);
	mes "["+ .@npc_name$ +"]";
	mes "Salut!";
	mes "Veux-tu que je";
	mes "répare tes objets ?";
	mes "Sur quoi tu peux compter pour les réparations !";
	mes "d'objets !";
	next;
	switch(select("Actually, I do have some items...:None at the moment.")) {
	case 1:
		.@checkitem = 1;
		while (1) {
			if (getbrokenid(.@checkitem) == 0) {
				break;
			}
			.@checkitem = .@checkitem+1;
		}
		.@checkitem = .@checkitem-1;
		if (!.@checkitem) {
			mes "["+ .@npc_name$ +"]";
			mes "Oh wow, c'est incroyable !";
			mes "Tu dois prendre soin de tes affaires. Aucun de tes objets n'est endommagé !";
			next;
			mes "["+ .@npc_name$ +"]";
			mes "Si tout le monde est comme toi, je vais être sans emploi !! Haha~!";
			close;
		}
		mes "["+ .@npc_name$ +"]";
		mes "Hmm...";
		mes "Voyons...";
		mes "Parmi tous tes objets,";
		mes "" + .@checkitem + " are damaged.";
		mes "Veux-tu les réparer ?";
		next;
		.@totalcost = .@repairprice*.@checkitem;
		mes "["+ .@npc_name$ +"]";
		mes "Each repair costs " + .@repairprice + " Zeny. So to repair all your damaged items would cost " + .@totalcost + " Zeny! Would you like to repair the items?";
		next;
		switch(select("Yes:No")) {
		case 1:
			if (Zeny < .@totalcost) {
				mes "["+ .@npc_name$ +"]";
				mes "Calme-toi...";
				mes "Vérifie ton portefeuille avant de recevoir la note de réparation ! Je ne peux rien réparer car tu n'as pas assez de Zeny.";
				close;
			}
			.@checkitem2 = 1;
			while (1) {
				if (getbrokenid(.@checkitem2) == 0) {
					break;
				}
				.@checkitem2 = .@checkitem2+1;
			}
			.@checkitem2 = .@checkitem2-1;
			if (.@checkitem == .@checkitem2) {
				Zeny = Zeny-.@totalcost;
				while (.@checkitem) {
					repair(.@checkitem);
					.@checkitem = .@checkitem-1;
				}
				mes "["+ .@npc_name$ +"]";
				mes "D'accord ! C'est fait. Maintenant, essaye d'être un peu plus prudent. Les objets ont aussi une vie, tu sais.";
				close;
			}
			else {
				mes "["+ .@npc_name$ +"]";
				mes "Hum ? Il y a un problème. Attends... Équipe les objets que tu veux réparer et reviens vers moi.";
				close;
			}
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "Bien, ce n'est pas mon affaire, mais il ne faut pas laisser des objets endommagés. Tu devrais les faire réparer dès que possible !";
			close;
		}
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "Hohoho...";
		mes "Tu n'as pas";
		mes "une affaire avec moi ?";
		mes "^FF0000si vous n'en avez aucun";
		mes "^FF0000articles à réparer.";
		close;
	}
}
