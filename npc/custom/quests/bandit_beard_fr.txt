umbala,126,129,4	script	Bearded Man#bandit	120,{
	if(BEARD_QUEST == 4 ) goto L_MAKE;
	if(BEARD_QUEST >= 5 ) goto L_DONE;
	if(BEARD_QUEST >= 1 ) goto L_THREAD;

	mes "[Homme barbu]";
	mes "Barbes ! Achetez vos merveilleuses barbes !";
	mes "Voudriez-vous une barbe ?";
	next;
	menu "Yes, I'd like some facial hair!",-,"No thanks, thats gross!",M_NO;

	mes "[Homme barbu]";
	mes "HoHo~ Vous voulez des poils de barbe.";
	emotion ET_SMILE;
	mes "Eh bien, je vous donnerais mes barbes mais elles ne sont pas bon marché.";
	mes "En fait, j'en ai encore une...";
	mes "mais je ne la vends pas.";
	mes "Mais n'allez pas en colère ! Je peux vous en faire une.";
	next;
	set BEARD_QUEST,1;
L_THREAD:
	mes "[Homme barbu]";
	mes "J'aurais besoin de ^8080FFdu fil^000000 pour la barbe.";
	mes "Parlez à mon ami à Izlude pour obtenir du ^8080FFfil^000000.";
	next;

	mes "[Homme barbu]";
	mes "Je vous demanderais aussi d'obtenir certaines choses pour la barbe.";
	mes "Ce sont :";
L_LIST:
	mes "^8080FFBande élastique^000000";
	mes "^8080FF5 Peaux d'animaux^000000";
	mes "^8080FFSubstance de teinture noire^000000";
	mes "^8080FFFaux barbe grand-père^000000";
	mes "^8080FF100 Morve collante^000000";
	mes "^8080FFContre-agent^000000";
	mes " et un ^8080FFDiamant de deux carats^000000";
	close;

M_NO:
	mes "[Homme barbu]";
	mes "Bien, alors comme ça!";
	emotion ET_ANGER;
	close;

L_DONE:
	mes "[Homme barbu]";
	mes "Excusez-moi, je ne peux en faire qu'un par personne.";
	mes "Nous ne voulons pas trop de barbes, n'est-ce pas?";
	emotion ET_KIK;
	close;

L_MAKE:
	mes "[Homme barbu]";
	mes "Allons-y, faisons votre barbe!";
	next;
	if(countitem(7200) < 1 || countitem(919) < 1 ||	countitem(983) < 1 || countitem(2241) < 1 ||
	countitem(938) < 100 || countitem(973) < 1 || countitem(731) < 1) goto L_NOITEMS;
	delitem 7200, 1;
	delitem 919, 1;
	delitem 983, 1;
	delitem 2241, 1;
	delitem 938, 100;
	delitem 973, 1;
	delitem 731, 1;
	mes "^8080FF~Vous le voyez couper et coudre la barbe ensemble~^000000";
	next;
	mes "^8080FF~Il vous tend la barbe terminée~^000000";
	next;
	getitem 2237, 1;
	set BEARD_QUEST, 5;
	mes "[Homme barbu]";
	mes "Bonne journée!";
	emotion ET_KIK;
	close;

L_NOITEMS:
	mes "[Homme barbu]";
	mes "Nom d'un chien! Je ne peux pas faire une barbe sans les objets!";
	mes "Voici la liste à nouveau:";
	emotion ET_HUK;
	goto L_LIST;
}

	next;

	mes "[Marchand de Fil]";
