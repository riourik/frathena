prontera,98,121,4	script	Acolyte#Prtclear	90,{
	mes "[Keiki]";
	mes "Bonjour, aventurier.";
	mes "J'ai étudié la magie partout dans Rune-Midgarts pour améliorer ce que je crois être l'une des plus grandes compétences disponibles pour la classe d'acolyte.";
	next;

	if (is_function("VIP_iRO_Acolyte"))
		callfunc "VIP_iRO_Acolyte","[Keiki]","Juno";

	mes "[Keiki]";
	mes "Je suis le seul et unique Acolyte à avoir atteint le niveau 10 de la compétence Warp Portal !";
	emotion ET_THINK, playerattached();
	next;
	mes "[Keiki]";
	mes "C'est exact! Et...";
	mes "Je vous promets que je n'oublie pas les lieux que j'ai déjà mémorisés.";
	mes "Un jour, j'améliorerai mes compétences pour me téléporter là où je veux ~";
	next;
	mes "[Keiki]";
	mes "Je suis prêt à vous téléporter vers les nombreux endroits que j'ai mémorisés pour une somme modique.";
	mes "Souhaitez-vous utiliser ce service ?";
	next;
	switch(select("Yes:No")) {
	case 1:
		mes "[Keiki]";
		mes "Où aimerais-tu aller ?";
		mes "Je vous souhaite bonne chance dans votre voyage.";
		next;

		setarray .@towns$[0], "Izlude", "Geffen", "Payon", "Morocc", "Alberta", "Al De Baran", "Comodo", "Umbala", "Juno";
		setarray .@cost[0],        600,     1200,    1200,     1200,      1800,          2200,     2200,     2200,   1800;

		set .@size, getarraysize(.@towns$);
		for(set .@i,0; .@i<.@size; set .@i,.@i+1)
			set .@menu$, .@menu$+.@towns$[.@i]+" -> "+.@cost[.@i]+"z:";
		set .@i, select(.@menu$+"Cancel")-1;
		if (.@i == .@size)
			close;
		if (Zeny < .@cost[.@i]) {
			mes "[Keiki]";
			mes "Je suis désolé, mais tu n'as pas";
			mes "assez zeny pour la téléportation";
			mes "Service. Les frais de téléportation";
			mes "to "+.@towns$[.@i]+" is "+.@cost[.@i]+" zeny.";
			close;
		}
		specialeffect2 EF_READYPORTAL;
		specialeffect2 EF_TELEPORTATION;
		specialeffect2 EF_PORTAL;
		next;
		set Zeny, Zeny - .@cost[.@i];
		switch(.@i) {
			case 0: if (checkre(0)) warp "izlude",128,98; else warp "izlude",91,105; break;
			case 1: warp "geffen",120,39; break;
			case 2: warp "payon",161,58; break;
			case 3: warp "morocc",156,46; break;
			case 4: warp "alberta",117,56; break;
			case 5: warp "aldebaran",168,112; break;
			case 6: warp "comodo",209,143; break;
			case 7: warp "umbala",100,154; break;
			case 8: warp "yuno",158,125; break;
		}
		close;
	case 2:
		close;
	}
}
