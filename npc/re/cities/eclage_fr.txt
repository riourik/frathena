ecl_in02,164,56,3	script	Receptionist#Laphine	835,{
	mes "[Réceptionniste]";
	mes "Bienvenue à la Clinique d'Eclage.";
	next;
	switch (select("Save Location:Rest:Finish the conversation")) {
	case 1:
		mes "[Réceptionniste]";
		mes "La clinique a été définie comme votre point de sauvegarde. Faites attention toutefois~";
		savepoint "ecl_in02",162,50,1,1;
		close;
	case 2:
		mes "[Réceptionniste]";
		mes "Récupérez vos forces mentales et physiques dans notre clinique. Elle est remplie de l'énergie pure d'Eclage.";
		mes "Nous facturons normalement 1 Splendide Coin, mais nous offrons une réduction spéciale aux voyageurs. Votre total s'élève à 5 000 zeny.";
		next;
		switch (select("Pay with Splendide Coin:Pay with Zeny:Forget it")) {
		case 1:
			if (!countitem(6081)) {
				mes "[Réceptionniste]";
				mes "Il semble que vous n'ayez pas assez de pièces. Pourriez-vous vérifier votre compte encore une fois, s'il vous plaît ?";
				close;
			}
			mes "[Réceptionniste]";
			mes "Faites comme chez vous~";
			close2;
			delitem 6081,1; //6081
			percentheal 100,100;
			warp "ecl_in02",167,49;
			end;
		case 2:
			if (Zeny < 5000) {
				mes "[Réceptionniste]";
				mes "Il semble que vous n'ayez pas assez de ressources pour continuer. Avez-vous l'esprit tranquille si nous vous demandons de vérifier votre compte à nouveau ?";
				close;
			}
			mes "[Réceptionniste]";
			mes "Faites comme chez vous~";
			close2;
			Zeny = Zeny - 5000;
			percentheal 100,100;
			warp "ecl_in02",167,49;
			end;
		case 3:
			close;
		}
	case 3:
		close;
	}
}
