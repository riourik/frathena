valkyrie,53,42,3	script	Champion#Valkyrie	52,{
	if (ADVJOB == 0 || Upper != 1) {
		set .@karma_d,rand(1,10);
		if (.@karma_d > 4) {
			mes "[Champion]";
			mes "Félicitations.";
			mes "Honneur aux guerriers !";
			close;
		}
		mes "[Champion]";
		mes "^FF0000Dextérité.^_";
		mes "^FF0000Vitesse.^_";
		mes "^FF0000Force.^_";
		mes "^FF0000Agilité.^_";
		next;
		mes "[Champion]";
		mes "Un Champion peut";
		mes "bénéficier de tout cela.";
		mes "Mais on ne peut maîtriser";
		mes "que tant de choses dans la vie...";
		mes "[Champion]";
		close;
	}
	else if (ADVJOB == Job_Champion && Class == Job_Acolyte_High && JobLevel > 39) {
		mes "[Champion]";
		mes "L'heure pour les grands héros";
		mes "de se lever face aux forces du mal qui";
		mes "nuisent au monde de Midgard !";
		mes "[Champion]";
		mes "devenir un Champion ?";
		next;
		mes "[Champion]";
		mes "Aimeriez-vous";
		mes "to become a Champion?";
		next;
		if (select("No.:Yes.") == 1) {
			mes "[Champion]";
			mes "[Grand Prêtre]";
			mes "n'hésitez pas à revenir.";
			mes "Honneur aux guerriers !";
			close;
		}
		if (SkillPoint) {
			mes "[Champion]";
			mes "Il est encore possible pour vous d'apprendre des compétences supplémentaires. Veuillez utiliser";
			mes "tous vos points de compétence restants avant de revenir me voir.";
			close;
		}
		jobchange Job_Champion;
		set ADVJOB,0;
		mes "[Champion]";
		mes "Félicitations!";
		mes "Vivre comme un Champion,";
		mes "et apporter de la lumière dans";
		mes "le monde grâce à la";
		mes "force de vos poings.";
		close;
	}
	mes "[Champion]";
	mes "Accueillir";
	mes "vers Valhalla,";
	mes "la Salle de l'Honneur.";
	next;
	mes "[Champion]";
	mes "S'il vous plaît faites-vous";
	mes "confortable";
	mes "pendant votre séjour ici.";
	mes "Honneur aux guerriers !";
	close;
}
