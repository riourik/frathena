	mes "J'ai 2 objets qui pourraient devenir utiles pour vous.";
	next;
	menu "Realgar Wine",-, "Exorcize Herb",M_HERB;

	mes "Combien en voulez-vous ?";
	mes "Vous ne pouvez acheter qu'un maximum de 5.";
	mes "Chacun d'eux coûte 20000z.";

	getitem 682,@input;

	mes "Voilà, j'espère que vous réussirez dans ma quête.";
	close;
  
L_SORRY:

	mes "Vous ne devez pas être avare, l'honnêteté est le chemin de Dieu.";

	mes "^FF0000Le chemin d'anéantir complètement le mal est long, aidez-moi sur la voie de Dieu.^000000";
	close;
  
M_HERB:
	mes "[Exorcist Master Fahae]";
	mes "Combien en voulez-vous ?";
	mes "Vous ne pouvez acheter qu'un maximum de 5.";
	mes "Chacun coûte 10000z.";
	mes "(Tapez 0 pour annuler)";
	next;
	input @input;
	if(@input==0) close;
	if(@input>5) goto L_SORRY;
	if(Zeny< @input*10000) goto L_NOZENY;
	set Zeny,Zeny-10000*@input;
	getitem 683,@input;
	mes "[Exorcist Master Fahae]";
	mes "Voilà, j'espère que vous réussirez dans ma quête.";
	close;

L_NOZENY:
	mes "[Exorcist Master Fahae]";
	mes "L'argent ne rend pas heureux tout le monde, mais nous en avons besoin pour soutenir le temple et moi-même. S'il vous plaît, tuez quelques monstres et prenez leurs drops.";
	close;
}
