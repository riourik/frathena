-	script	Mercenary Merchant#dummy::MercMerchant	-1,{
	mes "[Marchand de marchandises mercenaires]";
	mes "Bonjour, je vends des biens";
	mes "que les mercenaires peuvent";
	mes "utiliser. Y a-t-il quelque chose";
	mes "en particulier que";
	mes "tu cherches ?";
	next;
	setarray .@item, 12184,	12185,	12241,	12242,	12243;
	setarray .@cost, 2500,	5000,	800,	1500,	3000;
	for (.@size = getarraysize(.@item); .@i < .@size; .@i++)
		.@menu$ = .@menu$ + getitemname(.@item[.@i]) + ":";
	.@m = select(.@menu$)-1;
	.@str$ = .@cost[.@m] + "";
	.@len = getstrlen(.@str$);
	.@str_cost$ = (.@len <= 3)? .@str$ : insertchar(.@str$,",",.@len-3);
	mes "[Marchand de marchandises mercenaires]";
	mes getitemname(.@item[.@m]);
	mes "each cost " + .@str_cost$ + " zeny.";
	mes "Combien en voudriez-vous ?";
	next;
	input .@input;
	if (!.@input) {
		mes "[Marchand de marchandises mercenaires]";
		mes "Vous avez changé d'avis ?";
		mes "Très bien, n'hésitez pas à venir";
		mes "reviens vers moi quand tu veux";
		mes "pour acheter des potions de mercenaire.";
		close;
	}
	else if (.@input < 0 || .@input > 10000) {
		mes "[Marchand de marchandises mercenaires]";
		mes "Je suis désolé, mais toi";
		mes "ne peut acheter que jusqu'à";
		mes "10 000 de ces potions";
		mes "à la fois. Veuillez entrer";
		mes "un nombre de 1 à 10 000.";
		close;
	}
	else if (set(.@total_po, .@input * .@cost[.@m]) > Zeny) {
		mes "[Marchand de marchandises mercenaires]";
		mes "Je suis désolé, mais tu ne le fais pas";
		mes "j'ai assez de zeny pour ça";
		mes "de nombreuses potions. Eh bien, je serai";
		mes "sois là quand tu seras prêt";
		mes "acheter quelque chose";
		mes "pour vos mercenaires.";
		close;
	}
	else if (!checkweight(.@item[.@m], .@input)) {
		mes "[Marchand de marchandises mercenaires]";
		mes "Si je t'en ai donné autant";
		mes "potions, tu ne le serais pas";
		mes "pouvoir les emporter avec vous.";
		mes "S'il te plaît, reviens après";
		mes "tu libères de l'espace";
		mes "dans votre inventaire.";
		close;
	}
	mes "[Marchand de marchandises mercenaires]";
	if (.@input == 1) 
		mes "Here's your " + getitemname(.@item[.@m]) + ".";
	else {
		mes "Voilà, c'est exactement";
		mes callfunc("F_InsertPlural",.@input,getitemname(.@item[.@m])) + ".";
	}
	mes "Merci et s'il vous plaît venez";
	mes "encore une fois quand tu as besoin de plus";
	mes "potions pour vos mercenaires.";
	Zeny = Zeny - .@total_po;
	getitem .@item[.@m], .@input;
	close;
}

-	script	Mercenary_Switch	-1,{

	setarray .@name$, "Spear","Sword","Bow";
	.@npc$ = strnpcinfo(2);
	for (.@size = getarraysize(.@name$); .@type < .@size; .@type++)
		if (.@npc$ == .@name$[.@type]) break;

	mes "[Vérificateur]";
	mes "Veuillez saisir un mot de passe.";
	next;
	.@i = callfunc("F_GM_NPC",1854,0,0,10000);
	mes "[Vérificateur]";
	if (.@i == -2)
		mes "Erreur.";
	else if (.@i == 0)
		mes "Mauvais numéro";
	else if (.@i == 1) {
		mes "Veuillez sélectionner.";
		next;
		switch( select( "Turn off Mercenary NPC","Turn on Mercenary NPC" ) ) {
		case 1:
			mes "Les PNJ sont désactivés.";
			disablenpc "Mercenary Manager#" + .@name$[.@type];
			disablenpc "Mercenary Merchant#" + .@name$[.@type];
			break;
		case 2:
			mes "Les PNJ sont activés.";
			enablenpc "Mercenary Manager#" + .@name$[.@type];
			enablenpc "Mercenary Merchant#" + .@name$[.@type];
			break;
		}
	}
	close;
}
