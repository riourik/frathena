			mes "Party: " + .@p_name$ + "";
			mes "Leader: " + .@p_leader$ + "";
			mes "^0000ff" + .@md_name$ + " ^000000- Unknown error";
			close;

		mes "^ff0000Veuillez patienter.^000000";
		close;
	case 2:

			mes "Téléportation...";
			next;
			mes "[Machine]";
			mes "^ff0000Cette machine ne peut être accessible pendant 20 heures après votre dernière utilisation.^000000";
			next;

			close;
		case IE_NOMEMBER:
			mes "[Machine]";
			mes "^ff0000Seulement un membre du groupe peut entrer dans le Donjon Commémoratif.^000000";
			close;
		case IE_OK:
			setquest 13185;
			specialeffect2 EF_READYPORTAL2;
			specialeffect2 EF_PORTAL2;
