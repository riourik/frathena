L_displaycutin:
	.@arg = getarg(0);
	if (.@arg < 4) cutin "Ʈī",4;		// Ghostring Card
	else if (.@arg < 7) cutin "ī",4;	// Drops Card
	else if (.@arg < 10) cutin "Ƽī",4;	// Mantis Card
	else if (.@arg < 13) cutin "ī",4;		// Munak Card
	else if (.@arg < 16) cutin "Ÿī",4;	// Santa Poring Card
	else if (.@arg < 19) cutin "ī",4;// Soldier Skeleton Card
	else if (.@arg < 22) cutin "̷ī",4;	// Skeleton Card
	else if (.@arg < 25) cutin "尡ī",4;	// Eddga Card
	else if (.@arg < 28) cutin "ī",4;	// Angeling Card
	else if (.@arg < 31) cutin "ī",4;		// Poring Card
	else cutin "ī",4;// Poporing Card
	return;
}
