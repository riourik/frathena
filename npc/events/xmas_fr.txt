xmas_in,100,96,4	script	Father Christmas::Santa2	718,{
	mes "[Le père Noël]";
	if(xmas_npc==0) set xmas_npc, 1; // For Lutie & Bard quest
	if(#event_xmas > 0 && #event_xmas < 30 ) goto L_Start;
	mes "Joyeux noël!";
	if(Class==0 || #event_xmas>=30 ) close; //anti exploit protection
	mes "^FF0000J'ai un cadeau pour toi ! Ho Ho Ho !";
	getitem rand(664,667),1; //gives one of 4 gift boxes
	set #event_xmas,#event_xmas+1;
	close;

L_Start:
	mes "^FF0000Je rencontre un petit problème...";
	mes "^FF0000Veux-tu m'écouter ?";
	next;
	menu "Listen to Santa Claus.",M_0, "Give Santa Claus proof.",M_1, "Cancel.",M_End;

	M_0:
		mes "[Le père Noël]";
		mes "^FF0000Mon problème est le suivant.";
		mes "^FF0000Il semble qu'il y ait un homme là-bas";
		mes "^FF0000qui me fait passer pour un imposteur et qui crée";
		mes "^FF0000la terreur dans tout le pays.";
		next;
		mes "[Le père Noël]";
		mes "^FF0000Comme le Grinch de la légende, il prend";
		mes "^FF0000tous les jouets des enfants et s'en sert pour lui-même.";
		mes "^FF0000Je suis trop occupé ici à créer mon lot de";
		next;
		mes "[Le père Noël]";
		mes "^FF0000jouets pour l'année prochaine, donc je ne peux pas";
		mes "^FF0000aller le chercher moi-même.";
		mes "^FF0000Alors je voudrais que tu ailles et que tu";
		mes "^FF0000Destruction de cet homme pour moi.";
		mes "^FF0000Il possède l'un de mes sacs magiques, cependant, donc il";
		next;
		mes "[Le père Noël]";
		mes "^FF0000va s'échapper dans un autre endroit chaque fois que tu";
		mes "^FF0000l'attaques.";
		mes "escape into it to another place each";
		mes "time you attack him.";
		next;
		mes "[Le père Noël]";
		mes "^FF0000Cependant,";
		mes "^FF0000En se hâtant, il a tendance à laisser tomber des choses.";
		mes "^FF0000S'il laisse tomber un de ses Chaussettes";
		mes "^FF0000Avec des Tréfles qu'il utilise pour voler les jouets";
		mes "^FF0000des enfants pauvres, ramasse-le.";
		next;
		mes "[Le père Noël]";
		mes "^FF0000Si tu en collectes 3 de ces chaussettes, je te donnerai";
		mes "^FF0000une boîte mystère prototype que";
		mes "^FF0000j'ai gardée dans mon labo. Elle donne des cadeaux aléatoires";
		mes "^FF0000et me sauve beaucoup de travail.";
		mes "^FF0000Il semble que tu as fait du bon boulot en éliminant ces";
		close;
	M_1:
		mes "[Le père Noël]";
		if(countitem(7034) < 3) goto L_NotEnuf;
		delitem 7034,3;
		mes "^FF0000faux Pères Noël. Continue comme ça !";
		mes "^FF0000Voici ta récompense.";
		mes "^FF0000Si tu en obtiens 3 de plus, je te donnerai une autre boîte.";
		next;
		getitem 644,1; //Gift Box Prototype
		set #event_xmas,#event_xmas+1;
		mes "[Le père Noël]";
		mes "^FF0000J'espère que tu recevras un bon objet.";
		mes "^FF0000Tu n'as pas assez de chaussettes comme preuve.";
		mes "^FF0000Va éliminer ces Pères Noël malveillants";
		close;
	
		L_NotEnuf:
			mes "^FF0000et ramène-m'en plus et je te récompenserai.";
			mes "^FF0000Je vois. En tout cas, au moins nous nous reverrons le matin de Noël.";
			mes "and get more for me and I'll reward you.";
			close;
	M_End:
		mes "[Le père Noël]";
		mes "I see. Well, at the very least";
		mes "we shall meet again on Christmas morning.";
		close;

OnInit:
	disablenpc "Santa Claus";
	end;
}
