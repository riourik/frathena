payon,173,238,5	script	Jade#pay	754,{
	if (checkweight(1201,1) == 0) {
		mes "^3355FFAttendez une seconde ! En ce moment, vous transportez trop d'objets avec vous. Veuillez revenir après avoir déposé certains de vos objets dans le coffre-fort Kafra.^000000";
		close;
	}
	mes "[Jade]";
	mes "Apportez-moi deux";
	mes "Gemmes du";
	mes "même couleur, et je vais";
	mes "les changer en Gemmes";
	mes "d'une autre couleur.";
	next;
	switch(select("Blue Gemstones into Red ones!:Red Gemstones into Yellow ones!:Yellow Gemstones into Blue ones!")) {
	case 1: callsub S_TradeGems,717,716;
	case 2: callsub S_TradeGems,716,715;
	case 3: callsub S_TradeGems,715,717;
	}

S_TradeGems:
	.@item_req = getarg(0);
	.@item_id = getarg(1);

	if (countitem(.@item_req) < 2) {
		mes "[Jade]";
		mes "Hah...!";
		mes "Vous rigolez ?";
		mes "Je ne peux pas vous fournir ce service si vous ne";
		mes "me donnez au moins";
		mes "give me at least";
		mes "2 "+ getitemname(.@item_req) +"s!";
		close;
	}
	else {
		.@gems = countitem(.@item_req) /2;
		mes "[Jade]";
		mes "Je pense que je peux créer";
		mes "a total of "+ .@gems +" "+ getitemname(.@item_id) +"s";
		mes "using the "+ getitemname(.@item_req) +"s";
		mes "ce que vous avez actuellement.";
		mes "Que voulez-vous faire ?";
		next;
		switch(select("Give me as many as you can.:I want to set the amount.:I quit.")) {
		case 1:
			delitem .@item_req,.@gems * 2;
			getitem .@item_id,.@gems;
			break;
		case 2:
			mes "[Jade]";
			mes "Donc combien";
			mes "en voulez-vous ?";
			mes "Le nombre maximal";
			mes "que vous pouvez entrer est de 100.";
			next;
			while(1) {
				input .@input,0,101;
				if (.@input == 0) {
					mes "[Jade]";
					mes "Aucun du tout ?";
					mes "Je suppose que vous";
					mes "^FF0000...Vous avez changé d'avis...";
					close;
				}
				else if (.@input > 100) {
					mes "[Jade]";
					mes "Euh...";
					mes "^FF0000Je vous ai demandé d'entrer";
					mes "^FF0000un montant ne dépassant pas";
					mes "^FF0000100, si je me souviens bien...?";
					next;
				}
				else if (.@gems < .@input) {
					// Custom dialogue
					mes "[Jade]";
					mes "Euh...";
					mes "^FF0000Vous n'avez pas autant de gemmes à échanger...";
					mes "^FF0000Il est vrai que vous ne possédez pas";
					next;
				}
				else break;
			}
			delitem .@item_req,.@input * 2;
			getitem .@item_id,.@input;
			break;
		case 3:
			mes "[Jade]";
			mes "^FF0000D'accord, aucun problème.";
			mes "^FF0000N'hésitez pas à revenir quand vous voulez.";
			close;
		}
		mes "[Jade]";
		mes "Voilà.";
		mes "^FF0000Venez me voir de nouveau lorsque bon vous semble.";
		mes "^FF0000N'importe quel moment vous conviendra pour revenir.";
		mes "^FF0000Hm, que cherchez-vous à me demander avec cet air-là?";
		mes "^FF0000Il y a quelque chose sur mon visage?";
		close;
	}
}
