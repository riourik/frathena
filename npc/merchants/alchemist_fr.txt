alde_alche,24,188,3	script	Guild Dealer	740,{
	if (checkweight(1201,1) == 0) {
		mes "^FF0000- Actuellement, vous transportez -^";
		mes "^FF0000- Veuillez revenir plus tard -^";
		mes "- trop d'objets avec vous. -";
		mes "^FF0000- après avoir stocké certains objets dans le coffre de Kafra. -^";
		mes "[Nicholas Flamel]";
		close;
	}
	mes "[Gever Al Sharp]";
	mes "Bienvenue au";
	mes "Union des Alchimistes.";
	mes "Comment puis-je vous aider aujourd'hui?";
	next;
	switch(select("Purchase materials.:Purchase a production manual.:Quit.")) {
	case 1:
		mes "[Gever Al Sharp]";
		mes "Qu'est-ce que tu voudrais?";
		next;
		if (select("Medicine Bowl - 8 Zeny:Cancel.") == 1) {
			mes "[Gever Al Sharp]";
			mes "Combien en veux-tu ?";
			mes "Entrez '0' si vous souhaitez quitter.";
			next;
			while(1) {
				input .@input,0,2001;
				if (.@input == 0) {
					mes "[Gever Al Sharp]";
					mes "La transaction a été annulée.";
					mes "Revenez nous voir la prochaine fois.";
					close;
				}
				else if (.@input < 1 || .@input > 2000) {
					mes "[Gever Al Sharp]";
					mes "Le numéro doit";
					mes "être inférieur à 2000.";
					next;
				}
				else {
					break;
				}
			}
			set .@sell,.@input * 8;
			set .@item_weight,.@input * 10;
			if (Zeny < .@sell) {
				mes "[Gever Al Sharp]";
				mes "Vous n'avez pas assez de zeny.";
				mes "Vérifiez combien vous avez de zeny";
				mes "Vérifiez combien vous avez de zeny";
				mes "^FF0000tu as le premier.";
				close;
			}
			if ((MaxWeight - Weight) < .@item_weight) {
				mes "[Gever Al Sharp]";
				mes "Il semble que vous ne puissiez pas";
				mes "porter tout cela.";
				mes "Veuillez vérifier l'espace";
				mes "dans votre inventaire.";
				close;
			}
			set Zeny, Zeny-.@sell;
			getitem 7134,.@input; //Medicine_Bowl
			mes "[Gever Al Sharp]";
			mes "Merci.";
			mes "Revenez quand vous voulez.";
			close;
		}
		mes "[Gever Al Sharp]";
		mes "Bien alors,";
		mes "revenez lorsque";
		mes "vous avez besoin d'acheter";
		mes "des matériaux liés à la";
		mes "Alchimie, d'accord?";
		close;
	case 2:
		mes "[Gever Al Sharp]";
		mes "De quoi as-tu besoin?";
		mes "Les manuels coûtent généralement 100 000 zeny. Mais il y a quelques manuels spéciaux qui seront plus chers.";
		next;
		switch(select("Potion Creation Guide:Alcohol Creation Guide:Bottle Grenade Creation Guide:Acid Bottle Creation Guide:Plant Bottle Creation Guide:Marine Sphere Bottle Creation Guide:Glistening Coat Creation Guide:Condensed Potion Creation Guide:Cancel Deal.")) {
		case 1:
			callsub S_SellManual,7144,100000; //Normal_Potion_Book
			break;
		case 2:
			callsub S_SellManual,7127,100000; //Alcol_Create_Book
			break;
		case 3:
			callsub S_SellManual,7128,100000; //FireBottle_Create_Book
			break;
		case 4:
			callsub S_SellManual,7129,100000; //Acid_Create_Book
			break;
		case 5:
			callsub S_SellManual,7130,100000; //Plant_Create_Book
			break;
		case 6:
			callsub S_SellManual,7131,100000; //Mine_Create_Book
			break;
		case 7:
			callsub S_SellManual,7132,100000; //Coating_Create_Book
			break;
		case 8:
			callsub S_SellManual,7133,240000; //Slim_Potion_Create_Book
			break;
		case 9:
			mes "[Gever Al Sharp]";
			mes "Eh bien alors...";
			mes "revenez si vous";
			mes "avez besoin d'acheter";
			mes "un manuel de production.";
			close;
		}
	case 3:
		mes "[Gever Al Sharp]";
		mes "Très bien alors,";
		mes "passe une bonne journée.";
		close;
	}

S_SellManual:
	.@item_id = getarg(0);
	.@zeny_req = getarg(1);

	if (Zeny < .@zeny_req) {
		mes "[Gever Al Sharp]";
		mes "Vous n'avez pas assez de zeny.";
		mes "Vérifiez combien vous avez de zeny";
		mes "Vérifiez combien vous avez de zeny";
		mes "^FF0000tu as le premier.";
		close;
	}
	mes "[Gever Al Sharp]";
	mes getitemname(.@item_id) +"?";
	mes "That'll be "+ .@zeny_req +" zeny.";
	next;
	if (select("Purchase.:Quit.") == 1) {
		Zeny = Zeny - .@zeny_req;
		getitem .@item_id,1;
		mes "[Gever Al Sharp]";
		mes "Merci pour";
		mes "^FF0000votre parrainage.";
		close;
	}
	mes "[Gever Al Sharp]";
	mes "^FF0000revenez nous rendre visite.";
	mes "la prochaine fois.";
	close;
}
