airport,143,43,5	script	Airport Staff#airport1a::airport1	90,{
	mes "[Personnel de l'aéroport]";
	mes "Bienvenue au";
	mes "Aéroport d'Einbroch,";
	mes "où nous offrons non-stop";
	mes "vols vers les villes de";
	mes "Junon, Lighthalzen et Hugel.";
	next;
	if (select("Board the Airship:Cancel") == 1) {
		mes "[Personnel de l'aéroport]";
		mes "Les frais d'embarquement du dirigeable";
		mes "est de 1 200 zeny, mais si vous avez";
		mes "j'ai obtenu un billet gratuit pour un dirigeable,";
		mes "les frais seront supprimés. Volonté";
		mes "tu montes à bord du dirigeable ?";
		next;
		if (select("Yes:No") == 1) {
			if (countitem(7311) > 0) {
				delitem 7311,1; //Free_Flying_Ship_Ticket
				warp "airport",148,51;
				end;
			}
			if (Zeny >= 1200) {
				set Zeny, Zeny - 1200;
				warp "airport",148,51;
				end;
			}
			mes "[Personnel de l'aéroport]";
			mes "Je suis désolé, mais tu ne le fais pas";
			mes "avoir un billet gratuit pour";
			mes "Dirigeable et tu n'as pas";
			mes "assez zeny pour embarquer";
			mes "le dirigeable. Rappelez-vous, le";
			mes "les frais d'embarquement sont de 1 200 zeny.";
			close;
		}
	}
	mes "[Personnel de l'aéroport]";
	mes "Merci et";
	mes "passe une bonne journée.";
	close;
}

airport,143,49,3	script	Arrival Staff#airport2a::airport2	90,{
	mes "[Personnel d'arrivée]";
	mes "Bienvenue à l'aéroport d'Einbroch.";
	mes "Si vous arrivez de votre";
	mes "vol, laissez-moi vous guider vers le";
	mes "terminal principal. Sinon, veuillez monter à bord du dirigeable pour partir vers";
	mes "Junon, Lighthalzen et Hugel.";
	next;
	if (select("Exit to main terminal.:Cancel.") == 1) {
		mes "[Personnel d'arrivée]";
		mes "Une fois dans le terminal principal, vous devrez payer à nouveau les frais.";
		mes "monter à bord d'un dirigeable. Tu devrais";
		mes "sortie uniquement si la ville d'Einbroch";
		mes "est votre destination prévue.";
		mes "Continuer jusqu'au terminal principal ?";
		next;
		if (select("Yes:No") == 1) {
			warp "airport",142,40;
			end;
		}
	}
	mes "[Personnel d'arrivée]";
	mes "Très bien, merci";
	mes "pour votre mécénat";
	mes "et j'espère que tu as";
	mes "un vol agréable~";
	close;
}
