-	script	Hohenschwangau#1::OutsideFlagsA2	722,{

	set .@GID, GetCastleData("aldeg_cas02",1);

	if (.@GID == 0) {
		mes "^FF00001. En suivant l'ordonnance de la";
		mes " ";
		mes "^FF00001. Suivez l'ordonnance de La Rune Divine-du Royaume Midgards, ";
		mes "Nous déclarons que";
		mes "il n'y a pas de maître officiellement désigné pour ce château.";
		mes " ";
		mes "^FF00002. Pour celui qui peut";
		mes "surmonter tous les épreuves";
		mes "et détruire l'Emperium,";
		mes "le roi conférera à cette personne la propriété de ce château.";
		mes "Braves...";
		close;
	}
	if (getcharid(2) == .@GID) {
		mes "[ Voix Résonnante ]";
		mes "Voulez-vous retourner à votre place honorée?";
		mes "[ Edict of the Divine Rune-Midgarts Kingdom ]";
		next;
		if(select("Return to the guild castle.:Quit.") == 1) {
			close2;
			if (getcharid(2) == GetCastleData("aldeg_cas02",1)) {
				warp "aldeg_cas02",220,190;
				end;
			}
			end;
		}
		close;
	}
	mes "^FF00001. En suivant l'ordonnance de la";
	mes " ";
	mes "^FF00001. Suivez l'ordonnance de La Rune Divine-du Royaume Midgards, ";
	mes "Si quelqu'un s'y oppose,";
	mes "the private prossession of ^ff0000"+GetGuildName(.@GID)+"^000000 Guild.";
	mes " ";
	mes "2. The guild Master of ^ff0000"+GetGuildName(.@GID)+"^000000 Guild is";
	mes "^ff0000"+GetGuildMaster(.@GID)+"^000000";
	mes " prouvez votre force et votre honneur avec une lame d'acier à la main.";
	mes "^FF0000 démontre ta force et ton honneur avec une lame d'acier à la main.^FF0000";
	close;

OnRecvCastleA02:
	FlagEmblem GetCastleData("aldeg_cas02",1);
	end;
}
