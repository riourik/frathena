		initnpctimer;

	end;

	}
	end;

	if ($ep143_3_expbuff == 1)
		initnpctimer;
	end;

	if ($ep143_3_expbuff == 1)
		donpcevent "Guardian Nidhogg#143nyd01::OnEventOn";
	end;
