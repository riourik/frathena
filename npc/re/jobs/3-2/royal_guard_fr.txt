		if (.@input < 1 || .@input > 24) {
			mes "Définissez l'objet pour ajuster le Chevalier de Rune";
			mes "Vous ne pouvez définir qu'un nombre entre 1 et 24.";
		} else {
			set job_rune_edq, .@input;
			mes "Terminé";
		}
		close;
	case 3:
		mes "1~5";
		input .@input;
		next;
		if (.@input < 1 || .@input > 5) {
			mes "Définissez l'objet pour ajuster le Sorcier";
			mes "^FF0000[Guérisseur]: Vous ne pouvez définir qu'un nombre entre 1 et 5.";
		} else {
			set job_soc, .@input;
			mes "Terminé";
		}
		close;
	}
