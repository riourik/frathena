function	script	VIP_Third_Class	{
	if (!vip_status(VIP_STATUS_ACTIVE)) {
		mes "[Voix venue de nulle part]";
		mes "L'accès au contenu Premium est réservé à ceux qui possèdent une certaine quantité de Pierres de Réinitialisation.";
		next;
		if (countitem(6320) < 1) {
			mes "[Voix venue de nulle part]";
			mes "Pour passer en troisième classe, vous devez apporter 1 ^0000CCReset Stone^000000 ou avoir un compte Premium.";
			//mes "Veuillez visiter notre iShop pour acheter la ^0000CCReset Stone^000000.";
			close;
		} else {
			mes "[Voix venue de nulle part]";
			mes "Souhaitez-vous participer à cette quête contre 1 ^0000CCReset Stone^000000 ?";
			next;
			switch(select("Yes:No")) {
			case 1:
				mes "[Voix venue de nulle part]";
				mes "Bonne chance, aventurier.";
				delitem 6320,1; //Premium_Reset_Stone
				next;
				break;
			case 2:
				mes "[Voix venue de nulle part]";
				mes "Alors, laissez-moi";
				close;
			}
		}
	}
	return;
}
