ma_fild02,312,317,5	script	Guard#buwaya_cave	570,{
	if (BaseLevel < 130) {
		mes "[Garde]";
		mes "Les personnes sous ^ff0000niveau 130^000000";
		mes "ne peuvent pas entrer dans cet endroit.";
		mes "Cet endroit est dangereux. Veuillez retourner en arrière.";
		close;
	}

	set .@party_id,getcharid(1);
	set .@md_name$,"Buwaya Cave";

	if (!.@party_id) {
		mes "[Garde]";
		mes "Buwaya est dangereux.";
		mes "Veuillez revenir après avoir formé une";
		mes "équipe avec d'autres personnes.";
		close;
	}
	if (is_party_leader() == true) {
		mes "[Garde]";
		mes "Cet endroit est dangereux. Veuillez retourner en arrière.";
		mes "Si vous n'avez rien à faire ici, veuillez vous éloigner.";
		next;
		while(1) {
			switch(select("Why is it so dangerous?:I'm here to hunt down Buwaya.:End conversation.")) {
			case 1:
				mes "[Garde]";
				mes "Récemment, il y a eu un cas";
				mes "où des villageois ont disparu.";
				mes "Après avoir enquêté sur les traces,";
				mes "nous avons découvert que la monstruosité géante Buwaya";
				mes "en était responsable.";
				next;
				mes "[Garde]";
				mes "Des soldats et des mercenaires ont été dépêchés pour";
				mes "chasser Buwaya, mais sans jamais avoir réussi complètement.";
				mes "Buwaya enlève toujours les villageois à ce jour.";
				next;
				mes "[Garde]";
				mes "Veuillez rassembler quelques personnes pour éliminer Buwaya.";
				mes "Buwaya se cache toujours.";
				mes "Please gather some people to get rid of Buwaya.";
				next;
				break;
			case 2:
				set .@playtime, checkquest(4229,PLAYTIME);
				if (.@playtime == -1) {
					// fall through
				} else if (.@playtime == 0 || .@playtime == 1) {
					mes "[Garde]";
					mes "Buwaya se cache toujours.";
					mes "Même si vous entrez maintenant, Buwaya ne sortira pas.";
					mes "Veuillez revenir plus tard.";
					close;
				} else {
					erasequest 4229;
					// fall through
				}
				if (instance_create(.@md_name$) < 0) {
					mes "[Garde]";
					mes "Party name is... "+getpartyname(.@party_id)+".";
					mes "Party leader is... "+strcharinfo(0);
					mes "^0000ff"+.@md_name$+"^000000 cannot be opened now.";
					mes "Veuillez essayer un instant plus tard.";
					close;
				}
				mes "[Garde]";
				mes "I will open up the tunnel to ^0000ff"+.@md_name$+"^000000.";
				mes "^FF0000Je te souhaite bonne chance.";
				close;
			case 3:
				mes "[Garde]";
				mes "Cet endroit est dangereux. Vous ne pouvez pas entrer.";
				close;
			}
		}
	} else {
		mes "[Garde]";
		mes "Laissez-moi parler avec le chef de la partie.";
		mes "Veuillez faire apparaître le chef.";
		close;
	}
}

		mes "[Garde]";
		mes "Les personnes sous ^ff0000niveau 130^000000";
		mes "ne peuvent pas entrer dans cet endroit.";
		mes "Cet endroit est dangereux. Veuillez retourner en arrière.";
		close;
	}
	set .@playtime, checkquest(4229,PLAYTIME);
	if (.@playtime == -1) {
		// fall through
	} else if (.@playtime == 0 || .@playtime == 1) {
		mes "[Garde]";
		mes "Buwaya se cache toujours.";
		mes "Même si vous entrez maintenant, Buwaya ne sortira pas.";
		mes "Veuillez revenir plus tard.";
		close;
	} else {
		erasequest 4229;
		// fall through
	}
	switch(select("Enter.:Turn back.")) {
	case 1:
		switch(instance_enter("Buwaya Cave")) {
		case IE_OTHER:
			mes "[Garde]";
			mes "Oh, ce n'est pas le bon moment.";
			mes "Veuillez réessayer un instant plus tard.";
			close;
		case IE_NOINSTANCE:
		case IE_NOMEMBER:
			mes "[Garde]";
			mes "Cet endroit est dangereux. Veuillez ne pas entrer.";
			close;
		case IE_OK:
			mapannounce "ma_fild02",getpartyname(getcharid(1))+" party's "+strcharinfo(0)+" member began hunting Buwaya in Buwaya Cave.",bc_map,"0x00ff99"; //FW_NORMAL 12 0 0
			setquest 4229;
			//warp "1@ma_c",35,57;
			end;
		default:
			mes "- Impossible d'entrer en raison d'une force inconnue. -";
			close;
		}
	case 2:
		mes "[Garde]";
		mes "Vous devez valoriser votre vie.";
		mes "Il vaut mieux réfléchir avant d'agir.";
		close;
	}
}
