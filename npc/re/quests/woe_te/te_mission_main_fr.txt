		callsub S_Event, "alde0"+ charat( .@map$,getstrlen(.@map$)-1 ) + "::OnCommandOn";// #popa_alde0[1-5]::OnCommandOn

	.@map$ = strnpcinfo(4);

	.@string$ = getarg(0);

	donpcevent "#popb_"+ .@string$;
	donpcevent "#popc_"+ .@string$;
	donpcevent "#popd_"+ .@string$;
	end;
}

			mes "[Gardien] Le voleur a probablement profité des heures de siège.";
			mes "It's a request to find ^4d4dff"+ .@num_items +" or more of "+ .@string$[2] +"^000000.";
			callsub( S_Acceptquest, .@main_quest, .@npc_name$ );
		}
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "[Gardien] Étant donné que la région TE a une atmosphère plus décontractée,";
		mes "[Gardien] il n'est pas amusant de ne faire que des batailles de siège, non ?";
		mes "[Gardien] Alors, j'ai apporté ^4d4dfdes missions que vous pouvez effectuer pendant les heures de siège^000000~";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "[Gardien] Vous pouvez accepter des missions ^4d4dffà partir d'une heure avant le début du siège et jusqu'à la fin de la bataille de siège TE^000000,";
		mes "[Gardien] et chaque mission a une ^4d4dfdurée limite de 4 heures^000000 pour être complétée.";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "[Gardien] En simplifiant, vous acceptez une mission de moi pendant les heures de siège et";
		mes "[Gardien] ^4dfffrevenez me rendre compte dans le délai imparti^000000.";
		next;
		mes "["+ .@npc_name$ +"]";
		mes "[Gardien] Essayez-le pour le fun.";
		close;
	case 3:
		mes "["+ .@npc_name$ +"]";
		mes "Je vois.";
		mes "[Gardien] Revenez quand vous voulez si vous changez d'avis.";
		mes "[Gardien] Je ne serai peut-être pas disponible, cependant. Hehehe.";
		close;
	}

S_Cancelquest:
	.@npc_name$ = getarg(1);
	if (WoeTETimeStart(3600) == 1 || agitcheck3()) {
		mes "[Gardien] Il semble que vous n'ayez pas encore accompli la mission ?";
		mes getarg(2);
		next;
		mes "["+ .@npc_name$ +"]";
		mes "[Gardien] Voulez-vous l'annuler par hasard ?";
		next;
		switch( select( "No.","Yes." ) ) {
		case 1:
			mes "["+ .@npc_name$ +"]";
			mes "Bonne chance!";
			close;
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "Comme vous l'avez demandé, cette mission est annulée.";
			erasequest getarg(0);
			close;
		}
	}
	else {
		mes "[Gardien] Hmm... Vous avez encore un peu de temps, mais ce n'est pas encore l'heure du siège.";
		mes "[Gardien] Je ne pense pas que vous deviez continuer.";
		mes "[Gardien] Vous devriez annuler cette mission. Je dirais qu'elle est échouée.";
		next;
		switch( select( "I would like to go on.","Yes, I'll cancel it." ) ) {
		case 1:
			mes "["+ .@npc_name$ +"]";
			mes "Je ne pense pas que ça vaille le coup. Bon courage !";
			close;
		case 2:
			mes "["+ .@npc_name$ +"]";
			mes "Comme vous l'avez demandé, cette mission est annulée.";
			erasequest getarg(0);
			close;
		}
	}

S_Acceptquest:
	.@npc_name$ = getarg(1);
	next;
	switch( select( "Accept it.","Do not accept it." ) ) {
	case 1:
		mes "["+ .@npc_name$ +"]";
		mes "^4d4dffLe temps imparti est de 4 heures.^000000";
		mes "Accomplissez la mission dans ce délai et revenez m'en rendre compte.";
		setquest getarg(0);
		close;
	case 2:
		mes "["+ .@npc_name$ +"]";
		mes "Essayez une autre mission alors.";
		close;
	}

S_Rentbox:
	setarray .@item_id[0], 2950,15067,20709,22012;// Rune_Ring, Rune_Suit, Mana_Manteau, Mana_Boots
	for ( .@count = 0; .@count < getarg(0); .@count++ ) {
		.@miss_item01 = rand(100);
		if (.@miss_item01 > 3)
			mes "Vous n'avez rien trouvé.";
		else {
			rentitem .@item_id[ .@miss_item01 ],86400;
			mes "You found '"+ getitemname(.@item_id[ .@miss_item01 ]) +" !'";
		}
	}
	close;

S_Menu:
	for ( .@i = 1; .@i < getargcount(); .@i += 2 ) {
		if ( isbegin_quest(getarg(.@i)) )
			.@menu$ = .@menu$ + getarg(.@i+1) + " ^4d4dffCheck the result^000000:";
		else
			.@menu$ = .@menu$ + getarg(.@i+1) + ":";
		set getelementofarray( getarg(0),.@index ), getarg(.@i);
		.@index++;
	}
	return .@menu$;
}
