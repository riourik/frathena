function	script	Func_Socket	{
	.@item_id = getarg(0);
	.@zeny_req = getarg(4);
	setarray .@item_req[0], getarg(5), getarg(7,0);
	setarray .@req_amount[0], getarg(6), getarg(8,0);

	disable_items;
	mes "[Seiyablem]";
	mes "You want to add a Slot to a "+ getitemname(.@item_id) +"?";
	mes "Alright, please bring me ^FF0000"+ .@req_amount[0] +" "+ getitemname(.@item_req[0]) + (.@req_amount[0] > 1 && .@item_req[0] != 999 ? "s":"") +"^000000, "+ (.@item_req[1] != 0 && .@req_amount[1] != 0 ? ("^FF0000"+ .@req_amount[1] +" "+ getitemname(.@item_req[1]) + (.@req_amount[1] > 1 && .@item_req[1] != 999 ? "s":"") + "^000000 ") : "") + "and my ^FF0000" + (.@zeny_req >= 1000 ? (.@zeny_req /1000 +",000") : .@zeny_req) +",000 zeny^000000 service fee.";
	mes "Ah, and don't forget to bring that "+ getitemname(.@item_id) +"!";
	next;
	mes "[Seiyablem]";
	if (getiteminfo(.@item_id, ITEMINFO_LOCATIONS) & EQP_HAND_R) // weapon
	{
		mes "Je peux tenter d'ajouter un slot maintenant si tu as les matériaux nécessaires et le zeny requis.";
		mes "Cependant, il faut que tu saches qu'il y a une chance pour que j'échoue.";
		mes "Par conséquent, je dois te donner une précaution d'avertissement...";
		next;
		mes "[Seiyablem]";
		mes "Si cette tentative d'ajout de Slot à ton ^FF0000Arme^000000 échoue, alors l' ^FF0000Arme^000000, ainsi que les Cartes qu'elle contient seront détruites^000000.";
	}
	else // armor
	{
		mes "Si tu as tous les matériaux requis, ma rémunération en zeny et l'équipement, alors nous pouvons procéder à la tentative d'ajout de Slot.";
		mes "Mais avant cela, je dois te prévenir du risque.";
		next;
		mes "[Seiyablem]";
		mes "Si cette tentative d'ajout de Slot à ton ^FF0000Armure^000000 échoue, alors l' ^FF0000Armure^000000, ses améliorations et les Cartes qu'elle contient seront détruites^000000.";
	}
	mes "Veux-tu toujours essayer d'ajouter un Slot?";
	next;
	switch(select("Attempt Slot Addition:Cancel"))
	{
		case 1:
			if ((Zeny >= .@zeny_req*1000) && countitem(.@item_req[0]) >= .@req_amount[0] && countitem(getarg(7,512)) >= .@req_amount[1] && countitem(.@item_id) > 0)// if there's no getarg(7) and getarg(8) provided, check itemcount against 0 (we still need a valid item ID though)
			{
				mes "[Seiyablem]";
				mes "^FF0000Bien, commençons le travail !";
				mes "Mieux vaut prier pour un résultat réussi.";
				next;
				set .@a, rand(1,100);
				if ((.@a > getarg(2)) && (.@a < getarg(3)))
				{
					specialeffect getarg(3) == 51 ? EF_LORD : EF_SANCTUARY; // only S Class items have Lord of Vermillion effect
					mes "[Seiyablem]";
					mes "Génial, cela semble avoir fonctionné.";
					mes "Ça a l'air bien fait. Félicitations !";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					getitem getarg(1),1;
					next;
					mes "[Seiyablem]";
					mes "À la prochaine, mon pote !";
					close;
				}
				else
				{
					specialeffect EF_SUI_EXPLOSION;
					mes "[Seiyablem]";
					mes "Wah! ...Je suis vraiment désolé, ça a échoué.";
					mes "Cependant, je suis complètement innocent.";
					mes "C'est votre chance et c'est destinée par Dieu, d'accord ?";
					mes "Ne soyez pas si déçu et essayez la prochaine fois.";
					delitem .@item_id,1;
					delitem .@item_req[0],.@req_amount[0];
					if (.@item_req[1] != 0 && .@req_amount[1] != 0)
						delitem .@item_req[1],.@req_amount[1];
					Zeny = Zeny - .@zeny_req *1000;
					next;
					mes "[Seiyablem]";
					mes "Je vous souhaite bonne chance pour la prochaine fois !";
					close;
				}
			}
			else
			{
				mes "[Seiyablem]";
				mes "J'aimerais poursuivre cette tentative d'ajout de rangée, mais vous manquez quelques choses.";
				mes "Vous êtes sûr d'avoir l'équipement, les matériaux nécessaires et la zeny ?";
				close;
			}
		case 2:
			mes "[Seiyablem]";
			mes "Vous avez besoin de temps pour y réfléchir, hein ?";
			mes "D'accord, je comprends.";
			mes "^FF0000Rappelez-vous que la vie ne vaut pas le coup si vous n'osez jamais prendre de risques~";
			close;
	}
}
