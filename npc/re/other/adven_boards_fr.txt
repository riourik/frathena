function	script	F_AvenBoard	{
	.@mo_board = select( "Look up monsters by floor",
					"Monster Information Retrieval System",
					( countitem(909) == 30000 ? "Request" : "" ),
					"Cancel" );
	if (.@mo_board == 1)
		return 1;
	else if (.@mo_board == 2) {
		while(1) {
			mes "[Système de Récupération d'Informations sur les Monstres]";
			mes "Ceci est le Système de Récupération d'Informations sur les Monstres. Vous pouvez rechercher des informations sur les monstres qui apparaissent dans ce donjon. Lors de la recherche, entrez le nom du monstre, y compris les espaces.";
			next;
			input .@inputstr$;
			setarray .@element$[0],"Neutral","Water","Earth","Fire","Wind","Poison","Holy","Shadow","Ghost","Undead";
			setarray .@size$[0],"Medium","Small","Big";
			setarray .@race$[0],"Formless","Undead","Animal","Plant","Insect","Fish","Demon","Demihuman","Angel","Dragon";
			.@size = getargcount();
			.@i = 0;
			while( .@i < .@size && getarg(.@i+1,"") != .@inputstr$ ) .@i += 3;
			mes "[Système de Récupération d'Informations sur les Monstres]";
			mes "Recherche en cours...";
			next;
			mes "[Système de Récupération d'Informations sur les Monstres]";
			mes "La recherche est terminée.";
			mes "Les résultats de la recherche sont maintenant affichés.";
			next;
			mes "[Système de Récupération d'Informations sur les Monstres]";
			if (.@i >= .@size) {
				mes "Aucune information disponible pour ce monstre.";
				mes "Veuillez vérifier le nom du monstre.";
			}
			else {
				.@mob_id = getarg(.@i);
				mes "- Name : "+ getarg(.@i+1);
				mes "- Element : "+ .@element$[ getmonsterinfo(.@mob_id,MOB_ELEMENT) ];
				mes "- Size : "+ .@size$[ getmonsterinfo(.@mob_id,MOB_SIZE) ];
				mes "- Race : "+ .@race$[ getmonsterinfo(.@mob_id,MOB_RACE) ];
				mes "--------------------------------^FD8D02";
				mes getarg(.@i+2);
			}
			next;
			mes "[Système de Récupération d'Informations sur les Monstres]";
			mes "Souhaitez-vous rechercher d'autres informations sur des monstres ?";
			next;
			if (select( "Search","Cancel" ) == 1)
				continue;
			mes "[Système de Récupération d'Informations sur les Monstres]";
			mes "Le système de recherche de monstres va maintenant se fermer.";
			next;
			return 2;
		}
	}
	else if (.@mo_board == 3) {
		mes "Les informations demandées apparaîtront - Il n'y a aucun problème si elles apparaissent";
		next;
		return 3;
	}
	else if (.@mo_board == 4)
		close;
	else {
		mes "Erreur d'opération.";
		close;
	}
}

function	script	F_InfoBoard	{
	next;
	mes "[Groupe d'Aventuriers]";
	mes "Pour plus d'informations sur les monstres";
	mes "Système de Récupération des Informations sur les Monstres";
	mes "Doit être utilisé.";
	next;
	return;
}
