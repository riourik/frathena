prt_church,184,41,4	script	Cleric#aco	60,{
	if (Upper == 1) {
		if (ADVJOB == Job_High_Priest || ADVJOB == Job_Champion) {
			if (Class == Job_Novice_High) {
				mes "[Père Mareusis]";
				mes "Ah, je sens que vous avez vécu une vie antérieure.";
				mes "Vous devez avoir appris beaucoup de choses avant d'entrer dans Valhalla.";
				next;
				if (!callfunc("F_CanChangeJob")) {
					mes "[Père Mareusis]";
					mes "Malheureusement, je ne pense pas que vous soyez prêt à devenir un Acolyte encore. Veuillez terminer l'apprentissage des Compétences de Base d'abord.";
					next;
					mes "[Père Mareusis]";
					mes "En attendant,";
					mes "je vais attendre jusqu'à ce que vous soyez prêt.";
					mes "avec vous.";
					mes "Que Dieu soit";
					mes "[Père Mareusis]";
					close;
				}
				mes "[Père Mareusis]";
				mes "vous souhaite bonne chance pour votre";
				mes "nouveau voyage de vie.";
				mes "[Père Mareusis]";
				mes "Maintenant, partez en quête pour trouver ceux qui ont besoin de votre aide. Que Dieu éclaire votre chemin.";
				next;
				skill "NV_TRICKDEAD",0,SKILL_PERM;
				jobchange Job_Acolyte_High;
				skill "AL_HOLYLIGHT",1,SKILL_PERM;
				mes "[Père Mareusis]";
				mes "Maintenant, partez en quête pour trouver des gens qui ont besoin de votre aide. Que Dieu éclaire votre route.";
				close;
			}else{
				mes "[Père Mareusis]";
				mes "Maintenant, aventure-toi pour chercher ceux qui ont besoin de ton aide. Que Dieu éclaire ta voie.";
				close;
			}
		}else{
			mes "[Père Mareusis]";
			mes "Je sens que vous avez vécu une expérience de vie antérieure. Vous devriez avoir appris beaucoup de choses avant d'entrer dans Valhalla.";
			next;
			mes "[Père Mareusis]";
			mes "Cependant, je peux dire que vous n'êtes pas fait pour être un Acolyte. Souvenez-vous qui vous étiez dans votre vie antérieure et trouvez votre chemin.";
			close;
		}
	}
	mes "[Père Mareusis]";
	mes "Qu'est-ce que vous cherchez?";
	next;
	switch(select("Change your job to acolyte.:Ask the requirements to be an acolyte.:Quit it.")) {
	case 1:
		mes "[Père Mareusis]";
		if(BaseJob != Job_Novice) {
			if(BaseJob == Job_Acolyte)
				mes "Ça va aujourd'hui ? Je peux dire par votre habillement que vous êtes déjà un Acolyte. Vous ne plaisantez pas, n'est-ce pas ?";
			else
				mes "Je suis désolé, mais il semble que tu aies déjà ton propre métier, n'est-ce pas ?";
			close;
		}else{
			mes "Veux-tu vraiment devenir un serviteur de Dieu ?";
			mes "Voyons si tu es prêt ou non... Hmm...";
			next;
			if (!callfunc("F_CanChangeJob")) {
				mes "[Père Mareusis]";
				mes "Oh mon dieu ! Tu n'as pas encore accompli l'exercice de base ?! Il te reste beaucoup de chemin à parcourir ! Reviens après avoir augmenté ton niveau d'emploi !";
				close;
			}
			mes "[Père Mareusis]";
			mes "Hmm... ton niveau d'emploi est suffisant...";
			mes "Bien. Maintenant, je vais te donner la qualification pour devenir un Acolyte.";
			next;
			mes "[Père Mareusis]";
			mes "N'oublie jamais d'être reconnaissant envers Dieu, qui nous prend soin à tout moment. Dans le chaos et les moments de difficulté, affronte tes épreuves avec une foi inébranlable.";
			callfunc "Job_Change",Job_Acolyte;
			getitem 1545,1;	// N_Mace
			next;
			mes "[Père Mareusis]";
			mes "Enfin, je tiens sincèrement à vous féliciter pour avoir persévéré dans votre épreuve de pénitence.";
			close;
		}
	case 2:
		mes "[Père Mareusis]";
		mes "Veux-tu devenir un Acolyte ?";
		mes "Alors, tu dois accomplir les exigences suivantes en considérant qu'elles sont les pratiques données par Dieu.";
		next;
		mes "[Père Mareusis]";
		mes "D'abord, tu dois atteindre au moins le niveau d'emploi Novice 9 et apprendre toutes les Compétences de Base.";
		mes "C'est la chose la plus élémentaire à faire, donc tu dois y voir un moyen de te former toi-même.";
		next;
		mes "[Père Mareusis]";
		mes "Quand tu penses avoir rempli cette exigence, reviens alors vers moi. Ensuite, tu auras un métier sacré dans lequel tu pourras répandre la volonté de Dieu.";
		close;
	case 3:
		close;
	}
}
