//= rAthena Project
//===== Description: =========================================
//= Click the chicken and try retrieve the item at a low 
//= chance. If you fail he will nuke, freeze, stone, 
//= stun, or make you fall asleep.
//= The prize is configurable and triggered by the NPC.
//===== Additional Comments: =================================
//= 1.0 First release.
//= 1.1 Using 'switch rand' instead.
//= 1.2 Cleaned and standardized. [Euphy]
//= 1.2a Switched 'atcommand' to 'unitskilluseid'.
//= 1.3 Switched 'unitskilluseid' to 'unitkill'.
//============================================================

prontera,156,219,4	script	Cluckers	800,{

	set .@GMAccess,80;	// GM level required to access NPC

	if (.startcluck) {
		specialeffect2 EF_HIT3;
		switch(rand(15)) {
			case 0: npctalk "Cluckers : CLUUUUUUCK!!!"; unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; emotion ET_HUK; break;
			case 1: npctalk "Cluckers : Cluuuuck !~"; break;
			case 2: unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; break;
			case 3: sc_start SC_Freeze,10000,0; break;
			case 4: npctalk "Cluckers : CLUUUUUUUUUCK!!!"; unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; emotion ET_HUK; break;
			case 5: sc_start SC_Sleep,10000,0; break;
			case 6: sc_start SC_Stone,10000,0; emotion ET_KIK; break;
			case 7: npctalk "Cluckers : CLUUUUUUCK!!!"; unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; emotion ET_HUK; break;
			case 8: npctalk "Cluckers : Cocorico ! COCOORICO !!"; unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; emotion ET_HUK; break;
			case 9: sc_start SC_Stun,10000,0; break;
			case 10: sc_start SC_Sleep,10000,0; emotion ET_KIK; break;
			case 11: npctalk "Cluckers : Couac ! Couac !"; break;
			case 12: sc_start SC_Stun,10000,0; break;
			case 13: unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1; break;
			default:
				if (rand(50) < 4) {
					npctalk "Cluckers : OUIF !...........";
					specialeffect2 EF_SPHERE;
					announce "[Cluck! Cluck! Boom!] " + strcharinfo(0) + " squeezed out the prize! Well done!",0;
					getitem $cluck_item_id,$cluck_item_amount;
					set .startcluck,0;
				} else {
					npctalk "Cluckers : Cocorico ! COCOORICO !!";
					unitkill getcharid(3); skilleffect "NPC_SELFDESTRUCTION",1;
				}
				break;
		}
		end;
	}
	mes "[Crickets]";
	if (getgmlevel() >= .@GMAccess) {
		mes "Cluck cluck! Cluuuuuck? ^FF0000~Hi GM " + strcharinfo(0) + ", Wanna play today?~^000000";
		mes "Cocorico... COCORICO ! ^FF0000~Dites-moi juste ce que je dois faire !~^000000";
		next;
		while(1) switch(select("Start Event:Check Prize:Set Prize:Not today Cluckers")) {
			case 1:
				mes "[Crickets]";
				mes "CLUCK! ^FF0000~Bien sûr !~^000000";
				emotion ET_OK;
				close2;
				announce "[Cluck! Cluck! Boom!] va commencer à Prontera !",bc_blue;
				initnpctimer;
				end;
			case 2:
				mes "[Crickets]";
				mes "Cluck, cluck cluck... Cluck! ^FF0000~The current prize is^000000 ^008000"+ $cluck_item_amount +" "+ getitemname($cluck_item_id) +".^000000";
				next;
				break;
			case 3:
				mes "[Crickets]";
				mes "Cocorico ? Cocorico ??? ^FF0000~Quelle devrait être la récompense pour avoir gagné ? Veuillez entrer l'ID.~^000000";
				input $cluck_item_id;
				next;
				mes "[Crickets]";
				mes "Cliquetis ? Cluuuick ? ^FF0000~Combien de cet objet dois-je donner en échange ?~^000000";
				input $cluck_item_amount;
				next;
				mes "[Crickets]";
				mes "Cluck cluck..? Cluck. ^FF0000~So, the prize is^000000 ^008000"+ $cluck_item_amount +" "+ getitemname($cluck_item_id) +"^000000? ^FF0000Great.~^000000";
				emotion ET_OK;
				next;
				break;
			case 4:
				mes "[Crickets]";
				mes "Cocorico cocorico cocorico...";
				close;
		}
	}
	mes "Cocorico ! Cocooooorico ?";
	mes "Cocorico...";
	close;

OnTimer10000:
	announce "Vite derrière la fontaine si vous voulez jouer avec le poulet fou !",bc_blue;
	end;
OnTimer20000:
	announce "^FF0000Cluckers^a a mangé l'un de mes objets ! J'ai trop peur d'aller le récupérer !",bc_blue;
	end;
OnTimer30000:
	announce "Cliquez sur le poulet fou et essayez de faire sortir l'objet, si vous avez de la chance, vous gagnerez ! Êtes-vous prêt ?",bc_blue;
	end;
OnTimer40000:
	announce "Allez ! Cliquez sur le poulet pour obtenir le prix !",bc_blue;
	stopnpctimer;
	if (!$cluck_item_id) set $cluck_item_id,512;
	if (!$cluck_item_amount) set $cluck_item_amount,1;
	set .startcluck,1;
	end;
}
